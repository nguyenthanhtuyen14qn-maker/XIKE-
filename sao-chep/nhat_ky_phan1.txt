  }
  function isISO(s){
    return typeof s === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(s) && !isNaN(new Date(s + 'T00:00:00').getTime());

  function api(method, args){

      var id = 'g' + (++seq);

      }
