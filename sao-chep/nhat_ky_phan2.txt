  }

  function imgKey(t, idx){ return t + '_' + idx; }

  function imageNames(){
    return Object.keys(state.images).map(function(k){ return state.images[k]; })

  // ===== ĐỌC DỮ LIỆU ĐÃ LƯU =====
  function posNum(v, d){ v = Number(v); return isFinite(v) && v > 0 ? v : d; }
  function clampTier(v){ v = parseInt(v, 10); return v >= 1 && v <= TIER_COUNT ? v : 1; }
  function clampDays(v, d){ v = parseInt(v, 10); return v >= 1 ? Math.min(MAX_DAYS, v) : d; }
  function fit(list, n, clean){
    var out = [];
    for(var i=0; i<n; i++) out.push(clean(Array.isArray(list) ? list[i] : undefined));
    return out;
  }

  function ensureTierIn(s, t){
    if(!s.tierDays[t]){
      s.tierDays[t] = new Array(s.daysPerTier).fill(0);
      s.tierDayAmt[t] = new Array(s.daysPerTier).fill(0);
      s.tierDayDate[t] = new Array(s.daysPerTier).fill(null);
    }
  }
  function ensureTier(t){ ensureTierIn(state, t); }

  // Bậc hiện tại = bậc đầu tiên chưa ghi đủ ngày (giống checkTierCompletion của bản gốc)
  function highestTierOf(s){
    var highest = 1;
    for(var t = 1; t <= TIER_COUNT; t++){
      ensureTierIn(s, t);
      if(s.tierDays[t].every(function(v){ return v !== 0; }) && t < TIER_COUNT) highest = t + 1;
      else break;
    }
    return highest;
  }

  function normalizeState(src){
    if(src && src.cfg && Array.isArray(src.results)) return fromV4(src);
    var s = defaultState();
    if(!src || typeof src !== 'object') return s;
    if(isISO(src.startDate)) s.startDate = src.startDate;
    s.capital = posNum(src.capital, s.capital);
    s.percent = posNum(src.percent, s.percent);
    s.tpPrices = posNum(src.tpPrices, s.tpPrices);
    s.daysPerTier = clampDays(src.daysPerTier, s.daysPerTier);
    s.currentTier = clampTier(src.currentTier);
    s.viewTier = clampTier(src.viewTier || s.currentTier);
    var n = s.daysPerTier;
    for(var t=1; t<=TIER_COUNT; t++){
      var days = src.tierDays && src.tierDays[t];
      if(!Array.isArray(days)) continue;
      s.tierDays[t] = fit(days, n, function(v){ return v === 1 || v === -1 ? v : 0; });
      s.tierDayAmt[t] = fit(src.tierDayAmt && src.tierDayAmt[t], n, function(v){ return typeof v === 'number' && isFinite(v) ? v : 0; });
      s.tierDayDate[t] = fit(src.tierDayDate && src.tierDayDate[t], n, function(v){ return isISO(v) ? v : null; });
    }
    if(src.images && typeof src.images === 'object'){
      Object.keys(src.images).forEach(function(k){
        var v = src.images[k];
        if(/^\d+_\d+$/.test(k) && typeof v === 'string' && v) s.images[k] = v;
      });
    }
    return s;
  }

  // Dữ liệu của bản v4 trước (dạng cfg/results) đổi về dạng của bảng gốc
  function fromV4(src){
    var c = src.cfg || {};
    var s = defaultState();
    if(isISO(c.start)) s.startDate = c.start;
    s.capital = posNum(c.capital, s.capital);
    s.percent = posNum(c.pct, s.percent);
    s.tpPrices = posNum(c.tp, s.tpPrices);
    s.daysPerTier = clampDays(c.days, s.daysPerTier);
    var n = s.daysPerTier;
    src.results.forEach(function(r, i){
      if(!r || (r.r !== 'win' && r.r !== 'loss')) return;
      var t = Math.floor(i / n) + 1, idx = i % n;
      if(t > TIER_COUNT) return;
      ensureTierIn(s, t);
      s.tierDays[t][idx] = r.r === 'win' ? 1 : -1;
      s.tierDayAmt[t][idx] = Number(r.pnl) || 0;
      s.tierDayDate[t][idx] = isISO(r.date) ? r.date : null;
      if(typeof r.img === 'string' && r.img) s.images[imgKey(t, idx)] = r.img;
    });
    s.currentTier = highestTierOf(s);
    s.viewTier = s.currentTier;
    return s;
  }

  // ===== TÍNH TOÁN (giữ nguyên công thức bản gốc) =====
  function fmtFullDate(isoStr){
    if(!isoStr) return '';
    var parts = isoStr.split('-');
    if(parts.length === 3) return parts[2] + '/' + parts[1] + '/' + parts[0];
    return '';
  }

  function totalDaysLimit(){ return TIER_COUNT * state.daysPerTier; }
  function vonTier(t){ return state.capital * Math.pow(2, t-1); }
  function winTier(t){ return vonTier(t) * state.percent / 100; }

  function lotTier(t){
    var winAmt = winTier(t);
    var tp = state.tpPrices || 10;
    var lot = winAmt / (tp * 100);
    return Math.max(0.01, Math.round(lot * 100) / 100);
  }

  function fmtVND(n){ return Math.round(n).toLocaleString('vi-VN'); }
  function fmtLot(n){ return n.toLocaleString('vi-VN', {minimumFractionDigits:2, maximumFractionDigits:2}); }
  function fmtShort(n){
    if(n >= 1e6) return (Math.round((n/1e6)*10)/10) + 'M';
    if(n >= 1e3) return (Math.round((n/1e3)*10)/10).toString().replace('.0','') + 'K';
    return Math.round(n).toString();
  }

  function addDays(iso, n){
    if(!iso) iso = todayISO();
    var d = new Date(iso + 'T00:00:00');
    if(isNaN(d.getTime())) d = new Date();
    d.setDate(d.getDate() + n);
    return d.getFullYear() + '-' + pad(d.getMonth()+1) + '-' + pad(d.getDate());
  }

  function totalCompletedDays(){
    var sum = 0;
    for(var t=1; t<=TIER_COUNT; t++){
      ensureTier(t);
      state.tierDays[t].forEach(function(v){ if(v !== 0) sum++; });
    }
    return sum;
  }

  function computeTotals(){
    var profit = 0, maxLoss = 0;
    Object.keys(state.tierDayAmt).forEach(function(t){
      if(state.tierDayAmt[t] && parseInt(t, 10) <= TIER_COUNT){
        state.tierDayAmt[t].forEach(function(a){
          profit += a;
          if(a < 0 && Math.abs(a) > maxLoss) maxLoss = Math.abs(a);
        });
      }
    });
    return { profit: profit, maxLoss: maxLoss };
  }

      danger: true,

  async function applyDays(v){
    var snapshot = JSON.stringify(state);
    state.daysPerTier = v;
    state.tierDays = {}; state.tierDayAmt = {}; state.tierDayDate = {}; state.images = {};
    state.currentTier = 1; state.viewTier = 1;
    try {
      await saveState();
    } catch(e) {
      state = JSON.parse(snapshot);
      toast('Không lưu được: ' + e.message, 4000, true);
      writeInputs(); renderAll();
      return;
    }
    if(HAS_BRIDGE) api('imgPrune', { keep: [] }).catch(function(){});
    writeInputs(); renderAll();
  }

  // Đổi ảnh về JPEG (cạnh dài tối đa 2600 điểm ảnh, vẫn nét để đọc chữ trên ảnh chụp sàn)
  function readFile(file){
    return new Promise(function(resolve, reject){
      var reader = new FileReader();
      reader.onload = function(){ resolve(reader.result); };
      reader.onerror = function(){ reject(new Error('Không đọc được ảnh.')); };
      reader.readAsDataURL(file);
    });
  }

  function loadImage(src){
    return new Promise(function(resolve, reject){
      var img = new Image();
      img.onload = function(){ resolve(img); };
      img.onerror = function(){ reject(new Error('Không mở được ảnh này. Hãy chọn ảnh khác.')); };
      img.src = src;
    });
  }

  async function shrink(src, maxSide, quality){
    var img = await loadImage(src);
    var scale = Math.min(1, maxSide / Math.max(img.naturalWidth, img.naturalHeight));
    var canvas = document.createElement('canvas');
    canvas.width = Math.max(1, Math.round(img.naturalWidth * scale));
    canvas.height = Math.max(1, Math.round(img.naturalHeight * scale));
    var ctx = canvas.getContext('2d');
    ctx.fillStyle = '#000';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    return canvas.toDataURL('image/jpeg', quality);
  }

    try {
      // Lưu ảnh trước, xong mới ghi kết quả, để không có kết quả trỏ tới ảnh chưa lưu
      if(dlgImage.data) newRef = await Store.putImage(dlgImage.data, 't' + t + 'd' + idx + '-' + Date.now());
      state.tierDays[t][idx] = type;
      state.tierDayAmt[t][idx] = type === 1 ? Math.abs(amtVal) : -Math.abs(amtVal);
      state.tierDayDate[t][idx] = dateVal;
      if(newRef) state.images[key] = newRef;
      else if(dlgImage.removed) delete state.images[key];
      advanced = checkTierCompletion();
      await saveState();
    } catch(e) {
      state = JSON.parse(snapshot);
      if(newRef) Store.delImage(newRef);
      toast('Không lưu được: ' + e.message, 4000, true);
      return;
    } finally {
      busy = false;
      btn.disabled = false;
      btn.textContent = 'Lưu';
    }
    // Ảnh cũ chỉ xoá sau khi kết quả mới đã lưu xong
    if(oldRef && oldRef !== (state.images[key] || '')) Store.delImage(oldRef);
    closeDialog();
    renderAll();
    if(advanced){
      launchConfetti();
      var from = state.currentTier - 1;
      toast('🎉 Hoàn thành Bậc ' + from + '! Lên Bậc ' + state.currentTier + ': vốn $' + fmtVND(vonTier(state.currentTier)) + ', lô ' + fmtLot(lotTier(state.currentTier)) + '.', 6000);
    } else if(type === 1) {
      launchConfetti();
    }
  };

  window.clearDayResult = async function(){
    if (currentEditDayIdx === null || busy) return;
    var t = currentEditTier, idx = currentEditDayIdx;
    ensureTier(t);
    var key = imgKey(t, idx);
    var oldRef = state.images[key] || '';
    var btn = $('dlgClearBtn');
    // Ngày đã có kết quả hoặc ảnh thì bấm hai lần mới xoá
    if((state.tierDays[t][idx] !== 0 || oldRef) && !arm(btn, 'Bấm lần nữa')) return;

    state.tierDayAmt[t][idx] = 0;
    state.tierDayDate[t][idx] = null;

    checkTierCompletion();

      await saveState();

      state = JSON.parse(snapshot);
      toast('Không xoá được: ' + e.message, 4000, true);

  function toggleZoom(ev){
    ev.stopPropagation();
    var box = $('modalScroll');
    var img = $('modalImg');
    if(box.classList.contains('zoom')){
      box.classList.remove('zoom');
      img.style.width = '';
      $('modalHint').textContent = 'Chạm vào ảnh để phóng to';
      return;
    }
    var rect = img.getBoundingClientRect();
    var fx = rect.width ? (ev.clientX - rect.left) / rect.width : 0.5;
    var fy = rect.height ? (ev.clientY - rect.top) / rect.height : 0.5;
    box.classList.add('zoom');
    img.style.width = Math.round(Math.max(img.naturalWidth, box.clientWidth * 2.2)) + 'px';
    // Giữ chỗ vừa chạm ở giữa màn hình
    requestAnimationFrame(function(){
      box.scrollLeft = fx * img.offsetWidth - box.clientWidth / 2;
      box.scrollTop = fy * img.offsetHeight - box.clientHeight / 2;
    });
    $('modalHint').textContent = 'Vuốt để xem, chạm lần nữa để thu nhỏ';
  }
  $('modalImg').addEventListener('click', toggleZoom);

  var toastTimer = null;
  function toast(msg, ms, isErr){
    var t = $('toast');
    var overlay = ['resultDialog', 'ask', 'imgModal'].some(function(id){ return $(id).style.display === 'flex'; });
    var cls = 'toast' + (isErr ? ' err' : '') + (overlay ? ' top' : '');
    t.textContent = msg;
    t.className = cls + ' show';
    clearTimeout(toastTimer);
    toastTimer = setTimeout(function(){ t.className = cls; }, ms || 2600);
  }

  function hideToast(){
    clearTimeout(toastTimer);
    $('toast').className = 'toast';
  }

  // Bấm hai lần để xác nhận
  function arm(btn, label){
    if(btn.dataset.armed === '1') return true;
    btn.dataset.idle = btn.textContent;
    btn.dataset.armed = '1';
    btn.textContent = label;
    btn._timer = setTimeout(function(){ disarm(btn); }, 3000);
    return false;
  }

  function disarm(btn){
    clearTimeout(btn._timer);
    if(btn.dataset.armed === '1') btn.textContent = btn.dataset.idle;
    btn.dataset.armed = '';
  }

  function launchConfetti(){
    var canvas = $('confettiCanvas');
    var ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;

    var pieces = [];
    var count = 75;
    var colors = ['#ffd700', '#00e676', '#00e5ff', '#ff1744', '#ffffff'];

    for(var i=0; i<count; i++){
      pieces.push({
        x: canvas.width / 2,
        y: canvas.height / 2,
        vx: (Math.random() - 0.5) * 15,
        vy: (Math.random() - 0.7) * 18,
        size: Math.random() * 6 + 4,
        color: colors[Math.floor(Math.random() * colors.length)],
        rotation: Math.random() * 360,
        rSpeed: (Math.random() - 0.5) * 8,
        life: 1
      });
    }

    function animate(){
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      var alive = false;
      pieces.forEach(function(p){
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.45;
        p.rotation += p.rSpeed;
        p.life -= 0.015;

        if(p.life > 0){
          alive = true;
          ctx.save();
          ctx.translate(p.x, p.y);
          ctx.rotate((p.rotation * Math.PI) / 180);
          ctx.fillStyle = p.color;
          ctx.globalAlpha = p.life;
          ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
          ctx.restore();
        }
      });

      if(alive) requestAnimationFrame(animate);
      else ctx.clearRect(0, 0, canvas.width, canvas.height);
    }
    requestAnimationFrame(animate);
  }

  async function init(){
    var raw = '', note = '', parsed = null;
    try {
      raw = await Store.load();
    } catch(e) {
      loadFailed = true;
      note = 'Không đọc được dữ liệu: ' + e.message;
    }
    if(raw){
      try {
        parsed = JSON.parse(raw);
      } catch(e) {
        var bak = '';
        try { bak = await Store.loadBak(); } catch(e2) { bak = ''; }
        try {
          parsed = JSON.parse(bak);
          note = 'File dữ liệu bị lỗi, đã mở bản lưu dự phòng gần nhất.';
        } catch(e3) {
          parsed = null;
          note = 'File dữ liệu bị lỗi và không có bản dự phòng, đang mở bảng trống.';
        }
      }
    }
    state = normalizeState(parsed);
    ready = true;
    writeInputs();
    renderAll();
    // Dữ liệu dạng v4 trước đã đổi sang dạng gốc: ghi lại ngay
    if(parsed && parsed.cfg) saveQuiet();
    if(note) toast(note, 6000, true);
  }
