const pending = {};
let seq = 0;

function api(method, args) {
  return new Promise((resolve, reject) => {
    if (!window.AndroidBridge) {
      reject(new Error("Trang này chỉ chạy trong app AI XIKE."));
      return;
    }
    const id = "r" + ++seq;
    pending[id] = { resolve, reject };
    window.AndroidBridge.call(method, JSON.stringify(args || {}), id);
  });
}

window.__resolve = function (id, raw) {
  const p = pending[id];
  if (!p) return;
  delete pending[id];
  let r;
  try {
    r = JSON.parse(raw);
  } catch (e) {
    p.reject(new Error("Không đọc được phản hồi."));
    return;
  }
  if (r.ok) {
    p.resolve(r.data);
  } else {
    const err = new Error(r.error || "Lỗi không rõ");
    err.setup = !!r.setup;
    p.reject(err);
  }
};

function escapeHtml(s) {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
}

// Dọn ký hiệu Markdown còn sót, in đậm tiêu đề mục, tô màu BUY và SELL
function formatAI(text) {
  let s = escapeHtml(text || "");
  s = s.replace(/^#{1,6}\s*/gm, "");
  s = s.replace(/\*\*(.+?)\*\*/g, "<b>$1</b>");
  s = s.replace(/^\s*[*-]\s+/gm, "• ");
  s = s.replace(/^(\d+\.\s[^:\n<]{1,60}:)/gm, "<b>$1</b>");
  s = s.replace(/\b(BUY P|BUY)\b/g, '<span class="buy">$1</span>');
  s = s.replace(/\b(SELL P|SELL)\b/g, '<span class="sell">$1</span>');
  return s;
}

function pad(n) {
  return String(n).padStart(2, "0");
}

function fmtDate(value) {
  const d = new Date(value);
  if (isNaN(d.getTime())) return "";
  return pad(d.getDate()) + "/" + pad(d.getMonth() + 1) + "/" + d.getFullYear() + " " + pad(d.getHours()) + ":" + pad(d.getMinutes());
}

function showNote(id, text, kind) {
  const el = $(id);
  if (!text) {
    el.hidden = true;
    return;
  }
  el.textContent = text;
  el.className = "note " + (kind || "info");
  el.hidden = false;
}

function setBusy(btn, busy, label) {
  if (busy) {
    btn.dataset.label = btn.textContent;
    btn.textContent = label || "Đang xử lý…";
    btn.disabled = true;
  } else {
    if (btn.dataset.label) btn.textContent = btn.dataset.label;
    btn.disabled = false;
  }
}

// Bấm hai lần để xác nhận, thay cho hộp thoại
function confirmTap(btn, action, armedLabel) {
  if (btn.dataset.armed === "1") {
    clearTimeout(btn._timer);
    btn.dataset.armed = "";
    btn.textContent = btn.dataset.idle;
    action();
    return;
  }
  btn.dataset.idle = btn.textContent;
  btn.dataset.armed = "1";
  btn.textContent = armedLabel || "Bấm lần nữa để xác nhận";
  btn._timer = setTimeout(() => {
    btn.dataset.armed = "";
    btn.textContent = btn.dataset.idle;
  }, 3000);
}

function readFile(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result);
    reader.onerror = () => reject(new Error("Không đọc được ảnh."));
    reader.readAsDataURL(file);
  });
}

function loadImage(src) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error("Ứng dụng không đọc được ảnh này. Hãy chọn ảnh khác."));
    img.src = src;
  });
}

// Thu nhỏ ảnh về JPEG để gửi nhanh và đỡ tốn lượt
async function shrink(src, maxSide, quality) {
  const img = await loadImage(src);
  const scale = Math.min(1, maxSide / Math.max(img.naturalWidth, img.naturalHeight));
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(img.naturalWidth * scale));
  canvas.height = Math.max(1, Math.round(img.naturalHeight * scale));
  const ctx = canvas.getContext("2d");
  ctx.fillStyle = "#000";
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
  const dataUrl = canvas.toDataURL("image/jpeg", quality);
  return { dataUrl, b64: dataUrl.split(",")[1], mime: "image/jpeg" };
}

async function takeImage(input, maxSide, quality) {
  const file = input.files && input.files[0];
  input.value = "";
  if (!file) return null;
  const full = await shrink(await readFile(file), maxSide, quality);
  // Bản thu nhỏ cho bong bóng chat (cạnh dài 900px): đủ nét trên màn hình mật độ cao
  const thumb = await shrink(full.dataUrl, 900, 0.8);
  return { dataUrl: full.dataUrl, b64: full.b64, mime: full.mime, thumb: thumb.dataUrl };
}

// ===== VẼ PHÂN TÍCH LÊN ẢNH =====
// AI trả một khối <<<VE {json} VE>>> ở cuối. App bỏ khối đó khỏi chữ, rồi tự vẽ các mức lên ảnh gốc.
function extractDraw(text) {
  const src = String(text || "");
  let m = src.match(/<<<\s*VE\s*([\s\S]*?)\s*VE\s*>>>/);
  if (!m) m = src.match(/```(?:json)?\s*(\{[\s\S]*?"axis"[\s\S]*?\})\s*```/);
  if (!m) return { clean: src.trim(), data: null };
  const clean = (src.slice(0, m.index) + src.slice(m.index + m[0].length)).replace(/\n{3,}/g, "\n\n").trim();
  let raw = m[1].trim().replace(/^```(?:json)?/, "").replace(/```$/, "").trim();
  const a = raw.indexOf("{"), b = raw.lastIndexOf("}");
  if (a < 0 || b <= a) return { clean, data: null };
  try {
    return { clean, data: JSON.parse(raw.slice(a, b + 1)) };
  } catch (e) {
    return { clean, data: null };
  }
}

function numOrNull(v) {
  const n = Number(v);
  return v === null || v === "" || !Number.isFinite(n) || n <= 0 ? null : n;
}
function numList(v) {
  return (Array.isArray(v) ? v : [v]).map(numOrNull).filter((n) => n !== null).slice(0, 4);
}
function fmtPrice(n) {
  const parts = Number(n).toFixed(2).split(".");
  return parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ",") + "." + parts[1];
}

// Chia chữ thành các dòng vừa chiều rộng
function wrapText(cx, text, maxW) {
  const words = String(text).split(/\s+/).filter(Boolean);
  const lines = [];
  let line = "";
  for (const w of words) {
    const t = line ? line + " " + w : w;
    if (cx.measureText(t).width <= maxW || !line) line = t;
    else { lines.push(line); line = w; }
  }
  if (line) lines.push(line);
  return lines;
}

function roundRect(cx, x, y, w, h, r) {
  cx.beginPath();
  cx.moveTo(x + r, y);
  cx.arcTo(x + w, y, x + w, y + h, r);
  cx.arcTo(x + w, y + h, x, y + h, r);
  cx.arcTo(x, y + h, x, y, r);
  cx.arcTo(x, y, x + w, y, r);
  cx.closePath();
}

    cx.setLineDash(o.dash || []);
    const ang = Math.atan2(y1 - y0, x1 - x0), head = o.head || u * 0.9;
    cx.beginPath();
    cx.moveTo(x0, y0);
    cx.lineTo(x1 - Math.cos(ang) * head * 0.7, y1 - Math.sin(ang) * head * 0.7);

    const srs = res.map((p) => ({ p, up: true })).concat(sup.map((p) => ({ p, up: false })));
    for (const L of srs) {
      if (!inside(yOf(L.p))) { notShown.push((L.up ? "Kháng cự " : "Hỗ trợ ") + fmtPrice(L.p)); continue; }
      const color = L.up ? "#ef4444" : "#22c55e";
      hline(L.p, color, [u * 0.9, u * 0.55]);
      tag(L.p, color);
      const txt = [(L.up ? "Kháng cự (Resistance):" : "Hỗ trợ (Support):"), "~" + fmtPrice(L.p)];
      const sz = u * 0.95, hh = sz * 1.22 * 2;
      let y = L.up ? yOf(L.p) - hh - u * 0.35 : yOf(L.p) + u * 0.35;
      for (const r of srLabels) if (y < r + hh && y + hh > r) y = L.up ? r - hh - u * 0.2 : r + hh + u * 0.2;
      srLabels.push(y);
      const onRight = bias === "BUY" ? !L.up : L.up;
      let lx = onRight ? rightX : leftX;
      cx.font = "700 " + Math.round(sz) + "px " + FONT;
      let lw = Math.max(...txt.map((t) => cx.measureText(t).width));
      const fitX = (x, w) => Math.max(w / 2 + u * 0.3, Math.min(W - tagW - u * 0.5 - w / 2, x));   // không lấn cột giá
      lx = fitX(lx, lw);
      let out = txt;
      if (hits({ x: lx - lw / 2, y, w: lw, h: hh })) {
        // Chật: thử nhãn 1 dòng sát đường, không được thì đổi bên
        const one = [(L.up ? "Kháng cự " : "Hỗ trợ ") + "~" + fmtPrice(L.p)];
        const y1 = L.up ? yOf(L.p) - sz * 1.22 - u * 0.3 : yOf(L.p) + u * 0.3;
        const w1 = cx.measureText(one[0]).width;
        if (!hits({ x: lx - w1 / 2, y: y1, w: w1, h: sz * 1.22 })) { out = one; y = y1; }
        else lx = onRight ? leftX : rightX;
      }
      lw = Math.max(...out.map((t) => cx.measureText(t).width));
      lx = fitX(lx, lw);
      taken.push({ x: lx - lw / 2, y, w: lw, h: out.length * sz * 1.22 });
      srDraw.push(() => lines(out, lx, y, L.up ? "#f87171" : "#4ade80", sz, "center"));
    }
    // Kế hoạch lệnh: chỉ vẽ ngắn từ nến cuối ra mép phải
    const plan = [];
    if (entry !== null) plan.push({ p: entry, n: "Entry", c: "#f5b301" });
    if (sl !== null) plan.push({ p: sl, n: "SL", c: "#fb7185" });
    tps.forEach((p, i) => plan.push({ p, n: "TP" + (i + 1), c: "#86efac" }));
    for (const L of plan) {
      if (!inside(yOf(L.p))) { notShown.push(L.n + " " + fmtPrice(L.p)); continue; }
      if (res.concat(sup).some((q) => Math.abs(q - L.p) < 0.005)) continue;   // trùng đường hỗ trợ/kháng cự
      hline(L.p, L.c, [u * 0.25, u * 0.3], lastX + u * 0.5);
      tag(L.p, L.c, L.n + " " + Math.round(L.p));
    }

    // Mũi tên lớn chỉ hướng giá, từ nến cuối
    if (now !== null && (bias === "BUY" || bias === "SELL")) {
      const up = bias === "BUY";
      const y0 = yOf(now);
      const ahead = (q) => (up ? q > now : q < now) && Math.abs(yOf(q) - y0) >= u * 3;
      const near = (a, b) => Math.abs(a - now) - Math.abs(b - now);
      const cands = (up ? res : sup).filter(ahead).concat(tps.filter(ahead)).sort(near);
      let y1 = cands.length ? yOf(cands[cands.length > 1 ? 1 : 0]) + (up ? -1 : 1) * u * 2 : y0 + (up ? -1 : 1) * H * 0.12;
      y1 = Math.max(pTop + u, Math.min(pBot - u, y1));
      const x0 = lastX + u * 1.2, x1 = Math.min(W - tagW - u * 1.5, x0 + W * 0.3);
      if (x1 > x0 + u * 2) arrow(x0, y0 + (up ? -u : u), x1, y1, { width: Math.max(4, u * 0.38), dash: [u * 1.2, u * 0.8], head: u * 1.8 });
    }
    srDraw.forEach((f) => f());   // chữ hỗ trợ/kháng cự vẽ sau cùng, nằm trên đường và mũi tên
  }

  // Chú thích kèm mũi tên chỉ vào chi tiết trên biểu đồ (EMA, MACD, tín hiệu...)
  const callouts = Array.isArray(d.callouts) ? d.callouts.slice(0, 5) : [];
  for (const c of callouts) {
    if (!c || !c.text) continue;
    const px = fx(c.x, NaN), py = fy(c.y, NaN), tx = fx(c.tx, NaN), ty = fy(c.ty, NaN);
    if (![px, py, tx, ty].every(Number.isFinite)) continue;
    const sz = u * 0.9;
    cx.font = "700 " + Math.round(sz) + "px " + FONT;
    const arr = wrapText(cx, String(c.text).slice(0, 80), Math.min(W * 0.42, u * 13)).slice(0, 3);
    const tw = Math.max(...arr.map((t) => cx.measureText(t).width)), th = arr.length * sz * 1.22;
    // Tìm chỗ trống gần vị trí AI chọn: dời lên/xuống, hoặc sang trái khung nhận định; tránh khung và chữ khác
    const clampX = (v) => Math.max(tw / 2 + u * 0.3, Math.min(W - tw / 2 - u * 0.3, v));
    const clampY = (v) => Math.max(u * 0.3, Math.min(H - th - u * 0.3, v));
    const by0 = clampY(ty - th / 2), step = th + u * 0.4;
    const xs = [clampX(tx), clampX(box.x - tw / 2 - u * 0.4)];
    const ys = [by0];
    for (let k = 1; k <= 4; k++) ys.push(clampY(by0 + k * step), clampY(by0 - k * step));
    const rect = (x, y) => ({ x: x - tw / 2 - u * 0.2, y: y - u * 0.15, w: tw + u * 0.4, h: th + u * 0.3 });
    let bx = xs[0], by = by0, found = false;
    for (const yv of ys) {
      for (const xv of xs) {
        if (![box].concat(taken).some((r) => overlap(rect(xv, yv), r))) { bx = xv; by = yv; found = true; break; }
      }
      if (found) break;
    }
    if (!found) {   // không có chỗ trống hẳn: ít nhất không nằm dưới khung nhận định
      outer: for (const yv of ys) for (const xv of xs) if (!overlap(rect(xv, yv), box)) { bx = xv; by = yv; break outer; }
    }
    taken.push({ x: bx - tw / 2, y: by, w: tw, h: th });
    lines(arr, bx, by, "#ffffff", sz, "center");
    // Mũi tên từ mép khung chữ gần điểm cần chỉ nhất
    const sx = Math.max(bx - tw / 2, Math.min(bx + tw / 2, px)), sy = py < by ? by - u * 0.2 : py > by + th ? by + th + u * 0.2 : by + th / 2;
    const ssx = sy === by + th / 2 ? (px < bx ? bx - tw / 2 - u * 0.3 : bx + tw / 2 + u * 0.3) : sx;
    if (Math.hypot(px - ssx, py - sy) > u * 1.2) arrow(ssx, sy, px, py, { width: Math.max(2, u * 0.13), head: u * 0.8 });
  }

  // Vẽ khung nhận định (sau cùng, nằm trên mọi thứ)
  const extra = yOf ? notShown : res.map((p) => "Kháng cự " + fmtPrice(p)).concat(sup.map((p) => "Hỗ trợ " + fmtPrice(p)),
    entry !== null ? ["Entry " + fmtPrice(entry)] : [], sl !== null ? ["SL " + fmtPrice(sl)] : [], tps.map((p, i) => "TP" + (i + 1) + " " + fmtPrice(p)));
  cx.font = "600 " + Math.round(noteSz * 0.9) + "px " + FONT;
  const extraLines = extra.length ? wrapText(cx, (yOf ? "Ngoài khung: " : "") + extra.join(" · "), boxW - notePad * 2).slice(0, 3) : [];
  const stamp = "AI XIKE · chỉ để tham khảo" + (yOf ? "" : " · không đọc được trục giá");
  if (extraLines.length) {
    boxW = Math.max(boxW, Math.min(W * 0.54, u * 16.5));
    box.w = boxW;
    box.x = boxX();
  }
  box.h = notePad * 2 + noteLh * noteLines.length + noteSz * 1.15 * extraLines.length;
  cx.fillStyle = "rgba(0,0,0,.85)";
  roundRect(cx, box.x, box.y, box.w, box.h, u * 0.3);
  cx.fill();
  cx.strokeStyle = "rgba(255,255,255,.85)";
  cx.lineWidth = Math.max(1.5, u / 10);
  cx.stroke();
  cx.textAlign = "left";
  cx.textBaseline = "top";
  cx.font = "700 " + Math.round(noteSz) + "px " + FONT;
  cx.fillStyle = "#ffffff";
  let yy = box.y + notePad;
  cx.fillText("NHẬN ĐỊNH: ", box.x + notePad, yy);
  cx.fillStyle = bias === "BUY" ? "#4ade80" : bias === "SELL" ? "#f87171" : "#fde68a";
  cx.fillText(biasText || "—", box.x + notePad + cx.measureText("NHẬN ĐỊNH: ").width, yy);
  yy += noteLh;
  cx.font = "600 " + Math.round(noteSz * 0.9) + "px " + FONT;
  cx.fillStyle = "#fde68a";
  for (const t of extraLines) { cx.fillText(t, box.x + notePad, yy); yy += noteSz * 1.15; }
  cx.font = "600 " + Math.round(u * 0.55) + "px " + FONT;
  cx.textAlign = "right";
  cx.fillStyle = "rgba(255,255,255,.6)";
  cx.fillText(stamp, box.x + box.w, box.y + box.h + u * 0.25);

  return { dataUrl: cv.toDataURL("image/jpeg", 0.9), scaled: !!yOf };
}

async function send(analyze) {
  if (chatBusy) return;
  const note = $("msg").value.trim();
  if (analyze && !attached) {
    hint("Chọn ảnh biểu đồ bằng nút 📎 Ảnh trước, rồi bấm Phân tích.");
    return;
  }
  if (!note && !attached) {
    hint("Nhập câu hỏi hoặc chọn ảnh biểu đồ.");
    return;
  }
  hint("");
  const isAnalyze = analyze || (!note && !!attached);
  const text = isAnalyze ? ANALYZE_PROMPT + (note ? "\n\nGhi chú của tôi: " + note : "") : note;
  const image = attached;
  const entry = {
    role: "user",
    text: image ? text + "\n[Có gửi kèm ảnh biểu đồ]" : text,
    display: note || "📈 Phân tích biểu đồ",
    thumb: image ? image.thumb : "",
    full: image ? image.dataUrl : "",
    t: Date.now()
  };
  history.push(entry);
  const bubble = addToLog(messageEl(entry));
  $("msg").value = "";
  growComposer();
  clearAttach();
  setChatBusy(true);
  const waiting = addToLog(typingEl(isAnalyze ? "AI đang đọc biểu đồ…" : "AI đang trả lời…"));

  let extra = "";
  if ($("withJournal").checked) {
    try {
      extra = await journalContext();
    } catch (e) {
      extra = "";
    }
  }
  // Gửi tối đa 12 tin gần nhất, bắt đầu bằng tin của người dùng
  const recent = history.slice(-12).map((h) => ({ role: h.role, text: h.text }));
  while (recent.length && recent[0].role !== "user") recent.shift();

  try {
    const r = await api("chat", {
      history: recent,
      imageB64: image ? image.b64 : "",
      imageMime: image ? image.mime : "",
      analyze: isAnalyze,
      query: note || (isAnalyze ? ANALYZE_QUERY : text),
      extraContext: extra
    });
    const ex = extractDraw(r.text);
    const reply = { role: "model", text: ex.clean, model: r.model, rules: r.rules || [], note: r.note || "", t: Date.now() };
    if (image && ex.data) {
      try {
        const out = await annotateChart(image.dataUrl, ex.data);
        reply.ann = out.dataUrl;
        reply.annThumb = (await shrink(out.dataUrl, 900, 0.8)).dataUrl;
        if (!out.scaled) reply.note = (reply.note ? reply.note + "\n" : "") + "AI không đọc được trục giá nên chỉ ghi các mức trong khung nhận định.";
      } catch (err) {
        reply.note = (reply.note ? reply.note + "\n" : "") + "Không vẽ được lên ảnh: " + (err.message || "lỗi không rõ") + ".";
      }
    } else if (image && isAnalyze) {
      reply.note = (reply.note ? reply.note + "\n" : "") + "AI không trả số liệu để vẽ lên ảnh lần này.";
    }
    waiting.remove();
    history.push(reply);
    addToLog(messageEl(reply));
    saveChat();
  } catch (e) {
    waiting.remove();
    bubble.remove();
    history.pop();
    // Trả lại nội dung và ảnh để bấm gửi lại
    $("msg").value = note;
    growComposer();
    if (image) setAttach(image);
    addToLog(errorEl(e.message));
  } finally {
    setChatBusy(false);
  }
}

// Nhận gói dữ liệu từ biểu đồ (nút 🤖 AI), gửi số liệu cho AI, vẽ kết quả lên ảnh biểu đồ đúng thang giá
async function sendLive(snap) {
  if (chatBusy) { show("chat"); hint("AI đang trả lời câu trước, đợi xong rồi bấm lại."); return; }
  const note = $("msg").value.trim();
  const text = LIVE_PROMPT + (note ? "\n\nGhi chú của tôi: " + note : "") + "\n\nDỮ LIỆU:\n" + snap.text;
  let thumb = "";
  try { thumb = (await shrink(snap.image, 900, 0.8)).dataUrl; } catch (e) { thumb = ""; }
  const entry = { role: "user", text, display: "📈 Phân tích dữ liệu live " + snap.tf + (note ? " · " + note : ""), thumb, full: snap.image, t: Date.now() };
  history.push(entry);
  const bubble = addToLog(messageEl(entry));
  $("msg").value = "";
  growComposer();
  setChatBusy(true);
  const waiting = addToLog(typingEl("AI đang đọc dữ liệu " + snap.tf + "…"));
  let extra = "";
  if ($("withJournal").checked) {
    try { extra = await journalContext(); } catch (e) { extra = ""; }
  }
  const recent = history.slice(-12).map((h) => ({ role: h.role, text: h.text }));
  while (recent.length && recent[0].role !== "user") recent.shift();
  try {
    const r = await api("chat", {
      history: recent, imageB64: "", imageMime: "", analyze: true,
      query: note || ANALYZE_QUERY, extraContext: extra
    });
    const ex = extractDraw(r.text);
    const reply = { role: "model", text: ex.clean, model: r.model, rules: r.rules || [], note: r.note || "", t: Date.now() };
    if (ex.data) {
      // Chụp lại biểu đồ với thang giá đủ các mức AI đưa ra (nếu tab Biểu đồ còn mở)
      const now0 = numOrNull(ex.data.now) || snap.now;
      const lv = [now0].concat(numList(ex.data.resistance), numList(ex.data.support), numList(ex.data.tp),
        [numOrNull(ex.data.entry), numOrNull(ex.data.sl)].filter((x) => x !== null)).filter((x) => Math.abs(x - now0) / now0 < 0.08);
      const shot = (await reshoot(Math.min(...lv), Math.max(...lv))) || snap;
      // Thang giá và vị trí nến cuối lấy từ biểu đồ, không cần AI đọc ảnh
      const d = Object.assign({}, ex.data, { axis: shot.axis, last_x: shot.last_x, pane: shot.pane, callouts: [] });
      if (!numOrNull(d.now)) d.now = snap.now;
      try {
        const out = await annotateChart(shot.image, d);
        reply.ann = out.dataUrl;
        reply.annThumb = (await shrink(out.dataUrl, 900, 0.8)).dataUrl;
      } catch (err) {
        reply.note = (reply.note ? reply.note + "\n" : "") + "Không vẽ được lên biểu đồ: " + (err.message || "lỗi không rõ") + ".";
      }
    } else {
      reply.note = (reply.note ? reply.note + "\n" : "") + "AI không trả số liệu để vẽ lên biểu đồ lần này.";
    }
    waiting.remove();
    history.push(reply);
    addToLog(messageEl(reply));
    saveChat();
  } catch (e) {
    waiting.remove();
    bubble.remove();
    history.pop();
    $("msg").value = note;
    growComposer();
    addToLog(errorEl(e.message));
  } finally {
    setChatBusy(false);
    $("actAI").disabled = false;
    $("actAI").textContent = "🤖 AI";
    show("chat");
  }
}

// Nhờ khung biểu đồ chụp lại với thang giá [lo, hi]; trả null nếu tab Biểu đồ đã đóng hoặc quá 4 giây
let shotSeq = 0;
function reshoot(lo, hi) {
  const f = $("xauFrame");
  if (!f || !f.contentWindow || !Number.isFinite(lo) || !Number.isFinite(hi)) return Promise.resolve(null);
  const id = "s" + ++shotSeq;
  return new Promise((resolve) => {
    const done = (v) => { window.removeEventListener("message", on); clearTimeout(timer); resolve(v); };
    const on = (e) => {
      if (e.source !== f.contentWindow || !e.data || e.data.type !== "xk-shot-done" || e.data.id !== id) return;
      done(e.data.image && /^data:image\//.test(e.data.image) ? e.data : null);
    };
    const timer = setTimeout(() => done(null), 4000);
    window.addEventListener("message", on);
    f.contentWindow.postMessage({ type: "xk-shot", id, lo, hi }, "*");
  });
}

window.addEventListener("message", (e) => {
  const f = $("xauFrame");
  if (!f || e.source !== f.contentWindow || !e.data || e.data.type !== "xk-analyze") return;
  const d = e.data;
  if (typeof d.text !== "string" || typeof d.image !== "string" || !/^data:image\//.test(d.image)) return;
  sendLive(d);
});

function saveChat() {
  clearTimeout(saveTimer);
  saveTimer = setTimeout(() => {
    const items = history.slice(-40).map((h) => ({
      role: h.role,
      text: h.text,
      display: h.display || "",
      thumb: h.thumb || "",
      annThumb: h.annThumb || "",
      model: h.model || "",
      rules: h.rules || [],
      note: h.note || "",
      t: h.t || 0
    }));
    api("save", { name: "chat", data: JSON.stringify({ v: 1, items }) }).catch(() => {});
  }, 300);
}

async function loadChat() {
  try {
    const raw = await api("load", { name: "chat" });

      const data = JSON.parse(raw);
      if (data && Array.isArray(data.items)) history = data.items;
    }
  } catch (e) {
    history = [];
  }
  renderLog();
}

// Tóm tắt bảng gấp thếp ở tab Nhật Ký để AI nhận xét khi bạn bật ô "Nhật ký".
// Dữ liệu cùng dạng với file index.html gốc: tierDays (1 thắng, -1 thua, 0 chưa ghi), tierDayAmt, tierDayDate.
async function journalContext() {
  const raw = await api("load", { name: "gapthep" });
  if (!raw) return "";
  const data = JSON.parse(raw);
  const capital = Number(data.capital) || 1000;
  const pct = Number(data.percent) || 5;
  const tp = Number(data.tpPrices) || 10;
  const days = Math.max(1, Math.round(Number(data.daysPerTier) || 20));
  const currentTier = Math.min(10, Math.max(1, Number(data.currentTier) || 1));
  const tierDays = data.tierDays || {};
  const tierDayAmt = data.tierDayAmt || {};
  const tierDayDate = data.tierDayDate || {};
  const usd = (n) => Math.round(n).toLocaleString("vi-VN");
  const signed = (n) => (n >= 0 ? "+$" : "-$") + usd(Math.abs(n));
  const base = (t) => capital * Math.pow(2, t - 1);
  const winDay = (t) => (base(t) * pct) / 100;
  const lot = (t) => Math.max(0.01, Math.round((winDay(t) / (tp * 100)) * 100) / 100);

  let sum = 0;
  let wins = 0;
  let losses = 0;
  let maxLoss = 0;
  const done = [];
  for (let t = 1; t <= 10; t++) {
    const list = Array.isArray(tierDays[t]) ? tierDays[t] : [];
    list.forEach((v, i) => {
      if (v !== 1 && v !== -1) return;
      const amt = Number((tierDayAmt[t] || [])[i]) || 0;
      sum += amt;
      if (v === 1) wins++;
      else losses++;
      if (amt < maxLoss) maxLoss = amt;
      done.push({ t, i, v, amt, date: (tierDayDate[t] || [])[i] || "" });
    });
  }

  const lines = [
    "NHẬT KÝ GẤP THẾP XIKE GOLD CỦA NGƯỜI DÙNG (1 pip = 0.1 giá, 10 giá = 100 pip, 1 lot = 100 oz):",
    "Kế hoạch: vốn đầu $" + usd(capital) + ", mỗi lệnh " + pct + "% vốn của bậc, TP " + tp + " giá (" + tp * 10 +
      " pip), " + days + " ngày mỗi bậc, qua mỗi bậc vốn gấp đôi, tổng 10 bậc."
  ];
  const curList = Array.isArray(tierDays[currentTier]) ? tierDays[currentTier] : [];
  const pendingIdx = curList.findIndex((v) => v !== 1 && v !== -1);
  const nextDay = pendingIdx === -1 ? (curList.length < days ? curList.length + 1 : days) : pendingIdx + 1;
  let tierProfit = 0;
  curList.forEach((v, i) => {
    if (v === 1 || v === -1) tierProfit += Number((tierDayAmt[currentTier] || [])[i]) || 0;
  });
  lines.push(
    "Đang ở Bậc " + currentTier + ", Ngày " + nextDay + "/" + days + ": vốn bậc $" + usd(base(currentTier)) + ", lô " +
      lot(currentTier).toFixed(2) + ", mục tiêu " + signed(winDay(currentTier)) + " mỗi ngày. Vốn hiện tại của bậc $" +
      usd(base(currentTier) + tierProfit) + "."
  );
  const rate = wins + losses ? Math.round((wins / (wins + losses)) * 100) + "%" : "chưa có";
  lines.push(
    "Đã ghi " + done.length + " ngày: thắng " + wins + ", thua " + losses + ", tỉ lệ thắng " + rate +
      ". Lợi nhuận tích luỹ " + signed(sum) + ". Lỗ lớn nhất " + (maxLoss < 0 ? signed(maxLoss) : "chưa có") + "."
  );
  if (done.length) {
    lines.push("Các ngày gần nhất:");
    for (const d of done.slice(-15).reverse()) {
      const date = /^\d{4}-\d{2}-\d{2}$/.test(d.date) ? " (" + d.date.split("-").reverse().join("/") + ")" : "";
      lines.push("- Bậc " + d.t + " Ngày " + (d.i + 1) + date + ": " + (d.v === 1 ? "WIN " : "LOSS ") + signed(d.amt));
    }
  }
  return lines.join("\n");
}

let rulesLoaded = false;
let rules = [];
let editingId = "";
let ruleImage = null;

async function loadRules() {
  $("ruleStats").textContent = "";
  $("ruleList").innerHTML = '<p class="hint">Đang tải…</p>';
  try {
    const r = await api("listRules");
    rules = r.rules || [];
    rulesLoaded = true;
    // Từ 6 quy tắc trở xuống app gửi hết; nhiều hơn thì chọn 5 quy tắc gần câu hỏi nhất
    $("ruleStats").textContent = !rules.length
      ? ""
      : rules.length + " quy tắc, " + r.imageCount + " ảnh mẫu. " +
        (rules.length <= 6 ? "AI dùng tất cả quy tắc mỗi lần phân tích." : "AI chọn 5 quy tắc hợp nhất cho mỗi câu hỏi.");
    renderRules();
  } catch (e) {
    rulesLoaded = false;
    $("ruleList").innerHTML = "";
    const p = document.createElement("p");
    p.className = "note err";
    p.textContent = e.message;
    $("ruleList").appendChild(p);
    if (/Cài đặt/.test(e.message)) {
      const b = document.createElement("button");
      b.className = "link";
      b.textContent = "Mở Cài đặt";
      b.addEventListener("click", () => show("settings"));
      $("ruleList").appendChild(b);
    }
  }
}

function renderRules() {
  const list = $("ruleList");
  list.innerHTML = "";
  if (!rules.length) {
    const p = document.createElement("p");
    p.className = "hint";
    p.textContent = "Chưa có quy tắc nào. Viết quy tắc đầu tiên ở khung phía trên, ví dụ cách đọc chỉ báo EMA 34/80.";
    list.appendChild(p);
    return;
  }
  for (const rule of rules) {
    const item = document.createElement("div");
    item.className = "rule";
    const head = document.createElement("div");
    head.className = "rule-head";
    const title = document.createElement("b");
    title.textContent = rule.title;
    const meta = document.createElement("span");
    meta.className = "rule-meta";
    meta.textContent = (rule.images ? "🖼 " + rule.images + "  " : "") + fmtDate(rule.createdAt);
    head.appendChild(title);
    head.appendChild(meta);
    const body = document.createElement("div");
    body.className = "rule-body";
    body.textContent = rule.text.split("\n").slice(1).join("\n").trim() || rule.text;
    const actions = document.createElement("div");
    actions.className = "rule-actions";
    const more = document.createElement("button");
    more.className = "link";
    more.textContent = "Xem đủ";
    more.addEventListener("click", () => {
      const open = body.classList.toggle("open");
      more.textContent = open ? "Thu gọn" : "Xem đủ";
    });
    const edit = document.createElement("button");
    edit.className = "link";
    edit.textContent = "Sửa";
    edit.addEventListener("click", () => startEdit(rule));
    const del = document.createElement("button");
    del.className = "link danger";
    del.textContent = "Xoá";
    del.addEventListener("click", () => confirmTap(del, () => deleteRule(rule, del), "Bấm lần nữa để xoá"));
    // Chỉ hiện "Xem đủ" khi nội dung dài hơn 3 dòng
    const bodyText = body.textContent;
    if (bodyText.length > 120 || bodyText.split("\n").length > 3) actions.appendChild(more);
    actions.appendChild(edit);
    actions.appendChild(del);
    item.appendChild(head);
    item.appendChild(body);
    item.appendChild(actions);
    list.appendChild(item);
  }
}

function startEdit(rule) {
  editingId = rule.id;
  $("ruleText").value = rule.text;
  $("ruleFormTitle").textContent = "Sửa quy tắc";
  $("ruleSave").textContent = "Lưu thay đổi";
  $("ruleCancel").hidden = false;
  $("ruleImgNote").textContent = rule.images ? "Chọn ảnh mới sẽ thay ảnh mẫu cũ" : "Không bắt buộc";
  showNote("ruleMsg", "");
  $("view-rules").querySelector(".scroll").scrollTop = 0;
}

function resetRuleForm() {
  editingId = "";
  ruleImage = null;
  $("ruleText").value = "";
  $("ruleFormTitle").textContent = "Dạy quy tắc mới";
  $("ruleSave").textContent = "Lưu quy tắc";
  $("ruleCancel").hidden = true;
  $("ruleImgBox").hidden = true;
  $("ruleImg").removeAttribute("src");
  $("ruleImgNote").textContent = "Không bắt buộc";
}

async function saveRule() {
  const text = $("ruleText").value.trim();
  if (!text) {
    showNote("ruleMsg", "Nhập nội dung quy tắc trước.", "err");
    return;
  }
  const btn = $("ruleSave");
  setBusy(btn, true, "Đang lưu…");
  const wasEditing = !!editingId;
  try {
    const r = await api("saveRule", {
      id: editingId,
      text,
      imageB64: ruleImage ? ruleImage.b64 : "",
      imageMime: ruleImage ? ruleImage.mime : ""
    });
    setBusy(btn, false);
    resetRuleForm();
    if (r.warning) showNote("ruleMsg", r.warning, "warn");
    else showNote("ruleMsg", wasEditing ? "Đã lưu thay đổi." : "Đã lưu quy tắc.", "ok");
    loadRules();
  } catch (e) {
    setBusy(btn, false);
    showNote("ruleMsg", e.message, "err");
  }
}

async function deleteRule(rule, btn) {
  btn.disabled = true;
  btn.textContent = "Đang xoá…";
  try {
    await api("deleteRule", { id: rule.id });
    if (editingId === rule.id) resetRuleForm();
    loadRules();
  } catch (e) {
    btn.disabled = false;
    btn.textContent = "Xoá";
    showNote("ruleMsg", e.message, "err");
  }
}

function applySettings(s) {
  $("sUrl").value = s.supabaseUrl || "";
  $("sKey").value = "";
  $("sKey").placeholder = s.supabaseKeyHint ? "Đã lưu: " + s.supabaseKeyHint : "Dán khoá sb_secret_...";
  $("gKey").value = "";
  $("gKey").placeholder = s.geminiKeyHint ? "Đã lưu: " + s.geminiKeyHint : "Dán khoá Gemini";
  $("mMain").value = s.model || "";
  $("mFallback").value = s.fallbackModel || "";
  $("mEmbed").value = s.embedModel || "";
  $("cKey").value = "";
  $("cKey").placeholder = s.claudeKeyHint ? "Đã lưu: " + s.claudeKeyHint : "Dán khoá sk-ant-...";
  $("cModel").value = s.claudeModel || "";
  $("provider").value = s.provider === "claude" ? "claude" : "gemini";
}

async function loadSettings() {
  try {
    const s = await api("getSettings");
    applySettings(s);
    if (!s.hasGemini && !s.hasClaude) {
      showNote("setIntro", "Nhập khoá Gemini hoặc khoá Claude để bắt đầu. Khoá Supabase cần cho phần quy tắc.", "info");
      show("settings");
    }
  } catch (e) {
    showNote("setMsg", e.message, "err");
  }
}

async function saveSettings() {
  const btn = $("setSave");
  setBusy(btn, true, "Đang lưu…");
  try {
    const s = await api("saveSettings", {
      supabaseUrl: $("sUrl").value.trim(),
      supabaseKey: $("sKey").value.trim(),
      geminiKey: $("gKey").value.trim(),
      model: $("mMain").value.trim(),
      fallbackModel: $("mFallback").value.trim(),
      embedModel: $("mEmbed").value.trim(),
      claudeKey: $("cKey").value.trim(),
      claudeModel: $("cModel").value.trim(),
      provider: $("provider").value
    });
    applySettings(s);
    showNote("setIntro", "");
    showNote("setMsg", "Đã lưu. Bấm Kiểm tra kết nối để thử.", "ok");
    rulesLoaded = false;
  } catch (e) {
    showNote("setMsg", e.message, "err");
  } finally {
    setBusy(btn, false);
  }
}

async function runDiagnose() {
  const btn = $("setTest");
  setBusy(btn, true, "Đang kiểm tra…");
  showNote("setMsg", "");
  const list = $("diagList");
  list.innerHTML = "";
  $("diagCard").hidden = false;
  try {
    const r = await api("diagnose");
    for (const item of r.items) {
      const li = document.createElement("li");
      const icon = document.createElement("span");
      icon.className = item.ok ? "ok" : "bad";
      icon.textContent = item.ok ? "✓" : "✗";
      const text = document.createElement("div");
      text.textContent = item.name;
      const small = document.createElement("small");
      small.textContent = item.detail;
      text.appendChild(small);
      li.appendChild(icon);
      li.appendChild(text);
      list.appendChild(li);
    }
    const allOk = r.items.every((i) => i.ok);
    $("sqlCard").classList.toggle("attn", !!r.needSql);
    if (r.needSql) {
      showNote("setMsg", "Supabase cần tạo bảng hoặc cấp quyền. Sao chép SQL bên dưới, dán vào SQL Editor của Supabase, bấm Run, rồi kiểm tra lại.", "warn");
      $("sqlCard").scrollIntoView({ behavior: "smooth", block: "start" });
    } else if (allOk) {
      showNote("setMsg", "Mọi thứ đã sẵn sàng.", "ok");
    }
  } catch (e) {
    showNote("setMsg", e.message, "err");
  } finally {
    setBusy(btn, false);
  }
}
