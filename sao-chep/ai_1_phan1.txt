function aggregate(base, sec) {
  const out = [];
  let cur = null, key = null;
  for (let i = 0; i &lt; base.length; i++) {
    const b = base[i];
    const k = Math.floor(b.t / sec) * sec;
    if (k !== key) {
      if (cur) out.push(cur);
      key = k;
      cur = { t: k, o: b.o, h: b.h, l: b.l, c: b.c, v: b.v };
    } else {
      if (b.h &gt; cur.h) cur.h = b.h;
      if (b.l &lt; cur.l) cur.l = b.l;
      cur.c = b.c;
      cur.v += b.v;
    }
  }
  if (cur) out.push(cur);
  return out;
}

// Trung bình động số mũ, giá trị đầu là SMA của n ô đầu tiên có số (như ta.ema của TradingView).
function smoothSeed(a, n, al) {
  const N = a.length, out = NAN_ARR(N);
  let cnt = 0, sum = 0, prev = NaN;
  for (let i = 0; i &lt; N; i++) {
    const x = a[i];
    if (!Number.isFinite(x)) continue;
    if (Number.isFinite(prev)) {
      prev += al * (x - prev);
      out[i] = prev;
    } else {
      sum += x;
      cnt++;
      if (cnt === n) { prev = sum / n; out[i] = prev; }
    }
  }
  return out;
}
const ema = (a, n) =&gt; smoothSeed(a, n, 2 / (n + 1));
const rma = (a, n) =&gt; smoothSeed(a, n, 1 / n);

// Trung bình đơn trượt, ô nào trong cửa sổ thiếu số thì ô đó là NaN.
function smaArr(a, n) {
  const N = a.length, out = NAN_ARR(N);
  let sum = 0, bad = 0;
  for (let i = 0; i &lt; N; i++) {
    const x = a[i];
    if (Number.isFinite(x)) sum += x; else bad++;
    if (i &gt;= n) {
      const y = a[i - n];
      if (Number.isFinite(y)) sum -= y; else bad--;
    }
    if (i &gt;= n - 1 &amp;&amp; bad === 0) out[i] = sum / n;
  }
  return out;
}

// Max hoặc min trượt bằng hàng đợi đơn điệu, chạy trong O(N).
function rollExt(a, n, wantMax) {
  const N = a.length, out = NAN_ARR(N), dq = new Int32Array(N);
  let head = 0, tail = 0;
  for (let i = 0; i &lt; N; i++) {
    const x = a[i];
    while (tail &gt; head &amp;&amp; (wantMax ? a[dq[tail - 1]] &lt;= x : a[dq[tail - 1]] &gt;= x)) tail--;
    dq[tail++] = i;
    if (dq[head] &lt;= i - n) head++;
    if (i &gt;= n - 1) out[i] = a[dq[head]];
  }
  return out;
}
const rollMax = (a, n) =&gt; rollExt(a, n, true);
const rollMin = (a, n) =&gt; rollExt(a, n, false);

// Nhãn tín hiệu của EMA 34/80 theo script XÌ KE V8, tính trên nến đã đóng:
// cắt EMA có lọc MACD (BUY, SELL) và hồi về vùng EMA sau nb nến (BUY P, SELL P).
// Trả về [{ i, text, up }]: up = true thì nhãn nằm dưới nến, false thì nằm trên nến.
function zoneMarks(h, l, c, e34, e80, mac, nb, bt, st) {
  const out = [];
  let bCnt = 0, bLock = false, sCnt = 0, sLock = false;
  for (let i = 0; i &lt; c.length; i++) {
    const f = e34[i], s = e80[i], m = mac[i], hi = h[i], lo = l[i], cl = c[i];
    const pf = i &gt; 0 ? e34[i - 1] : NaN, ps = i &gt; 0 ? e80[i - 1] : NaN;
    const bull = f &gt; s, bear = f &lt; s, macdUp = m &gt; 0, macdDn = m &lt; 0;
    if (f &gt; s &amp;&amp; pf &lt;= ps &amp;&amp; macdUp) out.push({ i, text: &#x27;BUY&#x27;, up: true });
    if (f &lt; s &amp;&amp; pf &gt;= ps &amp;&amp; macdDn) out.push({ i, text: &#x27;SELL&#x27;, up: false });

    const bOK = bull &amp;&amp; cl &gt; s &amp;&amp; macdUp;
    const bTouch = bOK &amp;&amp; lo &lt;= f &amp;&amp; !bLock;
    const bNext = bCnt &gt; 0 ? (bOK ? bCnt + 1 : 0) : (bTouch ? 1 : 0);
    const bFinal = bCnt === nb - 1;
    const buyLive = bFinal &amp;&amp; lo &lt;= f &amp;&amp; hi &gt;= s &amp;&amp; !bLock &amp;&amp; m &gt; bt;
    if (buyLive) out.push({ i, text: &#x27;BUY P&#x27;, up: true });
    bCnt = bFinal ? 0 : bNext;
    bLock = buyLive ? true : ((lo &gt; f || !bull) ? false : bLock);

    const sOK = bear &amp;&amp; cl &lt; s &amp;&amp; macdDn;
    const sTouch = sOK &amp;&amp; hi &gt;= f &amp;&amp; !sLock;
    const sNext = sCnt &gt; 0 ? (sOK ? sCnt + 1 : 0) : (sTouch ? 1 : 0);
    const sFinal = sCnt === nb - 1;
    const sellLive = sFinal &amp;&amp; hi &gt;= f &amp;&amp; lo &lt;= s &amp;&amp; !sLock &amp;&amp; m &lt; st;
    if (sellLive) out.push({ i, text: &#x27;SELL P&#x27;, up: false });
    sCnt = sFinal ? 0 : sNext;
    sLock = sellLive ? true : ((hi &lt; f || !bear) ? false : sLock);
  }
  return out;
}

// RSI theo Wilder: RMA của tăng và giảm, ô đầu không có thay đổi nên bị bỏ qua.
function rsiArr(c, n) {
  const N = c.length, g = NAN_ARR(N), l = NAN_ARR(N), out = NAN_ARR(N);
  for (let i = 1; i &lt; N; i++) {
    const d = c[i] - c[i - 1];
    g[i] = d &gt; 0 ? d : 0;
    l[i] = d &lt; 0 ? -d : 0;
  }
  const ag = rma(g, n), al = rma(l, n);
  for (let i = 0; i &lt; N; i++) {
    const a = ag[i], b = al[i];
    if (!Number.isFinite(a) || !Number.isFinite(b)) continue;
    out[i] = b === 0 ? (a === 0 ? 50 : 100) : 100 - 100 / (1 + a / b);
  }
  return out;
}

// Stoch RSI PRO (theo script Pine của người dùng): K = SMA(Stoch(RSI, RSI, RSI, stochLen), kSmooth), D = SMA(K, dSmooth)

  const N = c.length, r = rsiArr(c, rl), hh = rollMax(r, sl), ll = rollMin(r, sl), raw = NAN_ARR(N);

}

function niceStep(range, targets) {
  if (!(range &gt; 0)) return 1;
  const raw = range / targets;
  const mag = Math.pow(10, Math.floor(Math.log10(raw)));
  const f = raw / mag;
  return (f &lt; 1.5 ? 1 : f &lt; 3 ? 2 : f &lt; 7 ? 5 : 10) * mag;
}

function nearestIndex(bars, t) {
  let lo = 0, hi = bars.length - 1;
  while (lo &lt; hi) {
    const mid = (lo + hi) &gt;&gt; 1;
    if (bars[mid].t &lt; t) lo = mid + 1;
    else hi = mid;
  }
  if (lo &gt; 0 &amp;&amp; Math.abs(bars[lo - 1].t - t) &lt;= Math.abs(bars[lo].t - t)) return lo - 1;
  return lo;
}

function baseStep(bars) {
  const d = [];
  for (let i = 1; i &lt; bars.length; i++) {
    const x = bars[i].t - bars[i - 1].t;
    if (x &gt; 0) d.push(x);
  }
  if (!d.length) return 60;

}

function parseTime(raw) {
  const s = String(raw).trim();
  if (/^\d{9,13}(\.\d+)?$/.test(s)) {
    let v = Number(s);
    if (v &gt; 1e11) v /= 1000;
    return Math.floor(v);
  }
  let m = s.match(/^(\d{4})[.\-\/](\d{1,2})[.\-\/](\d{1,2})(?:[ T]+(\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
  if (m) return Math.floor(Date.UTC(+m[1], +m[2] - 1, +m[3], +(m[4] || 0), +(m[5] || 0), +(m[6] || 0)) / 1000);
  m = s.match(/^(\d{1,2})[.\-\/](\d{1,2})[.\-\/](\d{4})(?:[ T]+(\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
  if (m) return Math.floor(Date.UTC(+m[3], +m[2] - 1, +m[1], +(m[4] || 0), +(m[5] || 0), +(m[6] || 0)) / 1000);
  return NaN;
}

function parseCSV(text) {
  const lines = String(text).replace(/^﻿/, &#x27;&#x27;).split(/\r?\n/).filter(l =&gt; l.trim() !== &#x27;&#x27;);
  if (!lines.length) return [];
  const delim = lines[0].indexOf(&#x27;\t&#x27;) &gt;= 0 ? &#x27;\t&#x27; : (lines[0].indexOf(&#x27;;&#x27;) &gt;= 0 ? &#x27;;&#x27; : &#x27;,&#x27;);
  let rows = lines.map(l =&gt; l.split(delim).map(c =&gt; c.trim().replace(/^&quot;|&quot;$/g, &#x27;&#x27;)));
  let head = null;
  if (rows[0].some(c =&gt; /^[&lt;&quot;]?[A-Za-z]/.test(c))) {
    head = rows[0].map(c =&gt; c.toLowerCase().replace(/[&lt;&gt;\s&quot;]/g, &#x27;&#x27;));
    rows = rows.slice(1);
  }
  if (!rows.length) return [];
  const find = names =&gt; {
    if (!head) return -1;
    for (const nm of names) {
      const k = head.indexOf(nm);
      if (k &gt;= 0) return k;
    }
    return -1;
  };
  let timeOf, cO, cH, cL, cC, cV = -1;
  if (head) {
    cO = find([&#x27;open&#x27;]); cH = find([&#x27;high&#x27;]); cL = find([&#x27;low&#x27;]); cC = find([&#x27;close&#x27;]);
    cV = find([&#x27;volume&#x27;, &#x27;tickvol&#x27;, &#x27;vol&#x27;]);   // MT5: VOL thường bằng 0 với CFD, TICKVOL mới có ý nghĩa
    if (cO &lt; 0 || cH &lt; 0 || cL &lt; 0 || cC &lt; 0) return [];
    const cDT = find([&#x27;datetime&#x27;, &#x27;timestamp&#x27;]), cD = find([&#x27;date&#x27;]), cT = find([&#x27;time&#x27;]);
    if (cDT &gt;= 0) timeOf = r =&gt; parseTime(r[cDT]);
    else if (cD &gt;= 0 &amp;&amp; cT &gt;= 0) timeOf = r =&gt; parseTime(r[cD] + &#x27; &#x27; + r[cT]);
    else {
      const k = cT &gt;= 0 ? cT : 0;
      timeOf = r =&gt; parseTime(r[k]);
    }
  } else {
    const r0 = rows[0];
    const isPair = /^\d{4}[.\-\/]\d{1,2}[.\-\/]\d{1,2}$/.test(r0[0] || &#x27;&#x27;) &amp;&amp; /^\d{1,2}:\d{2}/.test(r0[1] || &#x27;&#x27;);
    const off = isPair ? 2 : 1;
    cO = off; cH = off + 1; cL = off + 2; cC = off + 3;
    cV = r0.length &gt; off + 4 ? off + 4 : -1;
    timeOf = isPair ? (r =&gt; parseTime(r[0] + &#x27; &#x27; + r[1])) : (r =&gt; parseTime(r[0]));
  }
  const numOf = s =&gt; {
    let t = String(s == null ? &#x27;&#x27; : s).trim();
    if (t === &#x27;&#x27;) return NaN;
    if (delim !== &#x27;,&#x27;) t = t.replace(&#x27;,&#x27;, &#x27;.&#x27;);
    return Number(t);
  };
  const out = [];
  for (const r of rows) {
    if (r.length &lt;= cC) continue;
    const t = timeOf(r);
    const o = numOf(r[cO]), h = numOf(r[cH]), l = numOf(r[cL]), c = numOf(r[cC]);
    if (!Number.isFinite(t) || ![o, h, l, c].every(x =&gt; Number.isFinite(x))) continue;
    const v = cV &gt;= 0 ? numOf(r[cV]) : 0;
    out.push({ t, o, h: Math.max(h, o, c), l: Math.min(l, o, c), c, v: Number.isFinite(v) ? v : 0 });
  }
  out.sort((a, b) =&gt; a.t - b.t);
  const ded = [];
  for (const b of out) {
    if (ded.length &amp;&amp; ded[ded.length - 1].t === b.t) ded[ded.length - 1] = b;
    else ded.push(b);
  }
  return ded;
}

function mulberry32(seed) {
  let a = seed &gt;&gt;&gt; 0;
  return function () {
    a = (a + 0x6d2b79f5) &gt;&gt;&gt; 0;
    let t = a;
    t = Math.imul(t ^ (t &gt;&gt;&gt; 15), t | 1);
    t ^= t + Math.imul(t ^ (t &gt;&gt;&gt; 7), t | 61);
    return ((t ^ (t &gt;&gt;&gt; 14)) &gt;&gt;&gt; 0) / 4294967296;
  };
}

function makeDemo(n, start, stepSec, endSec) {
  const rnd = mulberry32(20261120);
  const out = [];
  let p = start;
  const t0 = endSec - (n - 1) * stepSec;
  for (let i = 0; i &lt; n; i++) {
    const amp = 0.35 + 0.5 * Math.abs(Math.sin(i / 260));
    const drift = 0.03 * Math.sin(i / 1400);
    const o = p;
    const c = o + (rnd() - 0.5) * 2 * amp + drift;
    const h = Math.max(o, c) + rnd() * amp * 0.7;
    const l = Math.min(o, c) - rnd() * amp * 0.7;
    out.push({ t: t0 + i * stepSec, o: r2(o), h: r2(h), l: r2(l), c: r2(c), v: Math.round(40 + rnd() * 260) });
    p = c;
  }
  return out;
}

function pickPeriod(tfSec, barW) {
  for (const P of PERIODS) {
    if (P &gt;= tfSec &amp;&amp; (P / tfSec) * barW &gt;= 85) return P;
  }
  return PERIODS[PERIODS.length - 1];
}

function timeAtIdx(bars, sec, idx) {
  const n = bars.length;
  if (idx &lt;= n - 1) return bars[clamp(Math.round(idx), 0, n - 1)].t;
  return bars[n - 1].t + Math.round(idx - (n - 1)) * sec;
}

function series(tf) {
  if (!cache[tf]) {
    const def = TFS.find(x =&gt; x.id === tf) || TFS[0];
    const bars = aggregate(BASE, def.sec);
    cache[tf] = { sec: def.sec, bars, hasVol: bars.some(b =&gt; b.v &gt; 0), ind: {} };
  }
  return cache[tf];
}

/* ----- Gann Fan + Sq9 tự động, theo script XÌ KE + Gann Fan + TRIX Trend + Sq9 Auto (Pine v5). ----- */
// Gốc là đáy khi TRIX cắt lên, đỉnh khi TRIX cắt xuống, lấy cực trị từ lần cắt trước đến nay.
// Như bản Pine, chỉ vẽ fan, TP, SL và vùng entry của gốc gần nhất, tính theo nến cuối cùng.
function gannAutoCalc(bars, p) {
  const N = bars.length, none = { lines: [], levels: [], info: &#x27;Chưa có giao cắt TRIX&#x27;, fit: true };
  if (N &lt; 2) return none;
  const c = closesOf(bars), h = highsOf(bars), l = lowsOf(bars);
  const tr = NAN_ARR(N);
  for (let i = 0; i &lt; N; i++) {
    tr[i] = i === 0 ? h[0] - l[0] : Math.max(h[i] - l[i], Math.abs(h[i] - c[i - 1]), Math.abs(l[i] - c[i - 1]));
  }
  const atr = rma(tr, 14);
  const tx = trixCalc(c, p.tl, p.ts), T = tx.t, SG = tx.s;
  let runHigh = NaN, runHighBar = NaN, runLow = NaN, runLowBar = NaN, anchor = NaN, anchorBar = NaN;
  for (let i = 0; i &lt; N; i++) {
    const t = T[i], s = SG[i], pt = i &gt; 0 ? T[i - 1] : NaN, ps = i &gt; 0 ? SG[i - 1] : NaN;
    const bull = t &gt; s &amp;&amp; pt &lt;= ps, bear = t &lt; s &amp;&amp; pt &gt;= ps, up = t &gt; s;
    const hi = h[i], lo = l[i];
    const hPrev = Number.isNaN(runHigh) ? hi : runHigh, lPrev = Number.isNaN(runLow) ? lo : runLow;
    const newHigh = Math.max(hPrev, hi), newHighBar = hi &gt;= hPrev ? i : runHighBar;
    const newLow = Math.min(lPrev, lo), newLowBar = lo &lt;= lPrev ? i : runLowBar;
    if (bull) { anchor = runLow; anchorBar = runLowBar; }
    else if (bear) { anchor = runHigh; anchorBar = runHighBar; }
    if (bull) { runHigh = hi; runHighBar = i; }
    else if (up) { runHigh = newHigh; runHighBar = newHighBar; }
    else { runHigh = NaN; runHighBar = NaN; }
    if (bear) { runLow = lo; runLowBar = i; }
    else if (up) { runLow = NaN; runLowBar = NaN; }
    else { runLow = newLow; runLowBar = newLowBar; }
  }
  if (Number.isNaN(anchor)) return none;
  const k = N - 1, a = atr[k], cl = c[k], t = T[k], s = SG[k];
  if (!Number.isFinite(a) || !(cl &gt; 0)) return { lines: [], levels: [], info: &#x27;Chưa đủ nến&#x27;, fit: true };
  const up = t &gt; s, dir = up ? 1 : -1;
  const inc = p.ai &gt; 0 ? a / cl * p.ai : p.sm / 1000;
  const unit = a * p.fs;
  const fan = (slope, color, w, dash) =&gt; {
    const v = NAN_ARR(N);
    for (let i = anchorBar; i &lt; N; i++) v[i] = anchor + dir * slope * (i - anchorBar);
    return { name: &#x27;Fan&#x27;, v, color, w, dash, nofit: true, legend: false };
  };
  const sq = Math.sqrt(anchor);
  const tp = j =&gt; { const x = up ? sq + j * inc : sq - j * inc; return x * x; };
  const sl = up ? anchor - a * p.sb : anchor + a * p.sb;
  const edge = up ? anchor + a * p.ez : anchor - a * p.ez;
  return {
    lines: [fan(unit, PAL.cyan, 2, []), fan(unit * 2, PAL.blue, 1, [5, 4]), fan(unit * 0.5, PAL.violet, 1, [5, 4])],
    levels: [
      { v: tp(1), text: &#x27;TP1&#x27;, color: PAL.orange, right: true },
      { v: tp(2), text: &#x27;TP2&#x27;, color: PAL.orange, right: true },
      { v: tp(3), text: &#x27;TP3&#x27;, color: PAL.orange, right: true },
      { v: sl, text: &#x27;SL&#x27;, color: PAL.red, right: true },
      { v: anchor, text: &#x27;Entry&#x27;, color: PAL.gold, right: true },
      { v: edge, text: &#x27;Biên entry&#x27;, color: PAL.gold, right: true }
    ],
    info: (up ? &#x27;TĂNG · Buy&#x27; : &#x27;GIẢM · Sell&#x27;) + &#x27; · inc &#x27; + inc.toFixed(4),
    fit: true
  };
}

function cleanParam(f, v) {
  const blank = v === null || v === undefined || (typeof v === &#x27;string&#x27; &amp;&amp; v.trim() === &#x27;&#x27;);
  const x = blank ? NaN : Number(v);

}

function sanitizeInds(arr) {

  if (nPanes) {
    paneH = clamp(Math.floor(totalH * 0.2), 56, 112);
    mainH = totalH - nPanes * paneH;
    if (mainH &lt; 130) {
      mainH = 130;
      paneH = Math.max(40, Math.floor((totalH - 130) / nPanes));
    }
  }
  const volH = volumeOn() ? Math.round(mainH * 0.16) : 0;
  return {
    w, h, axisR, axisB, plotW, totalH, mainH, paneH, nPanes, volH,
    candleTop: 8, candleBottom: mainH - volH - 6
  };
}

    cx.fillText(FAN_LABELS[k], LAY.plotW - 6, yr - 8);
  });
  cx.fillStyle = COL.fan;
  cx.beginPath(); cx.arc(x1, y1, 3, 0, Math.PI * 2); cx.fill();
}

function updateReadout(cur, prv) {
  const px = $(&#x27;px&#x27;), chg = $(&#x27;chg&#x27;);
  if (!px || !chg) return;
  px.textContent = cur.c.toFixed(2);
  px.style.color = cur.c &gt;= cur.o ? COL.up : COL.down;
  const d = cur.c - prv.c, pct = prv.c ? d / prv.c * 100 : 0;
  chg.textContent = (d &gt;= 0 ? &#x27;+&#x27; : &#x27;&#x27;) + d.toFixed(2) + &#x27;  (&#x27; + (pct &gt;= 0 ? &#x27;+&#x27; : &#x27;&#x27;) + pct.toFixed(2) + &#x27;%)&#x27;;
  chg.style.color = d &gt;= 0 ? COL.up : COL.down;
}

function readTokens() {
  const cs = getComputedStyle(document.documentElement);
  const g = n =&gt; cs.getPropertyValue(n).trim();
  for (const k of [&#x27;bg&#x27;, &#x27;panel&#x27;, &#x27;line&#x27;, &#x27;text&#x27;, &#x27;muted&#x27;, &#x27;accent&#x27;, &#x27;up&#x27;, &#x27;down&#x27;]) {
    COL[k] = g(&#x27;--&#x27; + k) || COL[k];
  }
  FONT.num = g(&#x27;--f-num&#x27;) || FONT.num;
  FONT.ui = g(&#x27;--f-ui&#x27;) || FONT.ui;
}

function makeHatch() {
  try {
    const c = document.createElement(&#x27;canvas&#x27;);
    c.width = 8; c.height = 8;
    const g = c.getContext(&#x27;2d&#x27;);
    g.strokeStyle = &#x27;rgba(239,100,113,0.24)&#x27;;
    g.lineWidth = 2;
    g.beginPath();
    g.moveTo(0, 8); g.lineTo(8, 0);
    g.moveTo(-2, 2); g.lineTo(2, -2);
    g.moveTo(6, 10); g.lineTo(10, 6);
    g.stroke();
    return cx.createPattern(c, &#x27;repeat&#x27;);
  } catch (err) {
    return null;
  }
}

function render() {
  syncAutoBtn();
  if (!cx) return;
  LAY = layout();
  const L = LAY;
  cx.setTransform(DPR, 0, 0, DPR, 0, 0);
  cx.fillStyle = COL.bg;
  cx.fillRect(0, 0, L.w, L.h);
  cx.textBaseline = &#x27;middle&#x27;;
  cx.font = smallFont();
  if (!hasData) { drawEmpty(); updateLegend(0); return; }

  const ser = series(S.tf), bars = ser.bars, n = bars.length;
  let iL = Math.max(0, Math.floor(idxAtX(0)));
  let iR = Math.min(n - 1, Math.ceil(idxAtX(L.plotW)));
  if (iR &lt; iL) { iL = Math.max(0, n - 50); iR = n - 1; }

  const ovs = overlayInsts().map(inst =&gt; ({ def: IND_BY_ID[inst.id], r: instResult(inst) }));
  const pns = paneInsts().map(inst =&gt; ({ def: IND_BY_ID[inst.id], r: instResult(inst) }));

  // Khoảng giá: nến, cộng thêm các đường và dải đang hiển thị
  let lo = Infinity, hi = -Infinity, vmax = 0;
  for (let i = iL; i &lt;= iR; i++) {
    const b = bars[i];
    if (b.l &lt; lo) lo = b.l;
    if (b.h &gt; hi) hi = b.h;
    if (b.v &gt; vmax) vmax = b.v;
  }
  for (const o of ovs) {
    for (const ln of o.r.lines) {
      if (ln.nofit) continue;
      for (let i = iL; i &lt;= iR; i++) {
        const v = ln.v[i];
        if (Number.isFinite(v)) {
          if (v &lt; lo) lo = v;
          if (v &gt; hi) hi = v;
        }
      }
    }
    if (o.r.fit) {
      for (const lv of o.r.levels || []) {
        if (Number.isFinite(lv.v)) {
          if (lv.v &lt; lo) lo = lv.v;
          if (lv.v &gt; hi) hi = lv.v;
        }
      }
    }
  }
  if (XK_FIT) { lo = Math.min(lo, XK_FIT.lo); hi = Math.max(hi, XK_FIT.hi); }
  const pad = hi &gt; lo ? (hi - lo) * 0.08 : 1;
  VP = (Y_LOCK &amp;&amp; !XK_FIT) ? { pMin: Y_LOCK.pMin, pMax: Y_LOCK.pMax } : { pMin: lo - pad, pMax: hi + pad };

  const cr = S.cross;
  const crInPlot = !!cr &amp;&amp; cr.x &gt;= 0 &amp;&amp; cr.x &lt;= L.plotW;
  const crInMain = crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.mainH;
  const li = crInPlot ? clamp(Math.round(idxAtX(cr.x)), 0, n - 1) : n - 1;
  const tags = [];   // nhãn có nền trên trục phải
  const axis = [];   // nhãn trục phải thường

  // Lưới giá
  const step = niceStep(VP.pMax - VP.pMin, 6);
  const dec = clamp(-Math.floor(Math.log10(step)), 0, 2);
  for (let k = Math.ceil(VP.pMin / step); k * step &lt;= VP.pMax; k++) {
    const y = Math.round(yOf(k * step)) + 0.5;
    cx.strokeStyle = COL.line; cx.lineWidth = 1;
    cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
    axis.push({ y: y, text: (k * step).toFixed(dec), color: COL.muted });
  }

  // Lưới thời gian và nhãn trục thời gian
  const P = pickPeriod(ser.sec, S.barW);
  const tzSec = S.tz * 3600;
  let lastDay = null;
  cx.textAlign = &#x27;center&#x27;;
  cx.fillStyle = COL.muted;
  for (let i = Math.max(1, iL); i &lt;= iR; i++) {
    const tt = bars[i].t + tzSec, tp = bars[i - 1].t + tzSec;
    if (Math.floor(tt / P) === Math.floor(tp / P)) continue;
    const day = Math.floor(tt / 86400);
    const newDay = day !== lastDay;
    lastDay = day;
    const x = xOf(i);
    if (x &lt; 0 || x &gt; L.plotW) continue;
    let label;
    if (P &lt; 86400) label = fmtTs(bars[i].t, newDay ? &#x27;dhm&#x27; : &#x27;hm&#x27;);
    else if (P &lt; 2592000) label = fmtTs(bars[i].t, &#x27;dm&#x27;);
    else { const d = new Date(tt * 1000); label = pad2(d.getUTCMonth() + 1) + &#x27;/&#x27; + d.getUTCFullYear(); }
    const xl = Math.round(x) + 0.5;
    cx.strokeStyle = COL.line; cx.lineWidth = 1;
    cx.beginPath(); cx.moveTo(xl, 0); cx.lineTo(xl, L.totalH); cx.stroke();
    cx.fillText(label, x, L.totalH + L.axisB / 2);
  }

  // Giờ cấm: cột chéo đỏ phủ cả vùng giá lẫn vùng chỉ báo
  cx.save();
  cx.beginPath(); cx.rect(0, 0, L.plotW, L.totalH); cx.clip();
  if (S.showBlock &amp;&amp; blockPat) {
    cx.fillStyle = blockPat;
    for (let i = iL; i &lt;= iR; i++) {
      if (inBlock(bars[i].t)) cx.fillRect(xOf(i) - S.barW / 2, 0, S.barW, L.totalH);
    }
  }
  cx.restore();

  // Vùng giá: khối lượng, dải tô, nến, đường, mức, hình vẽ, giá cuối, con trỏ
  cx.save();
  cx.beginPath(); cx.rect(0, 0, L.plotW, L.mainH); cx.clip();

  if (L.volH &gt; 8 &amp;&amp; vmax &gt; 0) {
    const bw = Math.max(1, S.barW * 0.7), base = L.mainH - 2, maxH = L.volH - 6;
    for (let i = iL; i &lt;= iR; i++) {
      const b = bars[i], h = b.v / vmax * maxH;
      cx.fillStyle = b.c &gt;= b.o ? COL.volUp : COL.volDn;
      cx.fillRect(xOf(i) - bw / 2, base - h, bw, h);
    }
  }

  for (const o of ovs) if (o.r.band) drawBand(o.r, iL, iR);

  const bodyW = Math.max(1, Math.round(S.barW * 0.7));
  for (let i = iL; i &lt;= iR; i++) {
    const bc = bars[i], xc = Math.round(xOf(i)), col = bc.c &gt;= bc.o ? COL.up : COL.down;
    cx.fillStyle = col; cx.strokeStyle = col; cx.lineWidth = 1;
    cx.beginPath(); cx.moveTo(xc + 0.5, yOf(bc.h)); cx.lineTo(xc + 0.5, yOf(bc.l)); cx.stroke();
    const yo = yOf(bc.o), yc = yOf(bc.c);
    cx.fillRect(xc - Math.floor(bodyW / 2), Math.min(yo, yc), bodyW, Math.max(1, Math.abs(yc - yo)));
  }

  for (const o of ovs) {
    for (const ln of o.r.lines) {
      cx.setLineDash(ln.dash || []);
      polyline(ln.v, Math.max(0, iL - 1), iR, yOf, ln.color, ln.w || (ln.thin ? 1 : 1.5));
      cx.setLineDash([]);
    }
    for (const lv of o.r.levels || []) {
      if (lv.v &lt; VP.pMin || lv.v &gt; VP.pMax) continue;
      const y = Math.round(yOf(lv.v)) + 0.5;
      cx.setLineDash([5, 4]); cx.strokeStyle = lv.color; cx.lineWidth = 1;
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      cx.fillStyle = lv.color; cx.textAlign = lv.right ? &#x27;right&#x27; : &#x27;left&#x27;;
      cx.fillText(lv.text + &#x27;  &#x27; + lv.v.toFixed(2), lv.right ? L.plotW - 8 : 8, y - 8);
      cx.textAlign = &#x27;left&#x27;;
    }
  }

  // Nhãn tín hiệu trên giá: BUY và BUY P nằm dưới nến, SELL và SELL P nằm trên nến; nhãn cùng nến được xếp chồng
  const stacked = {};
  for (const o of ovs) {
    for (const mk of o.r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      const key = mk.i + (mk.up ? &#x27;u&#x27; : &#x27;d&#x27;), k = stacked[key] || 0;
      stacked[key] = k + 1;
      const bc = bars[mk.i], xc = xOf(mk.i);
      if (mk.up) drawMark(xc, yOf(bc.l) + 6 + k * 16, mk.text, COL.up);
      else drawMark(xc, yOf(bc.h) - 20 - k * 16, mk.text, COL.down);
    }
  }

  for (const d of S.drawings) {
    if (d.type === &#x27;fan&#x27;) { drawFan(d, bars); continue; }
    if (TREND_TYPES.has(d.type)) { drawTrend(d, bars); continue; }
    if (d.type === &#x27;hline&#x27;) {
      const yh = yOf(d.p);
      if (yh &lt; L.candleTop || yh &gt; L.candleBottom) continue;
      const y = Math.round(yh) + 0.5;
      cx.strokeStyle = COL.hline; cx.lineWidth = 1; cx.setLineDash([6, 4]);
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      tags.push({ y: yh, text: d.p.toFixed(2), bg: COL.hline, fg: COL.bg });
    }
  }

  if (selValid() &amp;&amp; S.mode === &#x27;pan&#x27;) {
    cx.save();
    cx.shadowColor = DRAW_COL;
    cx.shadowBlur = 10;
    drawSelShape(selDraw, bars);
    cx.restore();
    drawSelTag(bars);
  }
  drawPending(bars);
  if (S.pendingFan) {
    const i = nearestIndex(bars, S.pendingFan.t);
    cx.fillStyle = COL.fan;
    cx.beginPath(); cx.arc(xOf(i), yOf(S.pendingFan.p), 4, 0, Math.PI * 2); cx.fill();
  }

  const last = bars[n - 1], lastUp = last.c &gt;= last.o;
  if (last.c &gt;= VP.pMin &amp;&amp; last.c &lt;= VP.pMax) {
    const yl = Math.round(yOf(last.c)) + 0.5;
    cx.strokeStyle = lastUp ? COL.up : COL.down; cx.lineWidth = 1; cx.setLineDash([2, 3]);
    cx.beginPath(); cx.moveTo(0, yl); cx.lineTo(L.plotW, yl); cx.stroke();
    cx.setLineDash([]);
    tags.push({ y: yOf(last.c), text: last.c.toFixed(2), bg: lastUp ? COL.up : COL.down, fg: COL.bg });
  }

  if (crInMain) {
    const yc = Math.round(cr.y) + 0.5;
    cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
    cx.setLineDash([]);
    if (cr.y &gt;= L.candleTop &amp;&amp; cr.y &lt;= L.candleBottom) {
      tags.push({ y: cr.y, text: priceAtY(cr.y).toFixed(2), bg: COL.box, fg: COL.text });
    }
  }
  cx.restore();

  // Các vùng chỉ báo dao động, mỗi vùng có thang riêng
  pns.forEach((pn, k) =&gt; {
    const def = pn.def, r = pn.r;
    const top = L.mainH + k * L.paneH, bot = top + L.paneH;
    let lo2 = Infinity, hi2 = -Infinity;
    for (const ln of r.lines) {
      for (let i = iL; i &lt;= iR; i++) {
        const v = ln.v[i];
        if (Number.isFinite(v)) {
          if (v &lt; lo2) lo2 = v;
          if (v &gt; hi2) hi2 = v;
        }
      }
    }
    for (const lv of def.levels) {
      if (lv.v &lt; lo2) lo2 = lv.v;
      if (lv.v &gt; hi2) hi2 = lv.v;
    }
    if (def.range) {
      lo2 = def.range[0];
      hi2 = def.range[1];
    } else {
      if (!Number.isFinite(lo2) || !Number.isFinite(hi2)) { lo2 = 0; hi2 = 1; }
      if (def.zero) { lo2 = Math.min(lo2, 0); hi2 = Math.max(hi2, 0); }
      if (hi2 - lo2 &lt; 1e-9) hi2 = lo2 + 1;
      const pd = (hi2 - lo2) * 0.1;
      lo2 -= pd;
      hi2 += pd;
    }
    const yP = v =&gt; top + 6 + (hi2 - v) / (hi2 - lo2) * (L.paneH - 12);

    cx.strokeStyle = COL.line; cx.lineWidth = 1;
    cx.beginPath(); cx.moveTo(0, Math.round(top) + 0.5); cx.lineTo(L.w, Math.round(top) + 0.5); cx.stroke();

    cx.save();
    cx.beginPath(); cx.rect(0, top, L.plotW, L.paneH); cx.clip();
    cx.setLineDash([4, 4]); cx.strokeStyle = COL.grid; cx.lineWidth = 1;
    for (const lv of def.levels) {
      const y = Math.round(yP(lv.v)) + 0.5;
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
    }
    cx.setLineDash([]);
    for (const ln of r.lines) {
      if (ln.type === &#x27;hist&#x27;) {
        const bw = Math.max(1, Math.round(S.barW * 0.6)), y0 = yP(0);
        for (let i = iL; i &lt;= iR; i++) {
          const v = ln.v[i];
          if (!Number.isFinite(v)) continue;
          const yv = yP(v);
          cx.fillStyle = v &gt;= 0 ? ln.up : ln.dn;
          cx.fillRect(Math.round(xOf(i)) - Math.floor(bw / 2), Math.min(y0, yv), bw, Math.max(1, Math.abs(yv - y0)));
        }
      } else {
        polyline(ln.v, Math.max(0, iL - 1), iR, yP, ln.color, ln.thin ? 1 : 1.4);
      }
    }
    // Nhãn Buy / Sell của TRIX: Buy nằm sát đáy ô, Sell nằm sát đỉnh ô
    for (const mk of r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      drawMark(xOf(mk.i), mk.up ? bot - 17 : top + 3, mk.text, mk.up ? COL.up : COL.down);
    }
    if (crInPlot &amp;&amp; cr.y &gt;= top &amp;&amp; cr.y &lt; bot) {
      const yc = Math.round(cr.y) + 0.5;
      cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
      cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
      cx.setLineDash([]);
      const vv = hi2 - (cr.y - top - 6) / (L.paneH - 12) * (hi2 - lo2);
      tags.push({ y: cr.y, text: fmtVal(vv, def.dec), bg: COL.box, fg: COL.text });
    }
    cx.restore();

    // Nhãn trục phải: các mức tham chiếu và mốc đẹp
    const pstep = niceStep(hi2 - lo2, 3);
    const pdec = clamp(-Math.floor(Math.log10(pstep)), 0, 4);
    const levelYs = [];
    for (const lv of def.levels) {
      const y = yP(lv.v);
      levelYs.push(y);
      axis.push({ y: y, text: fmtVal(lv.v, pdec), color: COL.muted });
    }
    for (let kk = Math.ceil(lo2 / pstep); kk * pstep &lt;= hi2; kk++) {
      const y = yP(kk * pstep);
      if (y &lt; top + 4 || y &gt; bot - 4) continue;
      if (levelYs.some(ly =&gt; Math.abs(ly - y) &lt; 12)) continue;
      axis.push({ y: y, text: (kk * pstep).toFixed(pdec), color: COL.muted });
    }
  });

  // Đường dọc của con trỏ, trên toàn khung
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const xl = Math.round(cr.x) + 0.5;
    cx.save();
    cx.beginPath(); cx.rect(0, 0, L.plotW, L.totalH); cx.clip();
    cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(xl, 0); cx.lineTo(xl, L.totalH); cx.stroke();
    cx.setLineDash([]);
    cx.restore();
  }
