  // Nhãn trục (ngoài vùng cắt xén)
  cx.textAlign = &#x27;left&#x27;;
  for (const t of axis) {
    cx.fillStyle = t.color;
    cx.fillText(t.text, L.plotW + 8, t.y);
  }
  for (const t of tags) {
    cx.fillStyle = t.bg; cx.fillRect(L.plotW, t.y - 9, L.axisR, 18);
    cx.fillStyle = t.fg; cx.fillText(t.text, L.plotW + 6, t.y);
  }
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const txt = fmtTs(timeAtIdx(bars, ser.sec, idxAtX(cr.x)), &#x27;dhm&#x27;);
    cx.font = smallFont();
    const tw = cx.measureText(txt).width + 14;
    const bx = clamp(cr.x - tw / 2, 0, Math.max(0, L.plotW - tw));
    cx.fillStyle = COL.box; cx.fillRect(bx, L.totalH, tw, L.axisB);
    cx.fillStyle = COL.text; cx.textAlign = &#x27;center&#x27;;

  }

  const cur = bars[li], prv = bars[li &gt; 0 ? li - 1 : li];
  updateReadout(cur, prv);
  updateLegend(li);
}

// Chú thích trên góc trái: nến đang trỏ tới và giá trị của từng chỉ báo. Chạm một dòng để chỉnh chỉ báo đó.
function updateLegend(li) {
  const box = $(&#x27;legend&#x27;);
  if (!box) return;
  if (!hasData) {
    if (legendKey !== &#x27;empty&#x27;) { box.textContent = &#x27;&#x27;; legendKey = &#x27;empty&#x27;; }
    return;
  }
  const key = [li, S.tf, dataGen, indsVer, S.showBlock, S.tz, VIEW.ohlc, VIEW.inds].join(&#x27;|&#x27;);
  if (key === legendKey) return;
  legendKey = key;
  const b = series(S.tf).bars[li];
  const up = b.c &gt;= b.o;
  const parts = [];
  if (VIEW.ohlc &amp;&amp; VIEW.inds) parts.push(&#x27;&lt;div class=&quot;lg&quot;&gt;&lt;b&gt;&#x27; + S.tf + &#x27;&lt;/b&gt; &#x27; + fmtTs(b.t, &#x27;dhm&#x27;) +
    (S.showBlock &amp;&amp; inBlock(b.t) ? &#x27; &lt;span class=&quot;blk&quot;&gt;Giờ cấm 18–22h VN&lt;/span&gt;&#x27; : &#x27;&#x27;) + &#x27;&lt;/div&gt;&#x27;);
  if (VIEW.ohlc &amp;&amp; VIEW.inds) parts.push(&#x27;&lt;div class=&quot;lg&quot;&gt;O &lt;b&gt;&#x27; + b.o.toFixed(2) + &#x27;&lt;/b&gt; H &lt;b&gt;&#x27; + b.h.toFixed(2) +
    &#x27;&lt;/b&gt; L &lt;b&gt;&#x27; + b.l.toFixed(2) + &#x27;&lt;/b&gt; C &lt;b style=&quot;color:&#x27; + (up ? COL.up : COL.down) + &#x27;&quot;&gt;&#x27; +
    b.c.toFixed(2) + &#x27;&lt;/b&gt;&lt;/div&gt;&#x27;);
  for (const inst of S.inds) {
    if (!inst.on || !VIEW.inds) continue;
    const def = IND_BY_ID[inst.id];
    let vals = &#x27;&#x27;;
    if (def.volume) {
      vals = &#x27;&lt;b&gt;&#x27; + fmtCompact(b.v) + &#x27;&lt;/b&gt;&#x27;;
    } else {
      const r = instResult(inst);
      const vs = r.lines.filter(ln =&gt; ln.legend !== false).map(ln =&gt; &#x27;&lt;span style=&quot;color:&#x27; + (ln.color || ln.up) + &#x27;&quot;&gt;&#x27; + fmtVal(ln.v[li], def.dec) + &#x27;&lt;/span&gt;&#x27;);
      if (r.info) vs.push(&#x27;&lt;span&gt;&#x27; + r.info + &#x27;&lt;/span&gt;&#x27;);
      vals = vs.join(&#x27; &#x27;);
    }
    parts.push(&#x27;&lt;div class=&quot;lg&quot; data-uid=&quot;&#x27; + inst.uid + &#x27;&quot;&gt;&lt;i class=&quot;sw&quot; style=&quot;background:&#x27; + def.color +
      &#x27;&quot;&gt;&lt;/i&gt;&lt;b&gt;&#x27; + def.label(inst.p) + &#x27;&lt;/b&gt; &#x27; + vals + &#x27;&lt;/div&gt;&#x27;);
  }
  // Mũi tên thu gọn / mở thông số chỉ báo (chỉ hiện khi đang bật ít nhất một chỉ báo)
  {
    const chev = VIEW.inds ? &#x27;M4 10 L8 6 L12 10&#x27; : &#x27;M4 6 L8 10 L12 6&#x27;;
    parts.push(&#x27;&lt;button type=&quot;button&quot; class=&quot;lg-tog&quot; data-tog=&quot;1&quot; aria-label=&quot;&#x27; + (VIEW.inds ? &#x27;Thu gọn thông số&#x27; : &#x27;Hiện thông số&#x27;) +
      &#x27;&quot;&gt;&lt;svg width=&quot;16&quot; height=&quot;16&quot; viewBox=&quot;0 0 16 16&quot; fill=&quot;none&quot; stroke=&quot;currentColor&quot; stroke-width=&quot;1.8&quot; stroke-linecap=&quot;round&quot; stroke-linejoin=&quot;round&quot;&gt;&lt;path d=&quot;&#x27; + chev + &#x27;&quot;/&gt;&lt;/svg&gt;&lt;/button&gt;&#x27;);
  }
  box.innerHTML = parts.join(&#x27;&#x27;);
}

/* ===== Quản lý chỉ báo: lưu, thêm, bỏ, và bảng chọn kiểu bottom sheet ===== */
function saveInds() {
  try {
    localStorage.setItem(IND_KEY, JSON.stringify(S.inds.map(x =&gt; ({ id: x.id, p: x.p, on: x.on }))));
  } catch (err) { /* không có bộ nhớ trình duyệt */ }
}

function loadInds() {
  try {
    const v = JSON.parse(localStorage.getItem(IND_KEY) || &#x27;null&#x27;);
    if (Array.isArray(v)) {
      // Thêm Stoch RSI PRO một lần cho danh sách chỉ báo đã lưu từ bản trước
      let added = false;
      try { added = localStorage.getItem(&#x27;xau-chart-srsi-added&#x27;) === &#x27;1&#x27;; } catch (err) { added = true; }
      if (!added &amp;&amp; !v.some(x =&gt; x &amp;&amp; x.id === &#x27;stoch_rsi&#x27;)) v.splice(Math.max(0, v.length - 1), 0, { id: &#x27;stoch_rsi&#x27; });
      try { localStorage.setItem(&#x27;xau-chart-srsi-added&#x27;, &#x27;1&#x27;); } catch (err) { /* bỏ qua */ }
      S.inds = sanitizeInds(v);
      return;
    }
  } catch (err) { /* bỏ qua dữ liệu hỏng */ }
  S.inds = defaultInds();
}

function touchInds() {
  indsVer += 1;
  saveInds();
  LAY = layout();
  updateIndBadge();
  if (!$(&#x27;libView&#x27;).hidden) renderActive();
  schedule();
}

function updateIndBadge() {
  const n = S.inds.filter(x =&gt; x.on).length;
  const bd = $(&#x27;indCount&#x27;);
  bd.textContent = String(n);
  bd.hidden = n === 0;
}

function addInd(id) {
  const def = IND_BY_ID[id];
  if (!def || !IND_ENABLED[id]) return;
  if (S.inds.length &gt;= MAX_INDS) { say(&#x27;Tối đa &#x27; + MAX_INDS + &#x27; chỉ báo. Hãy bỏ bớt trước khi thêm.&#x27;); return; }
  if (def.volume &amp;&amp; S.inds.some(x =&gt; x.id === &#x27;volume&#x27;)) { say(&#x27;Khối lượng đã có trên biểu đồ.&#x27;); return; }
  const p = {};
  for (const f of def.params) p[f.k] = f.def;
  S.inds.push({ uid: newUid(), id: def.id, p, on: true });
  touchInds();
  say(&#x27;Đã thêm &#x27; + def.label(p) + &#x27;.&#x27;);
  closeSheet();
}

function removeInd(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  S.inds = S.inds.filter(x =&gt; x.uid !== uid);
  touchInds();
  say(&#x27;Đã bỏ &#x27; + IND_BY_ID[inst.id].label(inst.p) + &#x27; khỏi biểu đồ.&#x27;);
  if (editUid === uid) showSheetView(&#x27;lib&#x27;);
}

  $(&#x27;libView&#x27;).hidden = !lib;

function openEditor(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  const def = IND_BY_ID[inst.id];
  editUid = uid;
  const box = $(&#x27;edFields&#x27;);
  box.textContent = &#x27;&#x27;;
  for (const f of def.params) {
    const lb = document.createElement(&#x27;label&#x27;);
    lb.textContent = f.label;
    const inp = document.createElement(&#x27;input&#x27;);
    inp.type = &#x27;number&#x27;;
    inp.inputMode = f.float ? &#x27;decimal&#x27; : &#x27;numeric&#x27;;
    inp.min = String(f.min);
    inp.max = String(f.max);
    inp.step = f.float ? &#x27;0.01&#x27; : &#x27;1&#x27;;
    inp.value = String(inst.p[f.k]);
    inp.dataset.k = f.k;
    lb.appendChild(inp);
    box.appendChild(lb);
  }
  $(&#x27;edNote&#x27;).textContent = def.desc + (def.params.length ? &#x27;&#x27; : &#x27; Chỉ báo này không có tham số để chỉnh.&#x27;);
  $(&#x27;edReset&#x27;).hidden = !def.params.length;
  showSheetView(&#x27;edit&#x27;, def.label(inst.p));
}

function saveEditor() {
  const inst = S.inds.find(x =&gt; x.uid === editUid);
  if (!inst) { showSheetView(&#x27;lib&#x27;); return; }
  const def = IND_BY_ID[inst.id];
  for (const inp of Array.from($(&#x27;edFields&#x27;).querySelectorAll(&#x27;input&#x27;))) {
    const f = def.params.find(x =&gt; x.k === inp.dataset.k);
    if (f) inst.p[f.k] = cleanParam(f, inp.value);
  }
  touchInds();
  say(&#x27;Đã áp dụng cài đặt &#x27; + def.label(inst.p) + &#x27;.&#x27;);
  showSheetView(&#x27;lib&#x27;);
}

function resetEditor() {
  const inst = S.inds.find(x =&gt; x.uid === editUid);
  if (!inst) return;
  const def = IND_BY_ID[inst.id];
  for (const inp of Array.from($(&#x27;edFields&#x27;).querySelectorAll(&#x27;input&#x27;))) {
    const f = def.params.find(x =&gt; x.k === inp.dataset.k);
    if (f) inp.value = String(f.def);
  }
}

function drawTrend(d, bars) {
  const P = trendPts(d, bars);
  if (!P.length) return;
  const K = 1000;
  cx.save();
  cx.strokeStyle = DRAW_COL;
  cx.fillStyle = DRAW_COL;
  cx.lineWidth = 1.5;
  switch (d.type) {
    case &#x27;tl&#x27;:
    case &#x27;info&#x27;:
    case &#x27;angle&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, P[1].x, P[1].y);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      if (d.type === &#x27;info&#x27;) {
        const dp = P[1].p - P[0].p;
        const pct = P[0].p ? dp / P[0].p * 100 : 0;
        const nBars = Math.abs(P[1].i - P[0].i);
        trendTag((P[0].x + P[1].x) / 2, (P[0].y + P[1].y) / 2 - 14, (dp &gt;= 0 ? &#x27;+&#x27; : &#x27;&#x27;) + dp.toFixed(2) + &#x27; (&#x27; + pct.toFixed(2) + &#x27;%) · &#x27; + nBars + &#x27; nến&#x27;);
      } else if (d.type === &#x27;angle&#x27;) {
        const deg = Math.atan2(-(P[1].y - P[0].y), Math.abs(P[1].x - P[0].x)) * 180 / Math.PI;
        trendTag(P[1].x, P[1].y - 14, deg.toFixed(1) + &#x27;°&#x27;);
      }
      break;
    }
    case &#x27;ext&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x - dx * K, P[0].y - dy * K, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      break;
    }
    case &#x27;ray&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x, P[0].y, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;hray&#x27;: {
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;vl&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      break;
    }
    case &#x27;cross&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      trendSeg(0, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;chan&#x27;: {
      if (P.length &lt; 3 || P[1].x === P[0].x) break;
      trendPlotClip();
      const m = (P[1].y - P[0].y) / (P[1].x - P[0].x);
      const yMain = x =&gt; P[0].y + m * (x - P[0].x);
      const dyOff = P[2].y - yMain(P[2].x);
      const yPar = x =&gt; yMain(x) + dyOff;
      const xL = -20, xR = LAY.plotW + 20;
      cx.beginPath();
      cx.moveTo(xL, yMain(xL)); cx.lineTo(xR, yMain(xR)); cx.lineTo(xR, yPar(xR)); cx.lineTo(xL, yPar(xL));
      cx.closePath();
      trendFill(0.12);
      trendSeg(xL, yMain(xL), xR, yMain(xR));
      trendSeg(xL, yPar(xL), xR, yPar(xR));
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    case &#x27;reg&#x27;: {
      if (P.length &lt; 2) break;
      const a = Math.min(P[0].i, P[1].i), b = Math.max(P[0].i, P[1].i);
      if (b - a &lt; 2) break;
      trendPlotClip();
      const n = b - a + 1;
      let sx = 0, sy = 0, sxy = 0, sxx = 0;
      for (let i = a; i &lt;= b; i++) { const x = i - a, y = bars[i].c; sx += x; sy += y; sxy += x * y; sxx += x * x; }
      const den = n * sxx - sx * sx;
      const slope = den ? (n * sxy - sx * sy) / den : 0;
      const icpt = (sy - slope * sx) / n;
      let ss = 0;
      for (let i = a; i &lt;= b; i++) { const r = bars[i].c - (icpt + slope * (i - a)); ss += r * r; }
      const sd = Math.sqrt(ss / n);
      const pAt = (i, k) =&gt; icpt + slope * (i - a) + k * sd;
      const edge = k =&gt; {
        cx.beginPath();
        for (let i = a; i &lt;= b; i++) { const x = xOf(i), y = yOf(pAt(i, k)); if (i === a) cx.moveTo(x, y); else cx.lineTo(x, y); }
      };
      edge(2);
      for (let i = b; i &gt;= a; i--) cx.lineTo(xOf(i), yOf(pAt(i, -2)));
      cx.closePath();
      trendFill(0.12);
      cx.save();
      cx.setLineDash([4, 3]);
      edge(2); cx.stroke();
      edge(-2); cx.stroke();
      cx.restore();
      edge(0); cx.stroke();
      break;
    }
    case &#x27;flat&#x27;: {
      if (P.length &lt; 3 || P[0].x === P[1].x) break;
      trendPlotClip();
      const x0 = Math.min(P[0].x, P[1].x), x1 = Math.max(P[0].x, P[1].x);
      const yTop = P[0].y, yBot = P[2].y;
      cx.beginPath();
      cx.rect(x0, Math.min(yTop, yBot), x1 - x0, Math.abs(yBot - yTop));
      trendFill(0.12);
      trendSeg(x0, yTop, x1, yTop);
      trendSeg(x0, yBot, x1, yBot);
      trendSeg(x0, yTop, x0, yBot);
      trendSeg(x1, yTop, x1, yBot);
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    default: break;
  }
  cx.restore();
}

/* ===== Xóa từng bản vẽ ở chế độ xem, và khôi phục ===== */
const DEL_TOL = 12;

function segDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len2 = dx * dx + dy * dy;
  const t = len2 ? clamp(((px - x0) * dx + (py - y0) * dy) / len2, 0, 1) : 0;
  return Math.hypot(px - (x0 + t * dx), py - (y0 + t * dy));
}
function rayDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len2 = dx * dx + dy * dy;
  if (!len2) return Math.hypot(px - x0, py - y0);
  const t = ((px - x0) * dx + (py - y0) * dy) / len2;
  if (t &lt;= 0) return Math.hypot(px - x0, py - y0);
  return Math.hypot(px - (x0 + t * dx), py - (y0 + t * dy));
}
function lineDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len = Math.hypot(dx, dy);
  if (!len) return Math.hypot(px - x0, py - y0);
  return Math.abs(dx * (py - y0) - dy * (px - x0)) / len;
}
function regDist(P, bars, px, py) {
  const a = Math.min(P[0].i, P[1].i), b = Math.max(P[0].i, P[1].i);
  if (b - a &lt; 2) return Infinity;
  const n = b - a + 1;
  let sx = 0, sy = 0, sxy = 0, sxx = 0;
  for (let i = a; i &lt;= b; i++) { const x = i - a, y = bars[i].c; sx += x; sy += y; sxy += x * y; sxx += x * x; }
  const den = n * sxx - sx * sx;
  const slope = den ? (n * sxy - sx * sy) / den : 0;
  const icpt = (sy - slope * sx) / n;
  let ss = 0;
  for (let i = a; i &lt;= b; i++) { const r = bars[i].c - (icpt + slope * (i - a)); ss += r * r; }
  const sd = Math.sqrt(ss / n);
  let best = Infinity;
  for (const k of [-2, 0, 2]) {
    for (let i = a; i &lt; b; i++) {
      const ya = yOf(icpt + slope * (i - a) + k * sd), yb = yOf(icpt + slope * (i + 1 - a) + k * sd);
      best = Math.min(best, segDist(px, py, xOf(i), ya, xOf(i + 1), yb));
    }
  }
  return best;
}
function drawingDist(d, bars, px, py) {
  if (d.type === &#x27;hline&#x27;) {
    const yh = yOf(d.p);
    return yh &lt; LAY.candleTop || yh &gt; LAY.candleBottom ? Infinity : Math.abs(py - yh);
  }
  if (d.type === &#x27;fan&#x27;) {
    const i1 = nearestIndex(bars, d.p1.t), i2 = nearestIndex(bars, d.p2.t);
    if (i1 === i2 || xOf(i1) &gt; LAY.plotW) return Infinity;
    const x1 = xOf(i1), y1 = yOf(d.p1.p), perBar = (d.p2.p - d.p1.p) / (i2 - i1), iR = idxAtX(LAY.plotW);
    let best = Infinity;
    for (const r of FAN_RATIOS) best = Math.min(best, segDist(px, py, x1, y1, LAY.plotW, yOf(d.p1.p + perBar * r * (iR - i1))));
    return best;
  }
  if (!TREND_TYPES.has(d.type)) return Infinity;
  const P = trendPts(d, bars);
  if (!P.length) return Infinity;
  switch (d.type) {
    case &#x27;tl&#x27;: case &#x27;info&#x27;: case &#x27;angle&#x27;:
      return P.length &lt; 2 ? Infinity : segDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;ext&#x27;:
      return P.length &lt; 2 ? Infinity : lineDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;ray&#x27;:
      return P.length &lt; 2 ? Infinity : rayDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;hray&#x27;:
      return segDist(px, py, P[0].x, P[0].y, LAY.plotW, P[0].y);
    case &#x27;vl&#x27;:
      return Math.abs(px - P[0].x);
    case &#x27;cross&#x27;:
      return Math.min(Math.abs(px - P[0].x), Math.abs(py - P[0].y));
    case &#x27;chan&#x27;: {
      if (P.length &lt; 3 || P[1].x === P[0].x) return Infinity;
      const m = (P[1].y - P[0].y) / (P[1].x - P[0].x);
      const yMain = x =&gt; P[0].y + m * (x - P[0].x);
      const dyOff = P[2].y - yMain(P[2].x);
      const xL = -20, xR = LAY.plotW + 20;
      return Math.min(segDist(px, py, xL, yMain(xL), xR, yMain(xR)), segDist(px, py, xL, yMain(xL) + dyOff, xR, yMain(xR) + dyOff));
    }
    case &#x27;reg&#x27;:
      return regDist(P, bars, px, py);
    case &#x27;flat&#x27;: {
      if (P.length &lt; 3 || P[0].x === P[1].x) return Infinity;
      const x0 = Math.min(P[0].x, P[1].x), x1 = Math.max(P[0].x, P[1].x), yTop = P[0].y, yBot = P[2].y;
      return Math.min(segDist(px, py, x0, yTop, x1, yTop), segDist(px, py, x0, yBot, x1, yBot), segDist(px, py, x0, yTop, x0, yBot), segDist(px, py, x1, yTop, x1, yBot));
    }
    default: return Infinity;
  }
}

function panTap(bars, px, py) {
  const r = delTagRect(bars);
  if (r &amp;&amp; px &gt;= r.x - 8 &amp;&amp; px &lt;= r.x + r.w + 8 &amp;&amp; py &gt;= r.y - 8 &amp;&amp; py &lt;= r.y + r.h + 8) {

  }
  const k = drawingAt(bars, px, py);
  if (k &gt;= 0) {

  hideUndo();
  saveDrawings();

  schedule();
}
function onTap(x, y) {

  if (S.mode === &#x27;pan&#x27;) { panTap(bars, x, y); schedule(); return; }
  const i = clamp(Math.round(idxAtX(x)), 0, bars.length - 1);
  const p = r2(priceAtY(y));
  if (S.mode === &#x27;hline&#x27;) {
    S.drawings.push({ type: &#x27;hline&#x27;, p });

    tapDraw(bars, i, p);

      say(&#x27;Đã đặt điểm gốc. Chạm điểm thứ hai để định nghĩa 1x1.&#x27;);

      say(&#x27;Đã vẽ Gann Fan. Bấm Gann Fan để vẽ thêm.&#x27;);
    }
  }
  schedule();
}

function bindPointer() {
  const pts = new Map();
  let drag = null, pinch = null;
  const rel = e =&gt; {
    const r = cv.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top };
  };

  cv.addEventListener(&#x27;pointerdown&#x27;, e =&gt; {
    e.preventDefault();
    try { cv.setPointerCapture(e.pointerId); } catch (err) { /* không bắt được con trỏ */ }
    const p = rel(e);
    pts.set(e.pointerId, p);
    if (pts.size === 1) {
      drag = {
        x0: p.x, y0: p.y, a0: S.anchor, moved: false,
        // Kéo trong ô giá: dọc = dời thang giá; kéo trên cột giá bên phải = co giãn thang giá
        inMain: p.y &lt; LAY.candleBottom + 6, onAxis: p.x &gt; LAY.plotW &amp;&amp; p.y &lt; LAY.candleBottom + 6,
        vp0: { pMin: VP.pMin, pMax: VP.pMax }, vert: false
      };
      pinch = null;
      S.cross = p;
    } else {
      // Hai ngón: không còn là một lần chạm. Co giãn ở chế độ vẽ chỉ bắt đầu khi hai ngón thật sự kéo đi.
      drag = null;
      if (pts.size === 2 &amp;&amp; hasData) {
        const [p1, p2] = [...pts.values()];
        pinch = {
          d0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
          w0: S.barW,
          iMid: idxAtX((p1.x + p2.x) / 2),
          mx0: (p1.x + p2.x) / 2,
          my0: (p1.y + p2.y) / 2,
          armed: S.mode === &#x27;pan&#x27;
        };
      }
    }
    schedule();
  });

  cv.addEventListener(&#x27;pointermove&#x27;, e =&gt; {
    const p = rel(e);
    if (pts.has(e.pointerId)) pts.set(e.pointerId, p);
    else if (e.pointerType !== &#x27;mouse&#x27;) return;
    if (!hasData) return;
    if (pinch &amp;&amp; pts.size &gt;= 2) {
      const [p1, p2] = [...pts.values()];
      const mx = (p1.x + p2.x) / 2, my = (p1.y + p2.y) / 2;
      const d = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
      if (!pinch.armed) {
        // Chế độ vẽ: ngón thứ hai chạm nhẹ không được làm biểu đồ phóng to hay trôi đi
        if (Math.abs(d - pinch.d0) &lt; 20 &amp;&amp; Math.hypot(mx - pinch.mx0, my - pinch.my0) &lt; 20) return;
        pinch.armed = true;
        pinch.d0 = d;
        pinch.w0 = S.barW;
        pinch.iMid = idxAtX(mx);
      }
      S.barW = clamp(pinch.w0 * d / pinch.d0, 1.5, 40);
      S.anchor = pinch.iMid - 0.5 + (LAY.plotW - mx) / S.barW;
      clampAnchor();
      S.cross = null;
    } else if (drag &amp;&amp; pts.size === 1) {
      // Chế độ vẽ/Gann/đường ngang: một ngón chỉ để ngắm và đặt điểm, không kéo biểu đồ
      const dx = p.x - drag.x0, dy = p.y - drag.y0;
      if (!drag.moved &amp;&amp; Math.hypot(dx, dy) &gt; 6) drag.moved = true;
      if (drag.moved &amp;&amp; S.mode === &#x27;pan&#x27;) {
        const ppx = (drag.vp0.pMax - drag.vp0.pMin) / Math.max(1, LAY.candleBottom - LAY.candleTop);
        if (drag.onAxis) {
          // Cột giá: kéo lên = giãn nến (thu hẹp khoảng giá), kéo xuống = co lại
          const k = Math.exp(dy * 0.006), mid = (drag.vp0.pMin + drag.vp0.pMax) / 2, half = (drag.vp0.pMax - drag.vp0.pMin) / 2 * k;
          Y_LOCK = { pMin: mid - half, pMax: mid + half };
        } else {
          S.anchor = drag.a0 - dx / S.barW;
          clampAnchor();
          if (drag.inMain &amp;&amp; (drag.vert || Math.abs(dy) &gt; 6)) {
            drag.vert = true;
            Y_LOCK = { pMin: drag.vp0.pMin + dy * ppx, pMax: drag.vp0.pMax + dy * ppx };
          }
        }
      }
      S.cross = p;
    } else if (e.pointerType === &#x27;mouse&#x27;) {
      S.cross = p;
    }
    schedule();
  });

  const end = (e, cancelled) =&gt; {
    pts.delete(e.pointerId);
    if (drag &amp;&amp; pts.size === 0) {
      // Chế độ vẽ: điểm đặt ở chỗ nhấc ngón; dấu thập đã theo ngón tay lúc đang giữ
      if (!cancelled &amp;&amp; hasData &amp;&amp; (S.mode !== &#x27;pan&#x27; || !drag.moved)) {
        const q = S.mode === &#x27;pan&#x27; ? { x: drag.x0, y: drag.y0 } : rel(e);
        // Chạm vào cột giá khi đang chỉnh tay: trở lại tự co giãn
        if (drag.onAxis &amp;&amp; Y_LOCK) yAuto();
        else onTap(q.x, q.y);
      }
      drag = null;
    }
    if (pts.size &lt; 2) pinch = null;
    if (pts.size === 0 &amp;&amp; e.pointerType !== &#x27;mouse&#x27;) S.cross = null;
    schedule();
  };
  cv.addEventListener(&#x27;pointerup&#x27;, e =&gt; end(e, false));
  cv.addEventListener(&#x27;pointercancel&#x27;, e =&gt; end(e, true));
  cv.addEventListener(&#x27;pointerleave&#x27;, e =&gt; {
    if (e.pointerType === &#x27;mouse&#x27; &amp;&amp; pts.size === 0) {
      S.cross = null;
      schedule();
    }
  });
  cv.addEventListener(&#x27;wheel&#x27;, e =&gt; {
    if (!hasData) return;
    e.preventDefault();
    const p = rel(e);
    zoomAt(p.x, Math.exp(-e.deltaY * 0.0015));
    S.cross = p;
    schedule();
  }, { passive: false });
}

// OANDA v20: GET /v3/instruments/{mã}/candles, giá giữa (mid), tối đa 5000 nến mỗi lần
async function oandaFetch(cfg, size) {
  const q = new URLSearchParams({ granularity: cfg.tf, count: String(Math.min(5000, size)), price: &#x27;M&#x27; });
  const url = OA_HOST[cfg.oaEnv] + &#x27;/v3/instruments/&#x27; + encodeURIComponent(cfg.sym) + &#x27;/candles?&#x27; + q.toString();
  const { res, data } = await fdGet(url, { Authorization: &#x27;Bearer &#x27; + cfg.oaKey }, &#x27;OANDA&#x27;);
  if (!res.ok) {
    throw new Error(res.status === 401 ? &#x27;Sai token OANDA, hoặc token không thuộc loại tài khoản đã chọn (demo/thật).&#x27;
      : res.status === 400 || res.status === 404 ? &#x27;OANDA không nhận mã &#x27; + cfg.sym + &#x27; (viết kiểu XAU_USD).&#x27;
      : res.status === 429 ? &#x27;Gọi OANDA quá nhanh, đợi rồi thử lại.&#x27;
      : (data.errorMessage || (&#x27;OANDA trả lỗi &#x27; + res.status + &#x27;.&#x27;)));
  }
  const list = Array.isArray(data.candles) ? data.candles : [];
  return list.filter(k =&gt; k &amp;&amp; k.mid).map(k =&gt; ({ t: tdTime(k.time), o: +k.mid.o, h: +k.mid.h, l: +k.mid.l, c: +k.mid.c, v: +k.volume || 0 }));

function aiSnapshot() {
  if (!hasData) return { error: &#x27;Chưa có nến. Kết nối nguồn dữ liệu (Twelve Data hoặc OANDA) trong Cài đặt, hoặc bấm Demo.&#x27; };

  const e34 = ema(c, 34), e80 = ema(c, 80), mac = macdLineArr(c, 12, 26), sig = ema(mac, 9), tr = trixCalc(c, 25, 15);

  hidden.forEach(x =&gt; { x.on = false; });
  indsVer += 1;

  XK_FIT = fit &amp;&amp; Number.isFinite(fit.lo) &amp;&amp; Number.isFinite(fit.hi) ? fit : null;

    out = {

  indsVer += 1;

  schedule();
  return out;
}

// Trả về { order } nếu hợp lệ, hoặc { error }
function ordRead(side) {
  const cfg = acctStored();
  const base = acctBase(cfg.url), token = String(cfg.token || &#x27;&#x27;);
  if (!base || !token) return { error: &#x27;Chưa kết nối VPS. Nhập địa chỉ và token ở trên trước.&#x27; };
  const symbol = $(&#x27;ordSym&#x27;).value.trim();   // giữ hoa/thường: Exness Standard dùng XAUUSDm
  if (!/^[A-Za-z0-9._#-]{3,20}$/.test(symbol)) return { error: &#x27;Mã không hợp lệ.&#x27; };
  const vol = ordNum(&#x27;ordVol&#x27;), max = ordNum(&#x27;ordMax&#x27;), sl = ordNum(&#x27;ordSl&#x27;), tp = ordNum(&#x27;ordTp&#x27;);
  if (!(vol &gt; 0)) return { error: &#x27;Khối lượng phải lớn hơn 0.&#x27; };
  if (Math.abs(vol * 100 - Math.round(vol * 100)) &gt; 1e-6) return { error: &#x27;Khối lượng chỉ lấy 2 số lẻ (bước 0.01).&#x27; };
  if (!(max &gt; 0)) return { error: &#x27;Lot tối đa phải lớn hơn 0.&#x27; };
  if (vol &gt; max + 1e-9) return { error: &#x27;Khối lượng &#x27; + vol + &#x27; vượt lot tối đa &#x27; + max + &#x27;.&#x27; };
  if (sl === null) return { error: &#x27;Phải có SL.&#x27; };
  if (!(sl &gt; 0)) return { error: &#x27;SL không hợp lệ.&#x27; };
  if (tp !== null &amp;&amp; !(tp &gt; 0)) return { error: &#x27;TP không hợp lệ.&#x27; };
  if (tp !== null) {
    if (side === &#x27;BUY&#x27; &amp;&amp; !(tp &gt; sl)) return { error: &#x27;Lệnh BUY: TP phải cao hơn SL.&#x27; };
    if (side === &#x27;SELL&#x27; &amp;&amp; !(tp &lt; sl)) return { error: &#x27;Lệnh SELL: TP phải thấp hơn SL.&#x27; };
  }
  const px = ordChartPrice();
  if (px !== null) {
    if (side === &#x27;BUY&#x27; &amp;&amp; !(sl &lt; px)) return { error: &#x27;Lệnh BUY: SL phải thấp hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (side === &#x27;SELL&#x27; &amp;&amp; !(sl &gt; px)) return { error: &#x27;Lệnh SELL: SL phải cao hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (tp !== null &amp;&amp; side === &#x27;BUY&#x27; &amp;&amp; !(tp &gt; px)) return { error: &#x27;Lệnh BUY: TP phải cao hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (tp !== null &amp;&amp; side === &#x27;SELL&#x27; &amp;&amp; !(tp &lt; px)) return { error: &#x27;Lệnh SELL: TP phải thấp hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
  }
  const note = $(&#x27;ordNote&#x27;).value.trim().replace(/[^\w .-]/g, &#x27;&#x27;).slice(0, 20);
  return { base, token, px, order: { symbol, side, volume: Math.round(vol * 100) / 100, sl, tp: tp === null ? 0 : tp, comment: note || &#x27;XIKE&#x27; } };
}
// Gửi POST tới VPS. Không tự thử lại: lỗi mạng có thể là lệnh đã khớp mà chưa kịp trả lời.
async function ordPost(base, token, path, body) {
  const ctl = new AbortController();
  const timer = setTimeout(() =&gt; ctl.abort(), 15000);
  try {
    const res = await fetch(base + path, {
      method: &#x27;POST&#x27;, cache: &#x27;no-store&#x27;, signal: ctl.signal,
      headers: { &#x27;X-Token&#x27;: token, &#x27;Content-Type&#x27;: &#x27;application/json&#x27; },
      body: JSON.stringify(body)
    });
    const data = await res.json().catch(() =&gt; ({}));
    if (res.status === 401) throw new Error(&#x27;Sai token.&#x27;);
    if (res.status === 404 || res.status === 405) throw new Error(&#x27;VPS chưa có cổng &#x27; + path + &#x27;. Cần cập nhật code VPS.&#x27;);
    if (!res.ok || data.ok === false) {
      const why = data.error || data.comment || (&#x27;VPS trả lỗi &#x27; + res.status);
      throw new Error(String(why) + (data.retcode ? &#x27; (mã &#x27; + data.retcode + &#x27;)&#x27; : &#x27;&#x27;) + &#x27;.&#x27;);
    }
    return data;
  } catch (err) {
    if (err.name === &#x27;AbortError&#x27;) throw new Error(&#x27;Hết thời gian chờ VPS. Lệnh CÓ THỂ đã khớp: xem lại Lệnh đang mở trước khi đặt lại.&#x27;);
    if (err instanceof TypeError) throw new Error(&#x27;Không gửi được tới VPS. Lệnh CÓ THỂ đã tới VPS: xem lại Lệnh đang mở trước khi đặt lại.&#x27;);
    throw err;
  } finally { clearTimeout(timer); }
}
function ordId() {
  const a = new Uint32Array(2);
  (window.crypto || {}).getRandomValues ? crypto.getRandomValues(a) : (a[0] = Math.random() * 4e9, a[1] = Date.now());
  return &#x27;xk&#x27; + a[0].toString(36) + a[1].toString(36);
}
