function cleanParam(f, v) {
  const blank = v === null || v === undefined || (typeof v === &#x27;string&#x27; &amp;&amp; v.trim() === &#x27;&#x27;);
  const x = blank ? NaN : Number(v);
  const val = clamp(Number.isFinite(x) ? x : f.def, f.min, f.max);
  return f.float ? r2(val) : Math.round(val);
}

function sanitizeInds(arr) {
  const out = [];
  for (const it of arr) {
    if (out.length &gt;= MAX_INDS) break;

  const totalH = Math.max(120, h - axisB);

  if (nPanes) {
    paneH = clamp(Math.floor(totalH * 0.2), 56, 112);
    mainH = totalH - nPanes * paneH;
    if (mainH &lt; 130) {
      mainH = 130;
      paneH = Math.max(40, Math.floor((totalH - 130) / nPanes));
    }
  }
  const volH = volumeOn() ? Math.round(mainH * 0.16) : 0;
  return {
    w, h, axisR, axisB, plotW, totalH, mainH, paneH, nPanes, volH,
    candleTop: 8, candleBottom: mainH - volH - 6
  };
}

  let iL = Math.max(0, Math.floor(idxAtX(0)));
  let iR = Math.min(n - 1, Math.ceil(idxAtX(L.plotW)));
  if (iR &lt; iL) { iL = Math.max(0, n - 50); iR = n - 1; }

  const ovs = overlayInsts().map(inst =&gt; ({ def: IND_BY_ID[inst.id], r: instResult(inst) }));
  const pns = paneInsts().map(inst =&gt; ({ def: IND_BY_ID[inst.id], r: instResult(inst) }));

  // Khoảng giá: nến, cộng thêm các đường và dải đang hiển thị
  let lo = Infinity, hi = -Infinity, vmax = 0;
  for (let i = iL; i &lt;= iR; i++) {
    const b = bars[i];
    if (b.l &lt; lo) lo = b.l;
    if (b.h &gt; hi) hi = b.h;
    if (b.v &gt; vmax) vmax = b.v;
  }
  for (const o of ovs) {
    for (const ln of o.r.lines) {
      if (ln.nofit) continue;
      for (let i = iL; i &lt;= iR; i++) {
        const v = ln.v[i];
        if (Number.isFinite(v)) {
          if (v &lt; lo) lo = v;
          if (v &gt; hi) hi = v;
        }
      }
    }
    if (o.r.fit) {
      // Mức SL/TP chỉ nới thang giá khi ở gần nến (như TradingView: nến luôn đủ lớn để đọc)
      const rg = hi &gt; lo ? hi - lo : 1, fLo = lo - rg * 0.5, fHi = hi + rg * 0.5;
      for (const lv of o.r.levels || []) {
        if (Number.isFinite(lv.v) &amp;&amp; lv.v &gt;= fLo &amp;&amp; lv.v &lt;= fHi) {
          if (lv.v &lt; lo) lo = lv.v;
          if (lv.v &gt; hi) hi = lv.v;
        }
      }
    }
  }
  if (XK_FIT) { lo = Math.min(lo, XK_FIT.lo); hi = Math.max(hi, XK_FIT.hi); }
  const pad = hi &gt; lo ? (hi - lo) * 0.08 : 1;
  VP = (Y_LOCK &amp;&amp; !XK_FIT) ? { pMin: Y_LOCK.pMin, pMax: Y_LOCK.pMax } : { pMin: lo - pad, pMax: hi + pad };

  const cr = S.cross;
  const crInPlot = !!cr &amp;&amp; cr.x &gt;= 0 &amp;&amp; cr.x &lt;= L.plotW;
  const crInMain = crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.mainH;
  const li = crInPlot ? clamp(Math.round(idxAtX(cr.x)), 0, n - 1) : n - 1;
  const tags = [];   // nhãn có nền trên trục phải
  const lvTxt = [];  // vị trí chữ của các mức SL/TP/Entry đã vẽ
  const axis = [];   // nhãn trục phải thường

  // Lưới giá
  const step = niceStep(VP.pMax - VP.pMin, 6);
  const dec = clamp(-Math.floor(Math.log10(step)), 0, 2);
  for (let k = Math.ceil(VP.pMin / step); k * step &lt;= VP.pMax; k++) {
    const y = crisp(yOf(k * step));
    cx.strokeStyle = COL.line; cx.lineWidth = hair();
    cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
    axis.push({ y: y, text: (k * step).toFixed(dec), color: COL.axis });
  }

  // Lưới thời gian và nhãn trục thời gian
  const P = pickPeriod(ser.sec, S.barW);
  const tzSec = S.tz * 3600;
  let lastDay = null;
  cx.textAlign = &#x27;center&#x27;;
  cx.fillStyle = COL.axis;
  for (let i = Math.max(1, iL); i &lt;= iR; i++) {
    const tt = bars[i].t + tzSec, tp = bars[i - 1].t + tzSec;
    if (Math.floor(tt / P) === Math.floor(tp / P)) continue;
    const day = Math.floor(tt / 86400);
    const newDay = day !== lastDay;
    lastDay = day;
    const x = xOf(i);
    if (x &lt; 0 || x &gt; L.plotW) continue;
    let label;
    if (P &lt; 86400) label = newDay ? fmtTs(bars[i].t, &#x27;dm&#x27;) : fmtTs(bars[i].t, &#x27;hm&#x27;);
    else if (P &lt; 2592000) label = fmtTs(bars[i].t, &#x27;dm&#x27;);
    else { const d = new Date(tt * 1000); label = pad2(d.getUTCMonth() + 1) + &#x27;/&#x27; + d.getUTCFullYear(); }
    const xl = crisp(x);
    cx.strokeStyle = COL.line; cx.lineWidth = hair();
    cx.beginPath(); cx.moveTo(xl, 0); cx.lineTo(xl, L.totalH); cx.stroke();
    cx.fillText(label, x, L.totalH + L.axisB / 2);
  }

  // Giờ cấm: cột chéo đỏ phủ cả vùng giá lẫn vùng chỉ báo
  cx.save();
  cx.beginPath(); cx.rect(0, 0, L.plotW, L.totalH); cx.clip();
  if (S.showBlock &amp;&amp; blockPat) {
    cx.fillStyle = blockPat;
    for (let i = iL; i &lt;= iR; i++) {
      if (inBlock(bars[i].t)) cx.fillRect(xOf(i) - S.barW / 2, 0, S.barW, L.totalH);
    }
  }
  cx.restore();

  // Vùng giá: khối lượng, dải tô, nến, đường, mức, hình vẽ, giá cuối, con trỏ
  cx.save();
  cx.beginPath(); cx.rect(0, 0, L.plotW, L.mainH); cx.clip();

  if (L.volH &gt; 8 &amp;&amp; vmax &gt; 0) {
    const bw = Math.max(1, S.barW * 0.7), base = L.mainH - 2, maxH = L.volH - 6;
    for (let i = iL; i &lt;= iR; i++) {
      const b = bars[i], h = b.v / vmax * maxH;
      cx.fillStyle = b.c &gt;= b.o ? COL.volUp : COL.volDn;
      cx.fillRect(xOf(i) - bw / 2, base - h, bw, h);
    }
  }

  for (const o of ovs) if (o.r.band) drawBand(o.r, iL, iR);

  // Nến vẽ theo điểm ảnh thật: thân và bấc thẳng hàng, mép sắc như TradingView
  const vt = CTYPE === &#x27;vol&#x27; &amp;&amp; ser.hasVol &amp;&amp; vmax &gt; 0;
  const wd = Math.max(1, Math.floor(DPR));
  for (let i = iL; i &lt;= iR; i++) {
    const bc = bars[i], col = bc.c &gt;= bc.o ? COL.up : COL.down;
    const bw = vt ? Math.max(1, S.barW * 0.94 * Math.max(0.12, bc.v / vmax)) : Math.max(1, S.barW * 0.72);
    let bwd = Math.max(wd, Math.round(bw * DPR));
    if ((bwd - wd) % 2) bwd += 1;
    const xl = Math.round(xOf(i) * DPR - bwd / 2), xw = xl + (bwd - wd) / 2;
    const yh = Math.round(yOf(bc.h) * DPR), yl = Math.round(yOf(bc.l) * DPR);
    const yt = Math.round(yOf(Math.max(bc.o, bc.c)) * DPR);
    const yb = Math.max(yt + Math.max(1, Math.round(DPR)), Math.round(yOf(Math.min(bc.o, bc.c)) * DPR));
    cx.fillStyle = col;
    cx.fillRect(xw / DPR, yh / DPR, wd / DPR, Math.max(1, yl - yh) / DPR);
    cx.fillRect(xl / DPR, yt / DPR, bwd / DPR, (yb - yt) / DPR);
  }

  for (const o of ovs) {
    for (const ln of o.r.lines) {
      cx.setLineDash(ln.dash || []);
      polyline(ln.v, Math.max(0, iL - 1), iR, yOf, ln.color, ln.w || (ln.thin ? 1 : 1.5));
      cx.setLineDash([]);
    }
    for (const lv of o.r.levels || []) {
      if (lv.v &lt; VP.pMin || lv.v &gt; VP.pMax) continue;
      const y = crisp(yOf(lv.v));
      cx.setLineDash([5, 4]); cx.strokeStyle = lv.color; cx.lineWidth = hair();
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      cx.fillStyle = lv.color; cx.textAlign = &#x27;right&#x27;;
      let ty = y - 9;
      if (lvTxt.some(v =&gt; Math.abs(v - ty) &lt; 14)) ty = y + 9;
      if (!lvTxt.some(v =&gt; Math.abs(v - ty) &lt; 14)) { cx.fillText(lv.text, L.plotW - 6, ty); lvTxt.push(ty); }
      tags.push({ y: y, text: lv.v.toFixed(2), bg: lv.color, fg: &#x27;#000000&#x27; });
      cx.textAlign = &#x27;left&#x27;;
    }
  }

  // Nhãn tín hiệu trên giá: BUY và BUY P nằm dưới nến, SELL và SELL P nằm trên nến; nhãn cùng nến được xếp chồng
  const stacked = {};
  for (const o of ovs) {
    for (const mk of o.r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      const key = mk.i + (mk.up ? &#x27;u&#x27; : &#x27;d&#x27;), k = stacked[key] || 0;
      stacked[key] = k + 1;
      const bc = bars[mk.i], xc = xOf(mk.i);
      if (mk.up) drawMark(xc, yOf(bc.l) + 6 + k * 16, mk.text, COL.up);
      else drawMark(xc, yOf(bc.h) - 20 - k * 16, mk.text, COL.down);
    }
  }

  for (const d of S.drawings) {
    if (d.type === &#x27;fan&#x27;) { drawFan(d, bars); continue; }
    if (TREND_TYPES.has(d.type)) { drawTrend(d, bars); continue; }
    if (d.type === &#x27;hline&#x27;) {
      const yh = yOf(d.p);
      if (yh &lt; L.candleTop || yh &gt; L.candleBottom) continue;
      const y = crisp(yh);
      cx.strokeStyle = COL.hline; cx.lineWidth = hair(); cx.setLineDash([6, 4]);
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      tags.push({ y: yh, text: d.p.toFixed(2), bg: COL.hline, fg: COL.bg });
    }
  }

  if (selValid() &amp;&amp; S.mode === &#x27;pan&#x27;) {
    cx.save();
    cx.shadowColor = DRAW_COL;
    cx.shadowBlur = 10;
    drawSelShape(selDraw, bars);
    cx.restore();
    drawSelTag(bars);
  }
  drawPending(bars);
  if (S.pendingFan) {
    const i = nearestIndex(bars, S.pendingFan.t);
    cx.fillStyle = COL.fan;
    cx.beginPath(); cx.arc(xOf(i), yOf(S.pendingFan.p), 4, 0, Math.PI * 2); cx.fill();
  }

  const last = bars[n - 1], lastUp = last.c &gt;= last.o;
  if (last.c &gt;= VP.pMin &amp;&amp; last.c &lt;= VP.pMax) {
    const yl = crisp(yOf(last.c));
    cx.strokeStyle = lastUp ? COL.up : COL.down; cx.lineWidth = hair(); cx.setLineDash([2, 3]);
    cx.beginPath(); cx.moveTo(0, yl); cx.lineTo(L.plotW, yl); cx.stroke();
    cx.setLineDash([]);
    // Giá cuối kèm đồng hồ đếm ngược tới lúc đóng nến (như TradingView), chỉ khi dữ liệu live
    const left = last.t + ser.sec - Date.now() / 1000;
    let sub = &#x27;&#x27;;
    if (/^Live/.test(baseName) &amp;&amp; left &gt; 0 &amp;&amp; left &lt;= ser.sec) {
      const sl = Math.floor(left), hh = Math.floor(sl / 3600), mm = Math.floor(sl % 3600 / 60), ss = sl % 60;
      sub = (hh ? hh + &#x27;:&#x27; + pad2(mm) : pad2(mm)) + &#x27;:&#x27; + pad2(ss);
    }
    tags.push({ y: yOf(last.c), text: last.c.toFixed(2), sub: sub, bg: lastUp ? COL.up : COL.down, fg: &#x27;#ffffff&#x27;, keep: true });
  }

  if (crInMain) {
    const yc = crisp(cr.y);
    cx.strokeStyle = COL.cross; cx.lineWidth = hair(); cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
    cx.setLineDash([]);
    if (cr.y &gt;= L.candleTop &amp;&amp; cr.y &lt;= L.candleBottom) {
      tags.push({ y: cr.y, text: priceAtY(cr.y).toFixed(2), bg: COL.box, fg: COL.text });
    }
  }
  cx.restore();

  // Các vùng chỉ báo dao động, mỗi vùng có thang riêng
  pns.forEach((pn, k) =&gt; {
    const def = pn.def, r = pn.r;
    const top = L.mainH + k * L.paneH, bot = top + L.paneH;
    let lo2 = Infinity, hi2 = -Infinity;
    for (const ln of r.lines) {
      for (let i = iL; i &lt;= iR; i++) {
        const v = ln.v[i];
        if (Number.isFinite(v)) {
          if (v &lt; lo2) lo2 = v;
          if (v &gt; hi2) hi2 = v;
        }
      }
    }
    for (const lv of def.levels) {
      if (lv.v &lt; lo2) lo2 = lv.v;
      if (lv.v &gt; hi2) hi2 = lv.v;
    }
    if (def.range) {
      lo2 = def.range[0];
      hi2 = def.range[1];
    } else {
      if (!Number.isFinite(lo2) || !Number.isFinite(hi2)) { lo2 = 0; hi2 = 1; }
      if (def.zero) { lo2 = Math.min(lo2, 0); hi2 = Math.max(hi2, 0); }
      if (hi2 - lo2 &lt; 1e-9) hi2 = lo2 + 1;
      const pd = (hi2 - lo2) * 0.1;
      lo2 -= pd;
      hi2 += pd;
    }
    const yP = v =&gt; top + 6 + (hi2 - v) / (hi2 - lo2) * (L.paneH - 12);

    cx.strokeStyle = COL.line; cx.lineWidth = hair();
    cx.beginPath(); cx.moveTo(0, crisp(top)); cx.lineTo(L.w, crisp(top)); cx.stroke();

    cx.save();
    cx.beginPath(); cx.rect(0, top, L.plotW, L.paneH); cx.clip();
    cx.setLineDash([4, 4]); cx.strokeStyle = COL.grid; cx.lineWidth = hair();
    for (const lv of def.levels) {
      const y = crisp(yP(lv.v));
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
    }
    cx.setLineDash([]);
    for (const ln of r.lines) {
      if (ln.type === &#x27;hist&#x27;) {
        const bw = Math.max(1, Math.round(S.barW * 0.6)), y0 = yP(0);
        for (let i = iL; i &lt;= iR; i++) {
          const v = ln.v[i];
          if (!Number.isFinite(v)) continue;
          const yv = yP(v);
          cx.fillStyle = v &gt;= 0 ? ln.up : ln.dn;
          cx.fillRect(Math.round(xOf(i)) - Math.floor(bw / 2), Math.min(y0, yv), bw, Math.max(1, Math.abs(yv - y0)));
        }
      } else {
        polyline(ln.v, Math.max(0, iL - 1), iR, yP, ln.color, ln.thin ? 1 : 1.4);
      }
    }
    // Nhãn Buy / Sell của TRIX: Buy nằm sát đáy ô, Sell nằm sát đỉnh ô
    for (const mk of r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      drawMark(xOf(mk.i), mk.up ? bot - 18 : top + 22, mk.text, mk.up ? COL.up : COL.down);
    }
    if (crInPlot &amp;&amp; cr.y &gt;= top &amp;&amp; cr.y &lt; bot) {
      const yc = crisp(cr.y);
      cx.strokeStyle = COL.cross; cx.lineWidth = hair(); cx.setLineDash([3, 3]);
      cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
      cx.setLineDash([]);
      const vv = hi2 - (cr.y - top - 6) / (L.paneH - 12) * (hi2 - lo2);
      tags.push({ y: cr.y, text: fmtVal(vv, def.dec), bg: COL.box, fg: COL.text });
    }
    cx.restore();

    // Nhãn trục phải: các mức tham chiếu và mốc đẹp
    const pstep = niceStep(hi2 - lo2, 3);
    const pdec = clamp(-Math.floor(Math.log10(pstep)), 0, 4);
    const levelYs = [];
    for (const lv of def.levels) {
      const y = yP(lv.v);
      levelYs.push(y);
      axis.push({ y: y, text: fmtVal(lv.v, pdec), color: COL.muted });
    }
    for (let kk = Math.ceil(lo2 / pstep); kk * pstep &lt;= hi2; kk++) {
      const y = yP(kk * pstep);
      if (y &lt; top + 4 || y &gt; bot - 4) continue;
      if (levelYs.some(ly =&gt; Math.abs(ly - y) &lt; 12)) continue;
      axis.push({ y: y, text: (kk * pstep).toFixed(pdec), color: COL.muted });
    }
  });

  // Đường dọc của con trỏ, trên toàn khung
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const xl = crisp(cr.x);
    cx.save();
    cx.beginPath(); cx.rect(0, 0, L.plotW, L.totalH); cx.clip();
    cx.strokeStyle = COL.cross; cx.lineWidth = hair(); cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(xl, 0); cx.lineTo(xl, L.totalH); cx.stroke();
    cx.setLineDash([]);
    cx.restore();
  }

  // Viền cột giá và trục thời gian
  cx.strokeStyle = COL.border; cx.lineWidth = hair();
  cx.beginPath();
  cx.moveTo(crisp(L.plotW), 0); cx.lineTo(crisp(L.plotW), L.h);
  cx.moveTo(0, crisp(L.totalH)); cx.lineTo(L.w, crisp(L.totalH));
  cx.stroke();

  // Nhãn có nền: không chồng lên nhau (giá cuối giữ nguyên chỗ, nhãn khác né ra)
  for (const t of tags) t.h = t.sub ? 32 : 18;
  {
    const mid = (t, y) =&gt; y - 9 + t.h / 2;
    const placed = tags.filter(t =&gt; t.keep);
    for (const t of tags.filter(t =&gt; !t.keep).sort((p, q) =&gt; p.y - q.y)) {
      const hit = y =&gt; placed.some(g =&gt; Math.abs(mid(g, g.y) - mid(t, y)) &lt; (g.h + t.h) / 2 + 1);
      let y = t.y;
      for (let d = 1; hit(y) &amp;&amp; d &lt; 600; d++) {
        if (!hit(t.y + d)) { y = t.y + d; break; }
        if (!hit(t.y - d)) { y = t.y - d; break; }
      }
      t.y = y;
      placed.push(t);
    }
  }
  // Nhãn trục (ngoài vùng cắt xén)
  cx.textAlign = &#x27;left&#x27;;
  for (const t of axis) {
    if (tags.some(g =&gt; t.y &gt; g.y - 9 - 8 &amp;&amp; t.y &lt; g.y - 9 + g.h + 8)) continue;
    cx.fillStyle = t.color;
    cx.fillText(t.text, L.plotW + 8, t.y);
  }
  for (const t of tags) {
    const top = Math.round(t.y - 9);
    cx.fillStyle = t.bg;
    if (cx.roundRect) { cx.beginPath(); cx.roundRect(L.plotW + 1, top, L.axisR - 2, t.h, 2); cx.fill(); }
    else cx.fillRect(L.plotW + 1, top, L.axisR - 2, t.h);
    cx.fillStyle = t.fg; cx.fillText(t.text, L.plotW + 7, top + 9);
    if (t.sub) cx.fillText(t.sub, L.plotW + 7, top + 23);
  }
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const txt = fmtTs(timeAtIdx(bars, ser.sec, idxAtX(cr.x)), &#x27;dhm&#x27;);
    cx.font = smallFont();
    const tw = cx.measureText(txt).width + 14;
    const bx = clamp(cr.x - tw / 2, 0, Math.max(0, L.plotW - tw));
    cx.fillStyle = COL.box; cx.fillRect(bx, L.totalH, tw, L.axisB);
    cx.fillStyle = COL.text; cx.textAlign = &#x27;center&#x27;;
    cx.fillText(txt, bx + tw / 2, L.totalH + L.axisB / 2);
  }

  const cur = bars[li], prv = bars[li &gt; 0 ? li - 1 : li];
  updateReadout(cur, prv);
  updateLegend(li);
}

// Chú thích trên góc trái: nến đang trỏ tới và giá trị của từng chỉ báo. Chạm một dòng để chỉnh chỉ báo đó.
function updateLegend(li) {
  const box = $(&#x27;legend&#x27;);
  if (!box) return;
  if (!hasData) {
    if (legendKey !== &#x27;empty&#x27;) { box.textContent = &#x27;&#x27;; legendKey = &#x27;empty&#x27;; }
    return;
  }
  const key = [li, S.tf, dataGen, indsVer, S.showBlock, S.tz, VIEW.ohlc, VIEW.inds, LAY.mainH, LAY.paneH].join(&#x27;|&#x27;);
  if (key === legendKey) return;
  legendKey = key;
  const b = series(S.tf).bars[li];
  const up = b.c &gt;= b.o;
  const parts = [];
  const src = /Demo/.test(baseName) ? &#x27;Demo&#x27; : (baseName.match(/OANDA|Twelve Data|Exness\/MT5/) || [&#x27;Live&#x27;])[0];
  const cc = up ? COL.up : COL.down;
  const bars0 = series(S.tf).bars, pb = bars0[li &gt; 0 ? li - 1 : li];
  const dch = b.c - pb.c, dpc = pb.c ? dch / pb.c * 100 : 0;
  if (VIEW.inds) parts.push(&#x27;&lt;div class=&quot;lg&quot;&gt;&lt;span class=&quot;sym&quot;&gt;XAUUSD · &#x27; + tfLabel(S.tf) + &#x27;&lt;/span&gt;&lt;span class=&quot;src&quot;&gt;· &#x27; + src + &#x27;&lt;/span&gt;&#x27; +
    (S.showBlock &amp;&amp; inBlock(b.t) ? &#x27; &lt;span class=&quot;blk&quot;&gt;Giờ cấm 18–22h VN&lt;/span&gt;&#x27; : &#x27;&#x27;) + &#x27;&lt;/div&gt;&#x27;);
  if (VIEW.ohlc &amp;&amp; VIEW.inds) parts.push(&#x27;&lt;div class=&quot;lg&quot; style=&quot;color:&#x27; + cc + &#x27;&quot;&gt;&lt;span class=&quot;k&quot;&gt;O&lt;/span&gt;&#x27; + b.o.toFixed(2) +

function saveInds() {
  try {
    localStorage.setItem(IND_KEY, JSON.stringify(S.inds.map(x =&gt; ({ id: x.id, p: x.p, on: x.on }))));
  } catch (err) { /* không có bộ nhớ trình duyệt */ }
}

function loadInds() {
  try {
    const v = JSON.parse(localStorage.getItem(IND_KEY) || &#x27;null&#x27;);
    if (Array.isArray(v)) {
      // Thêm Stoch RSI PRO một lần cho danh sách chỉ báo đã lưu từ bản trước
      let added = false;
      try { added = localStorage.getItem(&#x27;xau-chart-srsi-added&#x27;) === &#x27;1&#x27;; } catch (err) { added = true; }
      if (!added &amp;&amp; !v.some(x =&gt; x &amp;&amp; x.id === &#x27;stoch_rsi&#x27;)) v.splice(Math.max(0, v.length - 1), 0, { id: &#x27;stoch_rsi&#x27; });
      try { localStorage.setItem(&#x27;xau-chart-srsi-added&#x27;, &#x27;1&#x27;); } catch (err) { /* bỏ qua */ }
      S.inds = sanitizeInds(v);
      return;
    }
  } catch (err) { /* bỏ qua dữ liệu hỏng */ }
  S.inds = defaultInds();
}

function touchInds() {
  indsVer += 1;
  saveInds();
  LAY = layout();
  updateIndBadge();
  if (!$(&#x27;libView&#x27;).hidden) renderActive();
  schedule();
}

function updateIndBadge() {
  const n = S.inds.filter(x =&gt; x.on).length;
  const bd = $(&#x27;indCount&#x27;);
  bd.textContent = String(n);
  bd.hidden = n === 0;
}

function addInd(id) {
  const def = IND_BY_ID[id];
  if (!def || !IND_ENABLED[id]) return;
  if (S.inds.length &gt;= MAX_INDS) { say(&#x27;Tối đa &#x27; + MAX_INDS + &#x27; chỉ báo. Hãy bỏ bớt trước khi thêm.&#x27;); return; }
  if (def.volume &amp;&amp; S.inds.some(x =&gt; x.id === &#x27;volume&#x27;)) { say(&#x27;Khối lượng đã có trên biểu đồ.&#x27;); return; }
  const p = {};
  for (const f of def.params) p[f.k] = f.def;
  S.inds.push({ uid: newUid(), id: def.id, p, on: true });
  touchInds();
  say(&#x27;Đã thêm &#x27; + def.label(p) + &#x27;.&#x27;);
  closeSheet();
}

function removeInd(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  S.inds = S.inds.filter(x =&gt; x.uid !== uid);
  touchInds();
  say(&#x27;Đã bỏ &#x27; + IND_BY_ID[inst.id].label(inst.p) + &#x27; khỏi biểu đồ.&#x27;);
  if (editUid === uid) showSheetView(&#x27;lib&#x27;);
}

function openEditor(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  const def = IND_BY_ID[inst.id];
  editUid = uid;
  const box = $(&#x27;edFields&#x27;);
  box.textContent = &#x27;&#x27;;
  for (const f of def.params) {
    const lb = document.createElement(&#x27;label&#x27;);
    lb.textContent = f.label;
    const inp = document.createElement(&#x27;input&#x27;);
    inp.type = &#x27;number&#x27;;
    inp.inputMode = f.float ? &#x27;decimal&#x27; : &#x27;numeric&#x27;;
    inp.min = String(f.min);
    inp.max = String(f.max);
    inp.step = f.float ? &#x27;0.01&#x27; : &#x27;1&#x27;;
    inp.value = String(inst.p[f.k]);
    inp.dataset.k = f.k;
    lb.appendChild(inp);
    box.appendChild(lb);
  }
  $(&#x27;edNote&#x27;).textContent = def.desc + (def.params.length ? &#x27;&#x27; : &#x27; Chỉ báo này không có tham số để chỉnh.&#x27;);
  $(&#x27;edReset&#x27;).hidden = !def.params.length;
  showSheetView(&#x27;edit&#x27;, def.label(inst.p));
}

function saveEditor() {
  const inst = S.inds.find(x =&gt; x.uid === editUid);
  if (!inst) { showSheetView(&#x27;lib&#x27;); return; }
  const def = IND_BY_ID[inst.id];
  for (const inp of Array.from($(&#x27;edFields&#x27;).querySelectorAll(&#x27;input&#x27;))) {
    const f = def.params.find(x =&gt; x.k === inp.dataset.k);
    if (f) inst.p[f.k] = cleanParam(f, inp.value);
  }
  touchInds();
  say(&#x27;Đã áp dụng cài đặt &#x27; + def.label(inst.p) + &#x27;.&#x27;);
  showSheetView(&#x27;lib&#x27;);
}

function resetEditor() {
  const inst = S.inds.find(x =&gt; x.uid === editUid);
  if (!inst) return;
  const def = IND_BY_ID[inst.id];
  for (const inp of Array.from($(&#x27;edFields&#x27;).querySelectorAll(&#x27;input&#x27;))) {
    const f = def.params.find(x =&gt; x.k === inp.dataset.k);
    if (f) inp.value = String(f.def);
  }
}

  const list = DRAW_TOOLS.filter(t =&gt; !key || normText(t.name).includes(key));
  const grid = $(&#x27;drawGrid&#x27;);
  grid.innerHTML = list.map(t =&gt; &#x27;&lt;button type=&quot;button&quot; class=&quot;dtool&#x27; + (S.mode === &#x27;draw&#x27; &amp;&amp; S.tool === t.id ? &#x27; on&#x27; : &#x27;&#x27;) + &#x27;&quot; data-id=&quot;&#x27; + t.id + &#x27;&quot;&gt;&#x27; + drawIconSvg(t.id) + &#x27;&lt;span&gt;&#x27; + t.name + &#x27;&lt;/span&gt;&lt;/button&gt;&#x27;).join(&#x27;&#x27;);
  for (const b of Array.from(grid.querySelectorAll(&#x27;.dtool&#x27;))) b.addEventListener(&#x27;click&#x27;, () =&gt; pickDrawTool(b.dataset.id));
  $(&#x27;drawEmpty&#x27;).hidden = list.length &gt; 0;
}
function drawSheetOpen() {
  $(&#x27;drawSearch&#x27;).value = &#x27;&#x27;;
  renderDrawGrid(&#x27;&#x27;);
  $(&#x27;drawSheet&#x27;).hidden = false;
  $(&#x27;scrim&#x27;).hidden = false;
}
function drawSheetClose() {
  $(&#x27;drawSheet&#x27;).hidden = true;
  if ($(&#x27;sheet&#x27;).hidden) $(&#x27;scrim&#x27;).hidden = true;
}
function pickDrawTool(id) {
  const t = DRAW_BY_ID[id];
  if (!t) return;
  drawSheetClose();
  if (t.type === &#x27;hline&#x27;) { setMode(&#x27;hline&#x27;); return; }
  S.tool = id;
  setMode(&#x27;draw&#x27;);
}
function undoDraw() {
  if (S.pendingDraw.length) { S.pendingDraw = []; say(&#x27;Đã hủy điểm đang chọn.&#x27;); schedule(); return; }
  if (!S.drawings.length) { say(&#x27;Chưa có bản vẽ nào để hoàn tác.&#x27;); return; }
  S.drawings.pop();
  saveDrawings();
  say(&#x27;Đã hoàn tác bản vẽ cuối.&#x27;);
  schedule();
}
function bindDraw() {
  setIcon($(&#x27;drawClose&#x27;), &#x27;close&#x27;);
  $(&#x27;drawClose&#x27;).addEventListener(&#x27;click&#x27;, drawSheetClose);
  $(&#x27;drawUndo&#x27;).addEventListener(&#x27;click&#x27;, undoDraw);
  $(&#x27;drawExit&#x27;).addEventListener(&#x27;click&#x27;, () =&gt; { drawSheetClose(); setMode(&#x27;pan&#x27;); });
  $(&#x27;drawSearch&#x27;).addEventListener(&#x27;input&#x27;, () =&gt; renderDrawGrid($(&#x27;drawSearch&#x27;).value));
  $(&#x27;scrim&#x27;).addEventListener(&#x27;click&#x27;, drawSheetClose);
  renderDrawGrid(&#x27;&#x27;);
}
function tapDraw(bars, i, p) {
  const tool = DRAW_BY_ID[S.tool];
  if (!tool || tool.type === &#x27;hline&#x27;) return;
  const pt = { t: bars[i].t, p };
  if (S.pendingDraw.length === 1 &amp;&amp; S.pendingDraw[0].t === pt.t) {
    say(&#x27;Hai điểm đang nằm trên cùng một nến. Chạm nến khác.&#x27;);
    return;
  }
  S.pendingDraw.push(pt);
  if (S.pendingDraw.length &lt; tool.pts) {
    say(tool.name + &#x27;: đã đặt điểm &#x27; + S.pendingDraw.length + &#x27;/&#x27; + tool.pts + &#x27;. Chạm điểm tiếp theo.&#x27;);
    return;
  }
  S.drawings.push({ type: tool.type, pts: S.pendingDraw.slice() });
  S.pendingDraw = [];
  saveDrawings();
  setMode(&#x27;pan&#x27;);
  say(&#x27;Đã vẽ &#x27; + tool.name.toLowerCase() + &#x27;. Bấm Bản vẽ để vẽ thêm.&#x27;);
}
function trendPts(d, bars) {
  return d.pts.map(q =&gt; { const i = nearestIndex(bars, q.t); return { i, p: q.p, x: xOf(i), y: yOf(q.p) }; });
}
function trendSeg(x0, y0, x1, y1) { cx.beginPath(); cx.moveTo(x0, y0); cx.lineTo(x1, y1); cx.stroke(); }
function trendDot(x, y) { cx.beginPath(); cx.arc(x, y, 3, 0, Math.PI * 2); cx.fill(); }
function trendTag(x, y, text) {
  cx.font = smallFont();
  const w = Math.ceil(cx.measureText(text).width) + 10;
  cx.fillStyle = COL.box;
  cx.fillRect(Math.round(x - w / 2), Math.round(y - 9), w, 18);
  cx.fillStyle = COL.text;
  cx.textAlign = &#x27;center&#x27;;
  cx.textBaseline = &#x27;middle&#x27;;
  cx.fillText(text, x, y);
}
function trendPlotClip() { cx.beginPath(); cx.rect(0, LAY.candleTop, LAY.plotW, LAY.candleBottom - LAY.candleTop); cx.clip(); }
function trendFullClip() { cx.beginPath(); cx.rect(0, 0, LAY.plotW, LAY.totalH); cx.clip(); }
function trendFill(alpha) { cx.save(); cx.globalAlpha = alpha; cx.fill(); cx.restore(); }
function drawTrend(d, bars) {
  const P = trendPts(d, bars);
  if (!P.length) return;
  const K = 1000;
  cx.save();
  cx.strokeStyle = DRAW_COL;
  cx.fillStyle = DRAW_COL;
  cx.lineWidth = 1.5;
  switch (d.type) {
    case &#x27;tl&#x27;:
    case &#x27;info&#x27;:
    case &#x27;angle&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, P[1].x, P[1].y);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      if (d.type === &#x27;info&#x27;) {
        const dp = P[1].p - P[0].p;
        const pct = P[0].p ? dp / P[0].p * 100 : 0;
        const nBars = Math.abs(P[1].i - P[0].i);
        trendTag((P[0].x + P[1].x) / 2, (P[0].y + P[1].y) / 2 - 14, (dp &gt;= 0 ? &#x27;+&#x27; : &#x27;&#x27;) + dp.toFixed(2) + &#x27; (&#x27; + pct.toFixed(2) + &#x27;%) · &#x27; + nBars + &#x27; nến&#x27;);
      } else if (d.type === &#x27;angle&#x27;) {
        const deg = Math.atan2(-(P[1].y - P[0].y), Math.abs(P[1].x - P[0].x)) * 180 / Math.PI;
        trendTag(P[1].x, P[1].y - 14, deg.toFixed(1) + &#x27;°&#x27;);
      }
      break;
    }
    case &#x27;ext&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x - dx * K, P[0].y - dy * K, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      break;
    }
    case &#x27;ray&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x, P[0].y, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;hray&#x27;: {
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;vl&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      break;
    }
    case &#x27;cross&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      trendSeg(0, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;chan&#x27;: {
      if (P.length &lt; 3 || P[1].x === P[0].x) break;
      trendPlotClip();
      const m = (P[1].y - P[0].y) / (P[1].x - P[0].x);
      const yMain = x =&gt; P[0].y + m * (x - P[0].x);
      const dyOff = P[2].y - yMain(P[2].x);
      const yPar = x =&gt; yMain(x) + dyOff;
      const xL = -20, xR = LAY.plotW + 20;
      cx.beginPath();
      cx.moveTo(xL, yMain(xL)); cx.lineTo(xR, yMain(xR)); cx.lineTo(xR, yPar(xR)); cx.lineTo(xL, yPar(xL));
      cx.closePath();
      trendFill(0.12);
      trendSeg(xL, yMain(xL), xR, yMain(xR));
      trendSeg(xL, yPar(xL), xR, yPar(xR));
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    case &#x27;reg&#x27;: {
      if (P.length &lt; 2) break;
      const a = Math.min(P[0].i, P[1].i), b = Math.max(P[0].i, P[1].i);
      if (b - a &lt; 2) break;
      trendPlotClip();
      const n = b - a + 1;
      let sx = 0, sy = 0, sxy = 0, sxx = 0;
      for (let i = a; i &lt;= b; i++) { const x = i - a, y = bars[i].c; sx += x; sy += y; sxy += x * y; sxx += x * x; }
      const den = n * sxx - sx * sx;
      const slope = den ? (n * sxy - sx * sy) / den : 0;
      const icpt = (sy - slope * sx) / n;
      let ss = 0;
      for (let i = a; i &lt;= b; i++) { const r = bars[i].c - (icpt + slope * (i - a)); ss += r * r; }
      const sd = Math.sqrt(ss / n);
      const pAt = (i, k) =&gt; icpt + slope * (i - a) + k * sd;
      const edge = k =&gt; {
        cx.beginPath();
        for (let i = a; i &lt;= b; i++) { const x = xOf(i), y = yOf(pAt(i, k)); if (i === a) cx.moveTo(x, y); else cx.lineTo(x, y); }
      };
      edge(2);
      for (let i = b; i &gt;= a; i--) cx.lineTo(xOf(i), yOf(pAt(i, -2)));
      cx.closePath();
      trendFill(0.12);
      cx.save();
      cx.setLineDash([4, 3]);
      edge(2); cx.stroke();
      edge(-2); cx.stroke();
      cx.restore();
      edge(0); cx.stroke();
      break;
    }
    case &#x27;flat&#x27;: {
      if (P.length &lt; 3 || P[0].x === P[1].x) break;
      trendPlotClip();
      const x0 = Math.min(P[0].x, P[1].x), x1 = Math.max(P[0].x, P[1].x);
      const yTop = P[0].y, yBot = P[2].y;
      cx.beginPath();
      cx.rect(x0, Math.min(yTop, yBot), x1 - x0, Math.abs(yBot - yTop));
      trendFill(0.12);
      trendSeg(x0, yTop, x1, yTop);
      trendSeg(x0, yBot, x1, yBot);
      trendSeg(x0, yTop, x0, yBot);
      trendSeg(x1, yTop, x1, yBot);
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    default: break;
  }
  cx.restore();
}
function drawPending(bars) {
  if (S.mode !== &#x27;draw&#x27; || !S.pendingDraw.length) return;
  const tool = DRAW_BY_ID[S.tool];
  if (!tool) return;
  const pts = S.pendingDraw.slice();
  const c = S.cross;
  if (c &amp;&amp; c.x &gt;= 0 &amp;&amp; c.x &lt;= LAY.plotW &amp;&amp; c.y &gt;= LAY.candleTop &amp;&amp; c.y &lt;= LAY.candleBottom) {
    const i = clamp(Math.round(idxAtX(c.x)), 0, bars.length - 1);
    pts.push({ t: bars[i].t, p: r2(priceAtY(c.y)) });
  }
  cx.save();
  cx.fillStyle = DRAW_COL;
  for (const q of S.pendingDraw) { const i = nearestIndex(bars, q.t); trendDot(xOf(i), yOf(q.p)); }
  if (pts.length === tool.pts) { cx.globalAlpha = 0.6; drawTrend({ type: tool.type, pts }, bars); }
  cx.restore();
}

/* ===== Chế độ vẽ, cảm ứng, nạp dữ liệu, thay đổi kích thước ===== */
function setMode(m) {
  S.mode = m;
  selDraw = null;
  S.pendingFan = null;
  S.pendingDraw = [];
  setOn(&#x27;bFan&#x27;, m === &#x27;fan&#x27;);
  setOn(&#x27;bHl&#x27;, m === &#x27;hline&#x27;);
  setOn(&#x27;bDraw&#x27;, m === &#x27;draw&#x27;);
  setOn(&#x27;bCur&#x27;, m === &#x27;pan&#x27;);
  document.body.classList.toggle(&#x27;drawing&#x27;, m !== &#x27;pan&#x27;);
  if (m === &#x27;fan&#x27;) say(&#x27;Gann Fan: chạm điểm gốc, rồi chạm điểm thứ hai để định nghĩa 1x1.&#x27;);
  else if (m === &#x27;hline&#x27;) say(&#x27;Đường ngang: chạm mức giá muốn ghim.&#x27;);
  else if (m === &#x27;draw&#x27;) say(DRAW_BY_ID[S.tool].hint);
  else say(&#x27;Kéo để xem biểu đồ. Chạm nút vẽ để bắt đầu vẽ.&#x27;);
  schedule();
}
