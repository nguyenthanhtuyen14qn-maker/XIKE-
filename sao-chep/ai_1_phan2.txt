// Chuẩn hóa một tham số: trống hoặc sai thì lấy mặc định, ngoài khoảng thì kẹp lại.
function cleanParam(f, v) {
  const blank = v === null || v === undefined || (typeof v === &#x27;string&#x27; &amp;&amp; v.trim() === &#x27;&#x27;);
  const x = blank ? NaN : Number(v);
  const val = clamp(Number.isFinite(x) ? x : f.def, f.min, f.max);
  return f.float ? r2(val) : Math.round(val);
}

// Đọc danh sách chỉ báo đã lưu: bỏ id lạ, chuẩn hóa tham số, cấp uid mới.
function sanitizeInds(arr) {
  const out = [];
  for (const it of arr) {
    if (out.length &gt;= MAX_INDS) break;
    if (!it || typeof it !== &#x27;object&#x27; || !IND_ENABLED[it.id]) continue;
    const def = IND_BY_ID[it.id];
    const src = it.p &amp;&amp; typeof it.p === &#x27;object&#x27; ? it.p : {};
    const p = {};
    for (const f of def.params) p[f.k] = cleanParam(f, src[f.k]);
    out.push({ uid: newUid(), id: def.id, p, on: it.on !== false });
  }
  return out;
}

      mainH = 130;
      paneH = Math.max(40, Math.floor((totalH - 130) / nPanes));

    const yo = yOf(bc.o), yc = yOf(bc.c);
    cx.fillRect(xc - Math.floor(bodyW / 2), Math.min(yo, yc), bodyW, Math.max(1, Math.abs(yc - yo)));
  }

  for (const o of ovs) {
    for (const ln of o.r.lines) {
      cx.setLineDash(ln.dash || []);
      polyline(ln.v, Math.max(0, iL - 1), iR, yOf, ln.color, ln.w || (ln.thin ? 1 : 1.5));
      cx.setLineDash([]);
    }
    for (const lv of o.r.levels || []) {
      if (lv.v &lt; VP.pMin || lv.v &gt; VP.pMax) continue;
      const y = Math.round(yOf(lv.v)) + 0.5;
      cx.setLineDash([5, 4]); cx.strokeStyle = lv.color; cx.lineWidth = 1;
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      cx.fillStyle = lv.color; cx.textAlign = lv.right ? &#x27;right&#x27; : &#x27;left&#x27;;
      cx.fillText(lv.text + &#x27;  &#x27; + lv.v.toFixed(2), lv.right ? L.plotW - 8 : 8, y - 8);
      cx.textAlign = &#x27;left&#x27;;
    }
  }

  // Nhãn tín hiệu trên giá: BUY và BUY P nằm dưới nến, SELL và SELL P nằm trên nến; nhãn cùng nến được xếp chồng
  const stacked = {};
  for (const o of ovs) {
    for (const mk of o.r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      const key = mk.i + (mk.up ? &#x27;u&#x27; : &#x27;d&#x27;), k = stacked[key] || 0;
      stacked[key] = k + 1;
      const bc = bars[mk.i], xc = xOf(mk.i);
      if (mk.up) drawMark(xc, yOf(bc.l) + 6 + k * 16, mk.text, COL.up);
      else drawMark(xc, yOf(bc.h) - 20 - k * 16, mk.text, COL.down);
    }
  }

  for (const d of S.drawings) {
    if (d.type === &#x27;fan&#x27;) { drawFan(d, bars); continue; }
    if (TREND_TYPES.has(d.type)) { drawTrend(d, bars); continue; }
    if (d.type === &#x27;hline&#x27;) {
      const yh = yOf(d.p);
      if (yh &lt; L.candleTop || yh &gt; L.candleBottom) continue;
      const y = Math.round(yh) + 0.5;
      cx.strokeStyle = COL.hline; cx.lineWidth = 1; cx.setLineDash([6, 4]);
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
      cx.setLineDash([]);
      tags.push({ y: yh, text: d.p.toFixed(2), bg: COL.hline, fg: COL.bg });
    }
  }

  if (selValid() &amp;&amp; S.mode === &#x27;pan&#x27;) {
    cx.save();
    cx.shadowColor = DRAW_COL;
    cx.shadowBlur = 10;
    drawSelShape(selDraw, bars);
    cx.restore();
    drawSelTag(bars);
  }
  drawPending(bars);
  if (S.pendingFan) {
    const i = nearestIndex(bars, S.pendingFan.t);
    cx.fillStyle = COL.fan;
    cx.beginPath(); cx.arc(xOf(i), yOf(S.pendingFan.p), 4, 0, Math.PI * 2); cx.fill();
  }

  const last = bars[n - 1], lastUp = last.c &gt;= last.o;
  if (last.c &gt;= VP.pMin &amp;&amp; last.c &lt;= VP.pMax) {
    const yl = Math.round(yOf(last.c)) + 0.5;
    cx.strokeStyle = lastUp ? COL.up : COL.down; cx.lineWidth = 1; cx.setLineDash([2, 3]);
    cx.beginPath(); cx.moveTo(0, yl); cx.lineTo(L.plotW, yl); cx.stroke();
    cx.setLineDash([]);
    tags.push({ y: yOf(last.c), text: last.c.toFixed(2), bg: lastUp ? COL.up : COL.down, fg: COL.bg });
  }

  if (crInMain) {
    const yc = Math.round(cr.y) + 0.5;
    cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
    cx.setLineDash([]);
    if (cr.y &gt;= L.candleTop &amp;&amp; cr.y &lt;= L.candleBottom) {
      tags.push({ y: cr.y, text: priceAtY(cr.y).toFixed(2), bg: COL.box, fg: COL.text });
    }
  }
  cx.restore();

  // Các vùng chỉ báo dao động, mỗi vùng có thang riêng
  pns.forEach((pn, k) =&gt; {
    const def = pn.def, r = pn.r;
    const top = L.mainH + k * L.paneH, bot = top + L.paneH;
    let lo2 = Infinity, hi2 = -Infinity;
    for (const ln of r.lines) {
      for (let i = iL; i &lt;= iR; i++) {
        const v = ln.v[i];
        if (Number.isFinite(v)) {
          if (v &lt; lo2) lo2 = v;
          if (v &gt; hi2) hi2 = v;
        }
      }
    }
    for (const lv of def.levels) {
      if (lv.v &lt; lo2) lo2 = lv.v;
      if (lv.v &gt; hi2) hi2 = lv.v;
    }
    if (def.range) {
      lo2 = def.range[0];
      hi2 = def.range[1];
    } else {
      if (!Number.isFinite(lo2) || !Number.isFinite(hi2)) { lo2 = 0; hi2 = 1; }
      if (def.zero) { lo2 = Math.min(lo2, 0); hi2 = Math.max(hi2, 0); }
      if (hi2 - lo2 &lt; 1e-9) hi2 = lo2 + 1;
      const pd = (hi2 - lo2) * 0.1;
      lo2 -= pd;
      hi2 += pd;
    }
    const yP = v =&gt; top + 6 + (hi2 - v) / (hi2 - lo2) * (L.paneH - 12);

    cx.strokeStyle = COL.line; cx.lineWidth = 1;
    cx.beginPath(); cx.moveTo(0, Math.round(top) + 0.5); cx.lineTo(L.w, Math.round(top) + 0.5); cx.stroke();

    cx.save();
    cx.beginPath(); cx.rect(0, top, L.plotW, L.paneH); cx.clip();
    cx.setLineDash([4, 4]); cx.strokeStyle = COL.grid; cx.lineWidth = 1;
    for (const lv of def.levels) {
      const y = Math.round(yP(lv.v)) + 0.5;
      cx.beginPath(); cx.moveTo(0, y); cx.lineTo(L.plotW, y); cx.stroke();
    }
    cx.setLineDash([]);
    for (const ln of r.lines) {
      if (ln.type === &#x27;hist&#x27;) {
        const bw = Math.max(1, Math.round(S.barW * 0.6)), y0 = yP(0);
        for (let i = iL; i &lt;= iR; i++) {
          const v = ln.v[i];
          if (!Number.isFinite(v)) continue;
          const yv = yP(v);
          cx.fillStyle = v &gt;= 0 ? ln.up : ln.dn;
          cx.fillRect(Math.round(xOf(i)) - Math.floor(bw / 2), Math.min(y0, yv), bw, Math.max(1, Math.abs(yv - y0)));
        }
      } else {
        polyline(ln.v, Math.max(0, iL - 1), iR, yP, ln.color, ln.thin ? 1 : 1.4);
      }
    }
    // Nhãn Buy / Sell của TRIX: Buy nằm sát đáy ô, Sell nằm sát đỉnh ô
    for (const mk of r.marks || []) {
      if (mk.i &lt; iL || mk.i &gt; iR) continue;
      drawMark(xOf(mk.i), mk.up ? bot - 17 : top + 3, mk.text, mk.up ? COL.up : COL.down);
    }
    if (crInPlot &amp;&amp; cr.y &gt;= top &amp;&amp; cr.y &lt; bot) {
      const yc = Math.round(cr.y) + 0.5;
      cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
      cx.beginPath(); cx.moveTo(0, yc); cx.lineTo(L.plotW, yc); cx.stroke();
      cx.setLineDash([]);
      const vv = hi2 - (cr.y - top - 6) / (L.paneH - 12) * (hi2 - lo2);
      tags.push({ y: cr.y, text: fmtVal(vv, def.dec), bg: COL.box, fg: COL.text });
    }
    cx.restore();

    // Nhãn trục phải: các mức tham chiếu và mốc đẹp
    const pstep = niceStep(hi2 - lo2, 3);
    const pdec = clamp(-Math.floor(Math.log10(pstep)), 0, 4);
    const levelYs = [];
    for (const lv of def.levels) {
      const y = yP(lv.v);
      levelYs.push(y);
      axis.push({ y: y, text: fmtVal(lv.v, pdec), color: COL.muted });
    }
    for (let kk = Math.ceil(lo2 / pstep); kk * pstep &lt;= hi2; kk++) {
      const y = yP(kk * pstep);
      if (y &lt; top + 4 || y &gt; bot - 4) continue;
      if (levelYs.some(ly =&gt; Math.abs(ly - y) &lt; 12)) continue;
      axis.push({ y: y, text: (kk * pstep).toFixed(pdec), color: COL.muted });
    }
  });

  // Đường dọc của con trỏ, trên toàn khung
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const xl = Math.round(cr.x) + 0.5;
    cx.save();
    cx.beginPath(); cx.rect(0, 0, L.plotW, L.totalH); cx.clip();
    cx.strokeStyle = COL.cross; cx.lineWidth = 1; cx.setLineDash([3, 3]);
    cx.beginPath(); cx.moveTo(xl, 0); cx.lineTo(xl, L.totalH); cx.stroke();
    cx.setLineDash([]);
    cx.restore();
  }

  // Nhãn trục (ngoài vùng cắt xén)
  cx.textAlign = &#x27;left&#x27;;
  for (const t of axis) {
    cx.fillStyle = t.color;
    cx.fillText(t.text, L.plotW + 8, t.y);
  }
  for (const t of tags) {
    cx.fillStyle = t.bg; cx.fillRect(L.plotW, t.y - 9, L.axisR, 18);
    cx.fillStyle = t.fg; cx.fillText(t.text, L.plotW + 6, t.y);
  }
  if (crInPlot &amp;&amp; cr.y &gt;= 0 &amp;&amp; cr.y &lt;= L.totalH) {
    const txt = fmtTs(timeAtIdx(bars, ser.sec, idxAtX(cr.x)), &#x27;dhm&#x27;);
    cx.font = smallFont();
    const tw = cx.measureText(txt).width + 14;
    const bx = clamp(cr.x - tw / 2, 0, Math.max(0, L.plotW - tw));
    cx.fillStyle = COL.box; cx.fillRect(bx, L.totalH, tw, L.axisB);
    cx.fillStyle = COL.text; cx.textAlign = &#x27;center&#x27;;
    cx.fillText(txt, bx + tw / 2, L.totalH + L.axisB / 2);
  }

  const cur = bars[li], prv = bars[li &gt; 0 ? li - 1 : li];
  updateReadout(cur, prv);
  updateLegend(li);
}

// Chú thích trên góc trái: nến đang trỏ tới và giá trị của từng chỉ báo. Chạm một dòng để chỉnh chỉ báo đó.
function updateLegend(li) {
  const box = $(&#x27;legend&#x27;);
  if (!box) return;
  if (!hasData) {
    if (legendKey !== &#x27;empty&#x27;) { box.textContent = &#x27;&#x27;; legendKey = &#x27;empty&#x27;; }
    return;
  }
  const key = [li, S.tf, dataGen, indsVer, S.showBlock, S.tz].join(&#x27;|&#x27;);
  if (key === legendKey) return;
  legendKey = key;
  const b = series(S.tf).bars[li];
  const up = b.c &gt;= b.o;
  const parts = [];
  parts.push(&#x27;&lt;div class=&quot;lg&quot;&gt;&lt;b&gt;&#x27; + S.tf + &#x27;&lt;/b&gt; &#x27; + fmtTs(b.t, &#x27;dhm&#x27;) +
    (S.showBlock &amp;&amp; inBlock(b.t) ? &#x27; &lt;span class=&quot;blk&quot;&gt;Giờ cấm 18–22h VN&lt;/span&gt;&#x27; : &#x27;&#x27;) + &#x27;&lt;/div&gt;&#x27;);
  parts.push(&#x27;&lt;div class=&quot;lg&quot;&gt;O &lt;b&gt;&#x27; + b.o.toFixed(2) + &#x27;&lt;/b&gt; H &lt;b&gt;&#x27; + b.h.toFixed(2) +
    &#x27;&lt;/b&gt; L &lt;b&gt;&#x27; + b.l.toFixed(2) + &#x27;&lt;/b&gt; C &lt;b style=&quot;color:&#x27; + (up ? COL.up : COL.down) + &#x27;&quot;&gt;&#x27; +
    b.c.toFixed(2) + &#x27;&lt;/b&gt;&lt;/div&gt;&#x27;);
  for (const inst of S.inds) {
    if (!inst.on) continue;
    const def = IND_BY_ID[inst.id];
    let vals = &#x27;&#x27;;
    if (def.volume) {
      vals = &#x27;&lt;b&gt;&#x27; + fmtCompact(b.v) + &#x27;&lt;/b&gt;&#x27;;
    } else {
      const r = instResult(inst);
      const vs = r.lines.filter(ln =&gt; ln.legend !== false).map(ln =&gt; &#x27;&lt;span style=&quot;color:&#x27; + (ln.color || ln.up) + &#x27;&quot;&gt;&#x27; + fmtVal(ln.v[li], def.dec) + &#x27;&lt;/span&gt;&#x27;);
      if (r.info) vs.push(&#x27;&lt;span&gt;&#x27; + r.info + &#x27;&lt;/span&gt;&#x27;);
      vals = vs.join(&#x27; &#x27;);
    }
    parts.push(&#x27;&lt;div class=&quot;lg&quot; data-uid=&quot;&#x27; + inst.uid + &#x27;&quot;&gt;&lt;i class=&quot;sw&quot; style=&quot;background:&#x27; + def.color +
      &#x27;&quot;&gt;&lt;/i&gt;&lt;b&gt;&#x27; + def.label(inst.p) + &#x27;&lt;/b&gt; &#x27; + vals + &#x27;&lt;/div&gt;&#x27;);
  }
  box.innerHTML = parts.join(&#x27;&#x27;);
}

function schedule() {
  if (!raf) raf = requestAnimationFrame(() =&gt; { raf = 0; render(); });
}

/* ===== Quản lý chỉ báo: lưu, thêm, bỏ, và bảng chọn kiểu bottom sheet ===== */
function saveInds() {
  try {
    localStorage.setItem(IND_KEY, JSON.stringify(S.inds.map(x =&gt; ({ id: x.id, p: x.p, on: x.on }))));
  } catch (err) { /* không có bộ nhớ trình duyệt */ }
}

function loadInds() {
  try {
    const v = JSON.parse(localStorage.getItem(IND_KEY) || &#x27;null&#x27;);
    if (Array.isArray(v)) { S.inds = sanitizeInds(v); return; }
  } catch (err) { /* bỏ qua dữ liệu hỏng */ }
  S.inds = defaultInds();
}

function touchInds() {
  indsVer += 1;
  saveInds();
  LAY = layout();
  updateIndBadge();
  if (!$(&#x27;libView&#x27;).hidden) renderActive();
  schedule();
}

function updateIndBadge() {
  const n = S.inds.filter(x =&gt; x.on).length;
  const bd = $(&#x27;indCount&#x27;);
  bd.textContent = String(n);
  bd.hidden = n === 0;
}

function addInd(id) {
  const def = IND_BY_ID[id];
  if (!def || !IND_ENABLED[id]) return;
  if (S.inds.length &gt;= MAX_INDS) { say(&#x27;Tối đa &#x27; + MAX_INDS + &#x27; chỉ báo. Hãy bỏ bớt trước khi thêm.&#x27;); return; }
  if (def.volume &amp;&amp; S.inds.some(x =&gt; x.id === &#x27;volume&#x27;)) { say(&#x27;Khối lượng đã có trên biểu đồ.&#x27;); return; }
  const p = {};
  for (const f of def.params) p[f.k] = f.def;
  S.inds.push({ uid: newUid(), id: def.id, p, on: true });
  touchInds();
  say(&#x27;Đã thêm &#x27; + def.label(p) + &#x27;.&#x27;);
  closeSheet();
}

function removeInd(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  S.inds = S.inds.filter(x =&gt; x.uid !== uid);
  touchInds();
  say(&#x27;Đã bỏ &#x27; + IND_BY_ID[inst.id].label(inst.p) + &#x27; khỏi biểu đồ.&#x27;);
  if (editUid === uid) showSheetView(&#x27;lib&#x27;);
}

function removeEditing() { if (editUid !== null) removeInd(editUid); }

    list.appendChild(li);
  }

  const lib = which === &#x27;lib&#x27;;

function openEditor(uid) {
  const inst = S.inds.find(x =&gt; x.uid === uid);
  if (!inst) return;
  const def = IND_BY_ID[inst.id];
  editUid = uid;
  const box = $(&#x27;edFields&#x27;);
  box.textContent = &#x27;&#x27;;
  for (const f of def.params) {

function trendSeg(x0, y0, x1, y1) { cx.beginPath(); cx.moveTo(x0, y0); cx.lineTo(x1, y1); cx.stroke(); }

function drawTrend(d, bars) {
  const P = trendPts(d, bars);
  if (!P.length) return;
  const K = 1000;
  cx.save();
  cx.strokeStyle = DRAW_COL;
  cx.fillStyle = DRAW_COL;
  cx.lineWidth = 1.5;
  switch (d.type) {
    case &#x27;tl&#x27;:
    case &#x27;info&#x27;:
    case &#x27;angle&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, P[1].x, P[1].y);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      if (d.type === &#x27;info&#x27;) {
        const dp = P[1].p - P[0].p;
        const pct = P[0].p ? dp / P[0].p * 100 : 0;
        const nBars = Math.abs(P[1].i - P[0].i);
        trendTag((P[0].x + P[1].x) / 2, (P[0].y + P[1].y) / 2 - 14, (dp &gt;= 0 ? &#x27;+&#x27; : &#x27;&#x27;) + dp.toFixed(2) + &#x27; (&#x27; + pct.toFixed(2) + &#x27;%) · &#x27; + nBars + &#x27; nến&#x27;);
      } else if (d.type === &#x27;angle&#x27;) {
        const deg = Math.atan2(-(P[1].y - P[0].y), Math.abs(P[1].x - P[0].x)) * 180 / Math.PI;
        trendTag(P[1].x, P[1].y - 14, deg.toFixed(1) + &#x27;°&#x27;);
      }
      break;
    }
    case &#x27;ext&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x - dx * K, P[0].y - dy * K, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      trendDot(P[1].x, P[1].y);
      break;
    }
    case &#x27;ray&#x27;: {
      if (P.length &lt; 2) break;
      trendPlotClip();
      const dx = P[1].x - P[0].x, dy = P[1].y - P[0].y;
      trendSeg(P[0].x, P[0].y, P[0].x + dx * K, P[0].y + dy * K);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;hray&#x27;: {
      trendPlotClip();
      trendSeg(P[0].x, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;vl&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      break;
    }
    case &#x27;cross&#x27;: {
      trendFullClip();
      trendSeg(P[0].x, 0, P[0].x, LAY.totalH);
      trendSeg(0, P[0].y, LAY.plotW, P[0].y);
      trendDot(P[0].x, P[0].y);
      break;
    }
    case &#x27;chan&#x27;: {
      if (P.length &lt; 3 || P[1].x === P[0].x) break;
      trendPlotClip();
      const m = (P[1].y - P[0].y) / (P[1].x - P[0].x);
      const yMain = x =&gt; P[0].y + m * (x - P[0].x);
      const dyOff = P[2].y - yMain(P[2].x);
      const yPar = x =&gt; yMain(x) + dyOff;
      const xL = -20, xR = LAY.plotW + 20;
      cx.beginPath();
      cx.moveTo(xL, yMain(xL)); cx.lineTo(xR, yMain(xR)); cx.lineTo(xR, yPar(xR)); cx.lineTo(xL, yPar(xL));
      cx.closePath();
      trendFill(0.12);
      trendSeg(xL, yMain(xL), xR, yMain(xR));
      trendSeg(xL, yPar(xL), xR, yPar(xR));
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    case &#x27;reg&#x27;: {
      if (P.length &lt; 2) break;
      const a = Math.min(P[0].i, P[1].i), b = Math.max(P[0].i, P[1].i);
      if (b - a &lt; 2) break;
      trendPlotClip();
      const n = b - a + 1;
      let sx = 0, sy = 0, sxy = 0, sxx = 0;
      for (let i = a; i &lt;= b; i++) { const x = i - a, y = bars[i].c; sx += x; sy += y; sxy += x * y; sxx += x * x; }
      const den = n * sxx - sx * sx;
      const slope = den ? (n * sxy - sx * sy) / den : 0;
      const icpt = (sy - slope * sx) / n;
      let ss = 0;
      for (let i = a; i &lt;= b; i++) { const r = bars[i].c - (icpt + slope * (i - a)); ss += r * r; }
      const sd = Math.sqrt(ss / n);
      const pAt = (i, k) =&gt; icpt + slope * (i - a) + k * sd;
      const edge = k =&gt; {
        cx.beginPath();
        for (let i = a; i &lt;= b; i++) { const x = xOf(i), y = yOf(pAt(i, k)); if (i === a) cx.moveTo(x, y); else cx.lineTo(x, y); }
      };
      edge(2);
      for (let i = b; i &gt;= a; i--) cx.lineTo(xOf(i), yOf(pAt(i, -2)));
      cx.closePath();
      trendFill(0.12);
      cx.save();
      cx.setLineDash([4, 3]);
      edge(2); cx.stroke();
      edge(-2); cx.stroke();
      cx.restore();
      edge(0); cx.stroke();
      break;
    }
    case &#x27;flat&#x27;: {
      if (P.length &lt; 3 || P[0].x === P[1].x) break;
      trendPlotClip();
      const x0 = Math.min(P[0].x, P[1].x), x1 = Math.max(P[0].x, P[1].x);
      const yTop = P[0].y, yBot = P[2].y;
      cx.beginPath();
      cx.rect(x0, Math.min(yTop, yBot), x1 - x0, Math.abs(yBot - yTop));
      trendFill(0.12);
      trendSeg(x0, yTop, x1, yTop);
      trendSeg(x0, yBot, x1, yBot);
      trendSeg(x0, yTop, x0, yBot);
      trendSeg(x1, yTop, x1, yBot);
      trendDot(P[0].x, P[0].y); trendDot(P[1].x, P[1].y); trendDot(P[2].x, P[2].y);
      break;
    }
    default: break;
  }
  cx.restore();
}
function drawPending(bars) {
  if (S.mode !== &#x27;draw&#x27; || !S.pendingDraw.length) return;
  const tool = DRAW_BY_ID[S.tool];
  if (!tool) return;
  const pts = S.pendingDraw.slice();
  const c = S.cross;
  if (c &amp;&amp; c.x &gt;= 0 &amp;&amp; c.x &lt;= LAY.plotW &amp;&amp; c.y &gt;= LAY.candleTop &amp;&amp; c.y &lt;= LAY.candleBottom) {
    const i = clamp(Math.round(idxAtX(c.x)), 0, bars.length - 1);
    pts.push({ t: bars[i].t, p: r2(priceAtY(c.y)) });
  }
  cx.save();
  cx.fillStyle = DRAW_COL;
  for (const q of S.pendingDraw) { const i = nearestIndex(bars, q.t); trendDot(xOf(i), yOf(q.p)); }
  if (pts.length === tool.pts) { cx.globalAlpha = 0.6; drawTrend({ type: tool.type, pts }, bars); }
  cx.restore();
}

const DEL_TOL = 12;
let lastDel = null, undoTimer = 0;
function segDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len2 = dx * dx + dy * dy;
  const t = len2 ? clamp(((px - x0) * dx + (py - y0) * dy) / len2, 0, 1) : 0;
  return Math.hypot(px - (x0 + t * dx), py - (y0 + t * dy));
}
function rayDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len2 = dx * dx + dy * dy;
  if (!len2) return Math.hypot(px - x0, py - y0);
  const t = ((px - x0) * dx + (py - y0) * dy) / len2;
  if (t &lt;= 0) return Math.hypot(px - x0, py - y0);
  return Math.hypot(px - (x0 + t * dx), py - (y0 + t * dy));
}
function lineDist(px, py, x0, y0, x1, y1) {
  const dx = x1 - x0, dy = y1 - y0, len = Math.hypot(dx, dy);
  if (!len) return Math.hypot(px - x0, py - y0);
  return Math.abs(dx * (py - y0) - dy * (px - x0)) / len;
}
function regDist(P, bars, px, py) {
  const a = Math.min(P[0].i, P[1].i), b = Math.max(P[0].i, P[1].i);
  if (b - a &lt; 2) return Infinity;
  const n = b - a + 1;
  let sx = 0, sy = 0, sxy = 0, sxx = 0;
  for (let i = a; i &lt;= b; i++) { const x = i - a, y = bars[i].c; sx += x; sy += y; sxy += x * y; sxx += x * x; }
  const den = n * sxx - sx * sx;
  const slope = den ? (n * sxy - sx * sy) / den : 0;
  const icpt = (sy - slope * sx) / n;
  let ss = 0;
  for (let i = a; i &lt;= b; i++) { const r = bars[i].c - (icpt + slope * (i - a)); ss += r * r; }
  const sd = Math.sqrt(ss / n);
  let best = Infinity;
  for (const k of [-2, 0, 2]) {
    for (let i = a; i &lt; b; i++) {
      const ya = yOf(icpt + slope * (i - a) + k * sd), yb = yOf(icpt + slope * (i + 1 - a) + k * sd);
      best = Math.min(best, segDist(px, py, xOf(i), ya, xOf(i + 1), yb));
    }
  }
  return best;
}
function drawingDist(d, bars, px, py) {
  if (d.type === &#x27;hline&#x27;) {
    const yh = yOf(d.p);
    return yh &lt; LAY.candleTop || yh &gt; LAY.candleBottom ? Infinity : Math.abs(py - yh);
  }
  if (d.type === &#x27;fan&#x27;) {
    const i1 = nearestIndex(bars, d.p1.t), i2 = nearestIndex(bars, d.p2.t);
    if (i1 === i2 || xOf(i1) &gt; LAY.plotW) return Infinity;
    const x1 = xOf(i1), y1 = yOf(d.p1.p), perBar = (d.p2.p - d.p1.p) / (i2 - i1), iR = idxAtX(LAY.plotW);
    let best = Infinity;
    for (const r of FAN_RATIOS) best = Math.min(best, segDist(px, py, x1, y1, LAY.plotW, yOf(d.p1.p + perBar * r * (iR - i1))));
    return best;
  }
  if (!TREND_TYPES.has(d.type)) return Infinity;
  const P = trendPts(d, bars);
  if (!P.length) return Infinity;
  switch (d.type) {
    case &#x27;tl&#x27;: case &#x27;info&#x27;: case &#x27;angle&#x27;:
      return P.length &lt; 2 ? Infinity : segDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;ext&#x27;:
      return P.length &lt; 2 ? Infinity : lineDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;ray&#x27;:
      return P.length &lt; 2 ? Infinity : rayDist(px, py, P[0].x, P[0].y, P[1].x, P[1].y);
    case &#x27;hray&#x27;:
      return segDist(px, py, P[0].x, P[0].y, LAY.plotW, P[0].y);
    case &#x27;vl&#x27;:
      return Math.abs(px - P[0].x);
    case &#x27;cross&#x27;:
      return Math.min(Math.abs(px - P[0].x), Math.abs(py - P[0].y));
    case &#x27;chan&#x27;: {
      if (P.length &lt; 3 || P[1].x === P[0].x) return Infinity;
      const m = (P[1].y - P[0].y) / (P[1].x - P[0].x);
      const yMain = x =&gt; P[0].y + m * (x - P[0].x);
      const dyOff = P[2].y - yMain(P[2].x);
      const xL = -20, xR = LAY.plotW + 20;
      return Math.min(segDist(px, py, xL, yMain(xL), xR, yMain(xR)), segDist(px, py, xL, yMain(xL) + dyOff, xR, yMain(xR) + dyOff));
    }
    case &#x27;reg&#x27;:
      return regDist(P, bars, px, py);
    case &#x27;flat&#x27;: {
      if (P.length &lt; 3 || P[0].x === P[1].x) return Infinity;
      const x0 = Math.min(P[0].x, P[1].x), x1 = Math.max(P[0].x, P[1].x), yTop = P[0].y, yBot = P[2].y;
      return Math.min(segDist(px, py, x0, yTop, x1, yTop), segDist(px, py, x0, yBot, x1, yBot), segDist(px, py, x0, yTop, x0, yBot), segDist(px, py, x1, yTop, x1, yBot));
    }
    default: return Infinity;
  }
}

  });
  return best;
}

      say(&#x27;Đã vẽ Gann Fan. Bấm Gann Fan để vẽ thêm.&#x27;);
    }
  }
  schedule();
}

function bindPointer() {
  const pts = new Map();
  let drag = null, pinch = null;
  const rel = e =&gt; {
    const r = cv.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top };
  };

  cv.addEventListener(&#x27;pointerdown&#x27;, e =&gt; {
    e.preventDefault();
    try { cv.setPointerCapture(e.pointerId); } catch (err) { /* không bắt được con trỏ */ }
    const p = rel(e);
    pts.set(e.pointerId, p);
    if (pts.size === 1) {
      drag = { x0: p.x, y0: p.y, a0: S.anchor, moved: false };
      pinch = null;
      S.cross = p;
    } else {
      // Hai ngón: không còn là một lần chạm. Co giãn ở chế độ vẽ chỉ bắt đầu khi hai ngón thật sự kéo đi.
      drag = null;
      if (pts.size === 2 &amp;&amp; hasData) {
        const [p1, p2] = [...pts.values()];
        pinch = {
          d0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
          w0: S.barW,
          iMid: idxAtX((p1.x + p2.x) / 2),
          mx0: (p1.x + p2.x) / 2,
          my0: (p1.y + p2.y) / 2,
          armed: S.mode === &#x27;pan&#x27;
        };
      }
    }
    schedule();
  });

  cv.addEventListener(&#x27;pointermove&#x27;, e =&gt; {
    const p = rel(e);
    if (pts.has(e.pointerId)) pts.set(e.pointerId, p);
    else if (e.pointerType !== &#x27;mouse&#x27;) return;
    if (!hasData) return;
    if (pinch &amp;&amp; pts.size &gt;= 2) {
      const [p1, p2] = [...pts.values()];
      const mx = (p1.x + p2.x) / 2, my = (p1.y + p2.y) / 2;
      const d = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
      if (!pinch.armed) {
        // Chế độ vẽ: ngón thứ hai chạm nhẹ không được làm biểu đồ phóng to hay trôi đi
        if (Math.abs(d - pinch.d0) &lt; 20 &amp;&amp; Math.hypot(mx - pinch.mx0, my - pinch.my0) &lt; 20) return;
        pinch.armed = true;
        pinch.d0 = d;
        pinch.w0 = S.barW;
        pinch.iMid = idxAtX(mx);
      }
      S.barW = clamp(pinch.w0 * d / pinch.d0, 1.5, 40);
      S.anchor = pinch.iMid - 0.5 + (LAY.plotW - mx) / S.barW;
      clampAnchor();
      S.cross = null;
    } else if (drag &amp;&amp; pts.size === 1) {
      // Chế độ vẽ/Gann/đường ngang: một ngón chỉ để ngắm và đặt điểm, không kéo biểu đồ
      const dx = p.x - drag.x0, dy = p.y - drag.y0;
      if (!drag.moved &amp;&amp; Math.hypot(dx, dy) &gt; 6) drag.moved = true;
      if (drag.moved &amp;&amp; S.mode === &#x27;pan&#x27;) {
        S.anchor = drag.a0 - dx / S.barW;
        clampAnchor();
      }
      S.cross = p;
    } else if (e.pointerType === &#x27;mouse&#x27;) {
      S.cross = p;
    }
    schedule();
  });

  const end = (e, cancelled) =&gt; {
    pts.delete(e.pointerId);
    if (drag &amp;&amp; pts.size === 0) {
      // Chế độ vẽ: điểm đặt ở chỗ nhấc ngón; dấu thập đã theo ngón tay lúc đang giữ
      if (!cancelled &amp;&amp; hasData &amp;&amp; (S.mode !== &#x27;pan&#x27; || !drag.moved)) {
        const q = S.mode === &#x27;pan&#x27; ? { x: drag.x0, y: drag.y0 } : rel(e);
        onTap(q.x, q.y);
      }
      drag = null;
    }
    if (pts.size &lt; 2) pinch = null;
    if (pts.size === 0 &amp;&amp; e.pointerType !== &#x27;mouse&#x27;) S.cross = null;
    schedule();
  };
  cv.addEventListener(&#x27;pointerup&#x27;, e =&gt; end(e, false));
  cv.addEventListener(&#x27;pointercancel&#x27;, e =&gt; end(e, true));
  cv.addEventListener(&#x27;pointerleave&#x27;, e =&gt; {
    if (e.pointerType === &#x27;mouse&#x27; &amp;&amp; pts.size === 0) {
      S.cross = null;
      schedule();
    }
  });
  cv.addEventListener(&#x27;wheel&#x27;, e =&gt; {
    if (!hasData) return;
    e.preventDefault();
    const p = rel(e);
    zoomAt(p.x, Math.exp(-e.deltaY * 0.0015));
    S.cross = p;
    schedule();
  }, { passive: false });
}

    const v = JSON.parse(localStorage.getItem(LS_KEY) || &#x27;[]&#x27;);
    if (Array.isArray(v)) S.drawings = v;
  } catch (err) { /* bỏ qua dữ liệu hỏng */ }
}

function resize() {
  if (!wrapEl || !cv) return;
  const r = wrapEl.getBoundingClientRect();
  view.w = Math.max(160, Math.floor(r.width));
  view.h = Math.max(160, Math.floor(r.height));
  cv.width = Math.floor(view.w * DPR);
  cv.height = Math.floor(view.h * DPR);
  cv.style.width = view.w + &#x27;px&#x27;;
  cv.style.height = view.h + &#x27;px&#x27;;
  LAY = layout();
  if (hasData) clampAnchor();
  legendKey = &#x27;&#x27;;
  schedule();
}
