function bindPointer() {
  const pts = new Map();
  let drag = null, pinch = null;
  const rel = e =&gt; {
    const r = cv.getBoundingClientRect();
    return { x: e.clientX - r.left, y: e.clientY - r.top };
  };

  cv.addEventListener(&#x27;pointerdown&#x27;, e =&gt; {
    e.preventDefault();
    try { cv.setPointerCapture(e.pointerId); } catch (err) { /* không bắt được con trỏ */ }
    const p = rel(e);
    pts.set(e.pointerId, p);
    if (pts.size === 1) {
      drag = {
        x0: p.x, y0: p.y, a0: S.anchor, moved: false,
        // Kéo trong ô giá: dọc = dời thang giá; kéo trên cột giá bên phải = co giãn thang giá
        inMain: p.y &lt; LAY.candleBottom + 6, onAxis: p.x &gt; LAY.plotW &amp;&amp; p.y &lt; LAY.candleBottom + 6,
        vp0: { pMin: VP.pMin, pMax: VP.pMax }, vert: false
      };
      pinch = null;
      S.cross = p;
    } else {
      // Hai ngón: không còn là một lần chạm. Co giãn ở chế độ vẽ chỉ bắt đầu khi hai ngón thật sự kéo đi.
      drag = null;
      if (pts.size === 2 &amp;&amp; hasData) {
        const [p1, p2] = [...pts.values()];
        pinch = {
          d0: Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1,
          w0: S.barW,
          iMid: idxAtX((p1.x + p2.x) / 2),
          mx0: (p1.x + p2.x) / 2,
          my0: (p1.y + p2.y) / 2,
          armed: S.mode === &#x27;pan&#x27;
        };
      }
    }
    schedule();
  });

  cv.addEventListener(&#x27;pointermove&#x27;, e =&gt; {
    const p = rel(e);
    if (pts.has(e.pointerId)) pts.set(e.pointerId, p);
    else if (e.pointerType !== &#x27;mouse&#x27;) return;
    if (!hasData) return;
    if (pinch &amp;&amp; pts.size &gt;= 2) {
      const [p1, p2] = [...pts.values()];
      const mx = (p1.x + p2.x) / 2, my = (p1.y + p2.y) / 2;
      const d = Math.hypot(p1.x - p2.x, p1.y - p2.y) || 1;
      if (!pinch.armed) {
        // Chế độ vẽ: ngón thứ hai chạm nhẹ không được làm biểu đồ phóng to hay trôi đi
        if (Math.abs(d - pinch.d0) &lt; 20 &amp;&amp; Math.hypot(mx - pinch.mx0, my - pinch.my0) &lt; 20) return;
        pinch.armed = true;
        pinch.d0 = d;
        pinch.w0 = S.barW;
        pinch.iMid = idxAtX(mx);
      }
      S.barW = clamp(pinch.w0 * d / pinch.d0, 1.5, 40);
      S.anchor = pinch.iMid - 0.5 + (LAY.plotW - mx) / S.barW;
      clampAnchor();
      S.cross = null;
    } else if (drag &amp;&amp; pts.size === 1) {
      // Chế độ vẽ/Gann/đường ngang: một ngón chỉ để ngắm và đặt điểm, không kéo biểu đồ
      const dx = p.x - drag.x0, dy = p.y - drag.y0;
      if (!drag.moved &amp;&amp; Math.hypot(dx, dy) &gt; 6) drag.moved = true;
      if (drag.moved &amp;&amp; S.mode === &#x27;pan&#x27;) {
        const ppx = (drag.vp0.pMax - drag.vp0.pMin) / Math.max(1, LAY.candleBottom - LAY.candleTop);
        if (drag.onAxis) {
          // Cột giá: kéo lên = giãn nến (thu hẹp khoảng giá), kéo xuống = co lại
          const k = Math.exp(dy * 0.006), mid = (drag.vp0.pMin + drag.vp0.pMax) / 2, half = (drag.vp0.pMax - drag.vp0.pMin) / 2 * k;
          Y_LOCK = { pMin: mid - half, pMax: mid + half };
        } else {
          S.anchor = drag.a0 - dx / S.barW;
          clampAnchor();
          if (drag.inMain &amp;&amp; (drag.vert || Math.abs(dy) &gt; 6)) {
            drag.vert = true;
            Y_LOCK = { pMin: drag.vp0.pMin + dy * ppx, pMax: drag.vp0.pMax + dy * ppx };
          }
        }
      }
      S.cross = p;
    } else if (e.pointerType === &#x27;mouse&#x27;) {
      S.cross = p;
    }
    schedule();
  });

  const end = (e, cancelled) =&gt; {
    pts.delete(e.pointerId);
    if (drag &amp;&amp; pts.size === 0) {
      // Chế độ vẽ: điểm đặt ở chỗ nhấc ngón; dấu thập đã theo ngón tay lúc đang giữ
      if (!cancelled &amp;&amp; hasData &amp;&amp; (S.mode !== &#x27;pan&#x27; || !drag.moved)) {
        const q = S.mode === &#x27;pan&#x27; ? { x: drag.x0, y: drag.y0 } : rel(e);
        // Chạm vào cột giá khi đang chỉnh tay: trở lại tự co giãn
        if (drag.onAxis &amp;&amp; Y_LOCK) yAuto();
        else onTap(q.x, q.y);
      }
      drag = null;
    }
    if (pts.size &lt; 2) pinch = null;
    if (pts.size === 0 &amp;&amp; e.pointerType !== &#x27;mouse&#x27;) S.cross = null;
    schedule();
  };
  cv.addEventListener(&#x27;pointerup&#x27;, e =&gt; end(e, false));
  cv.addEventListener(&#x27;pointercancel&#x27;, e =&gt; end(e, true));
  cv.addEventListener(&#x27;pointerleave&#x27;, e =&gt; {
    if (e.pointerType === &#x27;mouse&#x27; &amp;&amp; pts.size === 0) {
      S.cross = null;
      schedule();
    }
  });
  cv.addEventListener(&#x27;wheel&#x27;, e =&gt; {
    if (!hasData) return;
    e.preventDefault();
    const p = rel(e);
    zoomAt(p.x, Math.exp(-e.deltaY * 0.0015));
    S.cross = p;
    schedule();
  }, { passive: false });
}

// Trả về { order } nếu hợp lệ, hoặc { error }
function ordRead(side) {
  const cfg = acctStored();
  const base = acctBase(cfg.url), token = String(cfg.token || &#x27;&#x27;);
  if (!base || !token) return { error: &#x27;Chưa kết nối VPS. Nhập địa chỉ và token ở trên trước.&#x27; };
  const symbol = $(&#x27;ordSym&#x27;).value.trim();   // giữ hoa/thường: Exness Standard dùng XAUUSDm
  if (!/^[A-Za-z0-9._#-]{3,20}$/.test(symbol)) return { error: &#x27;Mã không hợp lệ.&#x27; };
  const vol = ordNum(&#x27;ordVol&#x27;), max = ordNum(&#x27;ordMax&#x27;), sl = ordNum(&#x27;ordSl&#x27;), tp = ordNum(&#x27;ordTp&#x27;);
  if (!(vol &gt; 0)) return { error: &#x27;Khối lượng phải lớn hơn 0.&#x27; };
  if (Math.abs(vol * 100 - Math.round(vol * 100)) &gt; 1e-6) return { error: &#x27;Khối lượng chỉ lấy 2 số lẻ (bước 0.01).&#x27; };
  if (!(max &gt; 0)) return { error: &#x27;Lot tối đa phải lớn hơn 0.&#x27; };
  if (vol &gt; max + 1e-9) return { error: &#x27;Khối lượng &#x27; + vol + &#x27; vượt lot tối đa &#x27; + max + &#x27;.&#x27; };
  if (sl === null) return { error: &#x27;Phải có SL.&#x27; };
  if (!(sl &gt; 0)) return { error: &#x27;SL không hợp lệ.&#x27; };
  if (tp !== null &amp;&amp; !(tp &gt; 0)) return { error: &#x27;TP không hợp lệ.&#x27; };
  if (tp !== null) {
    if (side === &#x27;BUY&#x27; &amp;&amp; !(tp &gt; sl)) return { error: &#x27;Lệnh BUY: TP phải cao hơn SL.&#x27; };
    if (side === &#x27;SELL&#x27; &amp;&amp; !(tp &lt; sl)) return { error: &#x27;Lệnh SELL: TP phải thấp hơn SL.&#x27; };
  }
  const px = ordChartPrice();
  if (px !== null) {
    if (side === &#x27;BUY&#x27; &amp;&amp; !(sl &lt; px)) return { error: &#x27;Lệnh BUY: SL phải thấp hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (side === &#x27;SELL&#x27; &amp;&amp; !(sl &gt; px)) return { error: &#x27;Lệnh SELL: SL phải cao hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (tp !== null &amp;&amp; side === &#x27;BUY&#x27; &amp;&amp; !(tp &gt; px)) return { error: &#x27;Lệnh BUY: TP phải cao hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
    if (tp !== null &amp;&amp; side === &#x27;SELL&#x27; &amp;&amp; !(tp &lt; px)) return { error: &#x27;Lệnh SELL: TP phải thấp hơn giá hiện tại &#x27; + px + &#x27;.&#x27; };
  }
  const note = $(&#x27;ordNote&#x27;).value.trim().replace(/[^\w .-]/g, &#x27;&#x27;).slice(0, 20);
  return { base, token, px, order: { symbol, side, volume: Math.round(vol * 100) / 100, sl, tp: tp === null ? 0 : tp, comment: note || &#x27;XIKE&#x27; } };
}
// Gửi POST tới VPS. Không tự thử lại: lỗi mạng có thể là lệnh đã khớp mà chưa kịp trả lời.
async function ordPost(base, token, path, body) {
  const ctl = new AbortController();
  const timer = setTimeout(() =&gt; ctl.abort(), 15000);
  try {
    const res = await fetch(base + path, {
      method: &#x27;POST&#x27;, cache: &#x27;no-store&#x27;, signal: ctl.signal,
      headers: { &#x27;X-Token&#x27;: token, &#x27;Content-Type&#x27;: &#x27;application/json&#x27; },
      body: JSON.stringify(body)
    });
    const data = await res.json().catch(() =&gt; ({}));
    if (res.status === 401) throw new Error(&#x27;Sai token.&#x27;);
    if (res.status === 404 || res.status === 405) throw new Error(&#x27;VPS chưa có cổng &#x27; + path + &#x27;. Cần cập nhật code VPS.&#x27;);
    if (!res.ok || data.ok === false) {
      const why = data.error || data.comment || (&#x27;VPS trả lỗi &#x27; + res.status);
      throw new Error(String(why) + (data.retcode ? &#x27; (mã &#x27; + data.retcode + &#x27;)&#x27; : &#x27;&#x27;) + &#x27;.&#x27;);
    }
    return data;
  } catch (err) {
    if (err.name === &#x27;AbortError&#x27;) throw new Error(&#x27;Hết thời gian chờ VPS. Lệnh CÓ THỂ đã khớp: xem lại Lệnh đang mở trước khi đặt lại.&#x27;);
    if (err instanceof TypeError) throw new Error(&#x27;Không gửi được tới VPS. Lệnh CÓ THỂ đã tới VPS: xem lại Lệnh đang mở trước khi đặt lại.&#x27;);
    throw err;
  } finally { clearTimeout(timer); }
}
function ordId() {

  return &#x27;xk&#x27; + a[0].toString(36) + a[1].toString(36);
}

function xkBarFrom(o) {
  if (!o || typeof o !== &#x27;object&#x27;) return null;
  const b = { t: Math.floor(Number(o.t)), o: Number(o.o), h: Number(o.h), l: Number(o.l), c: Number(o.c), v: Number(o.v) || 0 };
  if (![b.t, b.o, b.h, b.l, b.c].every(Number.isFinite)) return null;
  b.h = Math.max(b.h, b.o, b.c);
  b.l = Math.min(b.l, b.o, b.c);
  return b;
}
